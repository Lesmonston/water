//! Water desktop host. F2 ships with an unavailable backend; F3 provides the service.
#![forbid(unsafe_code)]

mod commands;
mod data;
mod picker;
mod policy;
mod wiring;

pub use commands::*;
pub use data::{DataDirectories, DataDirectoryError, prepare_data_directories};
pub use picker::{pick_folder, picker_failure};
pub use policy::{
    CAPABILITY, CONFIG, CSP, ConfigError, is_bundled_navigation, validate_capability,
    validate_config,
};
pub use wiring::*;

use serde::Serialize;
use serde_json::Value;
use std::{path::PathBuf, sync::Arc};
use tauri::{
    Manager, WebviewUrl, WebviewWindowBuilder,
    webview::{NewWindowResponse, PermissionResponse},
    window::{Effect, EffectsBuilder},
};

#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
pub struct IpcFailure {
    pub code: String,
    pub message_zh: String,
}

pub trait Backend: Send + Sync + 'static {
    fn call(&self, command: &str, args: Value) -> Result<Value, IpcFailure>;
}

pub struct UnavailableBackend;

impl Backend for UnavailableBackend {
    fn call(&self, _command: &str, _args: Value) -> Result<Value, IpcFailure> {
        Err(IpcFailure {
            code: "unavailable".into(),
            message_zh: "Water 服务暂时不可用。".into(),
        })
    }
}

/// Self-test stub: startup without Tauri setup, so every failure returns an error.
pub fn run<F>(make_backend: F) -> Result<(), StartupError>
where
    F: FnOnce(&HostContext) -> Result<Arc<dyn Backend>, StartupError>,
{
    validate_config(CONFIG).map_err(|_| StartupError::Failed)?;
    let app = tauri::Builder::default()
        .setup(|_| Ok(()))
        .invoke_handler(tauri::generate_handler![
            commands::app_state,
            commands::scan_start,
            commands::scan_cancel,
            commands::scan_progress,
            commands::scan_result,
            commands::group_detail,
            commands::clean_prepare,
            commands::clean_start,
            commands::clean_stop,
            commands::clean_progress,
            commands::clean_result,
            commands::trends,
            commands::history_list,
            commands::locations_get,
            commands::locations_set_category,
            commands::custom_root_add,
            commands::custom_root_remove,
        ])
        .build(tauri::generate_context!())
        .map_err(|_| StartupError::Failed)?;
    let override_dir = std::env::var_os("WATER_DATA_DIR").map(PathBuf::from);
    let fallback = match override_dir {
        Some(_) => PathBuf::new(),
        None => app
            .path()
            .local_data_dir()
            .map_err(|_| StartupError::Failed)?,
    };
    let directories = prepare_data_directories(override_dir.as_deref(), &fallback)
        .map_err(|_| StartupError::Failed)?;
    let _lock = acquire_instance_lock(&directories.data)?;
    let home_dir = app
        .path()
        .home_dir()
        .ok()
        .and_then(|p| p.to_str().map(str::to_owned))
        .ok_or(StartupError::Failed)?;
    let backend = make_backend(&HostContext {
        data_dir: directories.data.clone(),
        home_dir,
    })?;
    app.manage(backend);
    WebviewWindowBuilder::new(&app, "main", WebviewUrl::App("index.html".into()))
        .title("Water")
        .inner_size(1240.0, 820.0)
        .min_inner_size(1040.0, 720.0)
        .center()
        .transparent(true)
        .decorations(true)
        .effects(EffectsBuilder::new().effect(Effect::Mica).build())
        .data_directory(directories.webview)
        .devtools(false)
        .disable_drag_drop_handler()
        .on_navigation(|url| is_bundled_navigation(url.as_str()))
        .on_new_window(|_url, _features| NewWindowResponse::Deny)
        .on_download(|_webview, _event| false)
        .on_permission_request(|_webview, _kind| PermissionResponse::Deny)
        .build()
        .map_err(|_| StartupError::Failed)?;
    app.run(|_, _| {});
    Ok(())
}
