#![forbid(unsafe_code)]

use std::{
    collections::hash_map::RandomState,
    hash::{BuildHasher, Hasher},
    time::{Duration, SystemTime, UNIX_EPOCH},
};

pub(crate) fn id_ok(s: &str) -> bool {
    (1..=256).contains(&s.len())
        && s.as_bytes()[0].is_ascii_alphanumeric()
        && s.bytes()
            .all(|b| b.is_ascii_alphanumeric() || b":._-".contains(&b))
}

pub(crate) fn contains(a: &str, b: &str) -> bool {
    let a = a.trim_end_matches('\\');
    b.eq_ignore_ascii_case(a)
        || (b.len() > a.len()
            && b.get(..a.len()).is_some_and(|v| v.eq_ignore_ascii_case(a))
            && b.as_bytes()[a.len()] == b'\\')
}

pub(crate) fn drive(s: &str) -> String {
    s.get(..3).unwrap_or("").to_ascii_uppercase()
}
pub(crate) fn bytes(v: &serde_json::Value) -> u64 {
    v.as_str().and_then(|s| s.parse().ok()).unwrap_or(0)
}
pub(crate) fn number(s: &str) -> u64 {
    s.parse().unwrap_or(0)
}

pub(crate) fn random_hex() -> String {
    // Each keyed hasher obtains fresh standard-library random state; no clock/PID token seeds.
    let mut out = String::new();
    for domain in [0_u8, 1] {
        let mut h = RandomState::new().build_hasher();
        h.write_u8(domain);
        out.push_str(&format!("{:016x}", h.finish()));
    }
    out
}

pub(crate) fn new_id(prefix: &str, at: SystemTime) -> String {
    format!(
        "{prefix}-{}-{}",
        timestamp(at).replace(['-', ':'], ""),
        &random_hex()[..4]
    )
}

pub(crate) fn parts(at: SystemTime) -> [u16; 6] {
    let secs = at.duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
    let z = (secs / 86400) as i64 + 719468;
    let era = z.div_euclid(146097);
    let doe = z - era * 146097;
    let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = doy - (153 * mp + 2) / 5 + 1;
    let m = mp + if mp < 10 { 3 } else { -9 };
    let y = era * 400 + yoe + i64::from(m <= 2);
    [
        y as u16,
        m as u16,
        d as u16,
        ((secs / 3600) % 24) as u16,
        ((secs / 60) % 60) as u16,
        (secs % 60) as u16,
    ]
}

pub(crate) fn timestamp(at: SystemTime) -> String {
    let [y, m, d, h, n, s] = parts(at);
    format!("{y:04}-{m:02}-{d:02}T{h:02}:{n:02}:{s:02}Z")
}

pub(crate) fn parse_time(s: &str) -> Option<SystemTime> {
    if s.len() != 20 || !s.is_ascii() {
        return None;
    }
    let y = s[0..4].parse::<i64>().ok()?;
    let m = s[5..7].parse::<i64>().ok()?;
    let d = s[8..10].parse::<i64>().ok()?;
    let h = s[11..13].parse::<u64>().ok()?;
    let n = s[14..16].parse::<u64>().ok()?;
    let sec = s[17..19].parse::<u64>().ok()?;
    if !(1970..=9999).contains(&y)
        || !(1..=12).contains(&m)
        || !(1..=31).contains(&d)
        || h > 23
        || n > 59
        || sec > 59
    {
        return None;
    }
    let y = y - i64::from(m <= 2);
    let era = y.div_euclid(400);
    let yo = y - era * 400;
    let mp = m + if m > 2 { -3 } else { 9 };
    let days = era * 146097 + yo * 365 + yo / 4 - yo / 100 + (153 * mp + 2) / 5 + d - 1 - 719468;
    let at = UNIX_EPOCH.checked_add(Duration::from_secs(
        u64::try_from(days).ok()? * 86400 + h * 3600 + n * 60 + sec,
    ))?;
    (timestamp(at) == s).then_some(at)
}

pub(crate) fn digest(input: &str) -> String {
    // SHA-256 over the token, avoiding any persisted plaintext or process-specific hash keys.
    const K: [u32; 64] = [
        0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4,
        0xab1c5ed5, 0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3, 0x72be5d74, 0x80deb1fe,
        0x9bdc06a7, 0xc19bf174, 0xe49b69c1, 0xefbe4786, 0x0fc19dc6, 0x240ca1cc, 0x2de92c6f,
        0x4a7484aa, 0x5cb0a9dc, 0x76f988da, 0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7,
        0xc6e00bf3, 0xd5a79147, 0x06ca6351, 0x14292967, 0x27b70a85, 0x2e1b2138, 0x4d2c6dfc,
        0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85, 0xa2bfe8a1, 0xa81a664b,
        0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070, 0x19a4c116,
        0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
        0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7,
        0xc67178f2,
    ];
    let mut data = input.as_bytes().to_vec();
    let bits = (data.len() as u64) * 8;
    data.push(0x80);
    while data.len() % 64 != 56 {
        data.push(0);
    }
    data.extend_from_slice(&bits.to_be_bytes());
    let mut hash: [u32; 8] = [
        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab,
        0x5be0cd19,
    ];
    for block in data.chunks_exact(64) {
        let mut w = [0u32; 64];
        for (i, b) in block.chunks_exact(4).enumerate() {
            w[i] = u32::from_be_bytes(b.try_into().expect("four bytes"));
        }
        for i in 16..64 {
            let a = w[i - 15];
            let b = w[i - 2];
            w[i] = w[i - 16]
                .wrapping_add(a.rotate_right(7) ^ a.rotate_right(18) ^ (a >> 3))
                .wrapping_add(w[i - 7])
                .wrapping_add(b.rotate_right(17) ^ b.rotate_right(19) ^ (b >> 10));
        }
        let [mut a, mut b, mut c, mut d, mut e, mut f, mut g, mut h] = hash;
        for i in 0..64 {
            let t = h
                .wrapping_add(e.rotate_right(6) ^ e.rotate_right(11) ^ e.rotate_right(25))
                .wrapping_add((e & f) ^ (!e & g))
                .wrapping_add(K[i])
                .wrapping_add(w[i]);
            let u = (a.rotate_right(2) ^ a.rotate_right(13) ^ a.rotate_right(22))
                .wrapping_add((a & b) ^ (a & c) ^ (b & c));
            h = g;
            g = f;
            f = e;
            e = d.wrapping_add(t);
            d = c;
            c = b;
            b = a;
            a = t.wrapping_add(u);
        }
        for (v, n) in hash.iter_mut().zip([a, b, c, d, e, f, g, h]) {
            *v = v.wrapping_add(n);
        }
    }
    hash.iter().map(|n| format!("{n:08x}")).collect()
}
