#![forbid(unsafe_code)]

use crate::{IpcError, util};
use serde_json::Value;
use std::{
    fs::{self, File, OpenOptions},
    io::{self, Write},
    os::windows::fs::MetadataExt,
    path::{Component, Path, PathBuf},
};

pub(crate) struct Store {
    pub root: PathBuf,
}

pub(crate) fn plain_directory(path: &Path) -> bool {
    if !path.is_absolute() {
        return false;
    }
    let mut current = PathBuf::new();
    for c in path.components() {
        if matches!(c, Component::ParentDir | Component::CurDir) {
            return false;
        }
        current.push(c);
        if current.as_os_str().len() <= 2 {
            continue;
        }
        let Ok(m) = fs::symlink_metadata(&current) else {
            return false;
        };
        if !m.is_dir() || m.file_attributes() & 0x400 != 0 {
            return false;
        }
    }
    true
}

fn unavailable<T>(_: T) -> IpcError {
    IpcError::new("unavailable")
}

impl Store {
    pub fn new(root: PathBuf) -> Result<Self, IpcError> {
        if !root.is_absolute()
            || root
                .components()
                .any(|c| matches!(c, Component::CurDir | Component::ParentDir))
        {
            return Err(unavailable(()));
        }
        // Check existing ancestors before creating the application's directory.
        let mut nearest = root.as_path();
        while !nearest.exists() {
            nearest = nearest.parent().ok_or_else(|| unavailable(()))?;
        }
        if !plain_directory(nearest) {
            return Err(unavailable(()));
        }
        fs::create_dir_all(&root).map_err(unavailable)?;
        let store = Self { root };
        for dir in ["scans", "snapshots", "plans", "cleans", "journal"] {
            let path = store.root.join(dir);
            if path.exists() && !plain_directory(&path) {
                return Err(unavailable(()));
            }
            fs::create_dir_all(path).map_err(unavailable)?;
        }
        Ok(store)
    }

    fn path(&self, dir: &str, name: &str) -> Result<PathBuf, IpcError> {
        if !["", "scans", "snapshots", "plans", "cleans", "journal"].contains(&dir)
            || name.is_empty()
            || !name
                .bytes()
                .all(|b| b.is_ascii_alphanumeric() || b"-_.".contains(&b))
        {
            return Err(unavailable(()));
        }
        let parent = self.root.join(dir);
        if !plain_directory(&parent) {
            return Err(unavailable(()));
        }
        let path = parent.join(name);
        if let Ok(m) = fs::symlink_metadata(&path)
            && (!m.is_file() || m.file_attributes() & 0x400 != 0)
        {
            return Err(unavailable(()));
        }
        Ok(path)
    }

    pub fn read(&self, dir: &str, name: &str) -> Option<Value> {
        serde_json::from_slice(&fs::read(self.path(dir, name).ok()?).ok()?).ok()
    }

    pub fn write(&self, dir: &str, name: &str, value: &Value) -> Result<(), IpcError> {
        let dest = self.path(dir, name)?;
        let tmp = self.path(dir, &format!(".{name}.{}.tmp", util::random_hex()))?;
        let mut file = OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(&tmp)
            .map_err(unavailable)?;
        serde_json::to_writer(&mut file, value).map_err(unavailable)?;
        file.write_all(b"\n").map_err(unavailable)?;
        file.sync_all().map_err(unavailable)?;
        drop(file);
        // On Windows rename replaces the destination; there is no remove-then-write gap.
        self.path(dir, name)?;
        fs::rename(tmp, dest).map_err(unavailable)
    }

    pub fn records(&self, dir: &str) -> Vec<(String, Value)> {
        let Ok(sentinel) = self.path(dir, "list.json") else {
            return vec![];
        };
        let Ok(entries) = fs::read_dir(sentinel.parent().expect("store parent")) else {
            return vec![];
        };
        entries
            .filter_map(Result::ok)
            .filter_map(|entry| {
                let name = entry.file_name().into_string().ok()?;
                let id = name.strip_suffix(".json")?;
                let prefix = match dir {
                    "scans" | "snapshots" => "scan-",
                    "plans" => "plan-",
                    "cleans" => "clean-",
                    _ => return None,
                };
                if !id.starts_with(prefix) || !util::id_ok(id) {
                    return None;
                }
                Some((id.to_owned(), self.read(dir, &name)?))
            })
            .collect()
    }

    pub fn prune(&self, dir: &str, keep: &[String]) -> Result<(), IpcError> {
        // Only our flat JSON record names; neither recursive traversal nor journal deletion.
        for (id, _) in self.records(dir) {
            if !keep.contains(&id) {
                fs::remove_file(self.path(dir, &format!("{id}.json"))?).map_err(unavailable)?;
            }
        }
        Ok(())
    }

    pub fn journal(&self, id: &str) -> Result<DurableFile, IpcError> {
        let file = OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(self.path("journal", &format!("{id}.jsonl"))?)
            .map_err(unavailable)?;
        Ok(DurableFile(file))
    }
}

pub(crate) struct DurableFile(File);
impl Write for DurableFile {
    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
        self.0.write(buf)
    }
    fn flush(&mut self) -> io::Result<()> {
        self.0.flush()?;
        self.0.sync_all()
    }
}
