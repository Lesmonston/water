//! One physical DFS feeds every logical target that covers a directory.
//! Listings live only on the DFS stack; cluster proposals retain measurements,
//! never the listings of completed subtrees.
use super::*;

pub(super) struct ClusterSearch {
    pub parent: String,
    pub category: Category,
    pub discovery: bool,
    pub group: String,
    pub members: Vec<(Child, Measured)>,
}

enum Purpose {
    Target(usize),
    Candidate,
}

struct Measurement {
    purpose: Purpose,
    child: Child,
    measured: Measured,
    identities: HashSet<(u32, [u8; 16])>,
    parent: Option<Arc<Vec<ParentEntry>>>,
    issues: Vec<&'static str>,
    declared: bool,
}

struct Search {
    parent: String,
    category: Category,
    discovery: bool,
    group: String,
    candidates: Vec<usize>,
}

enum Frame {
    Enter {
        child: Child,
        closed: Vec<usize>,
        measuring: Vec<usize>,
        starts: Vec<usize>,
        parent: Option<Arc<Vec<ParentEntry>>>,
    },
    Exit {
        path: String,
        starts: Vec<usize>,
        search: Option<Search>,
        group: String,
    },
}

fn relative<'a>(root: &str, path: &'a str) -> Option<Vec<&'a str>> {
    if path_eq(root, path) {
        Some(vec![])
    } else if is_proper_ancestor(root, path) {
        Some(
            path[root.trim_end_matches('\\').len() + 1..]
                .split('\\')
                .collect(),
        )
    } else {
        None
    }
}

fn start(
    pending: &mut HashMap<usize, Measurement>,
    serial: &mut usize,
    child: &Child,
    purpose: Purpose,
    parent: Option<Arc<Vec<ParentEntry>>>,
) -> usize {
    let id = *serial;
    *serial += 1;
    let mut measured = Measured::empty(child.mtime, child.creation);
    if child.kind == Kind::Dir {
        observe_name(&mut measured, child, &child.path);
    }
    pending.insert(
        id,
        Measurement {
            purpose,
            child: child.clone(),
            measured,
            identities: HashSet::new(),
            parent,
            issues: vec![],
            declared: false,
        },
    );
    id
}

fn owner(row: &Measurement, plan: &ScanPlan, fallbacks: &[Option<usize>]) -> Option<usize> {
    let Purpose::Target(index) = row.purpose else {
        return None;
    };
    let target = &plan.targets[index];
    match &target.match_spec {
        Match::Marker {
            conditions, marker, ..
        } => {
            if conditions_match(
                &row.child.path,
                conditions,
                row.parent.as_deref().map(Vec::as_slice),
                Some(marker),
                Some(row.measured.bytes),
            ) {
                Some(index)
            } else {
                fallbacks[index]
            }
        }
        Match::Pattern { conditions, .. } => conditions_match(
            &row.child.path,
            conditions,
            row.parent.as_deref().map(Vec::as_slice),
            None,
            Some(row.measured.bytes),
        )
        .then_some(index),
        Match::Discovery { .. } => Some(index),
    }
}

fn finish(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    row: Measurement,
    plan: &ScanPlan,
    fallbacks: &[Option<usize>],
    suppressed: &mut Vec<String>,
) {
    let Purpose::Target(primary) = row.purpose else {
        return;
    };
    let resolved = owner(&row, plan, fallbacks);
    let target = &plan.targets[resolved.unwrap_or(primary)];
    for code in &row.issues {
        engine.add_issue(code, &target.group_id);
    }
    if engine.stopped(&target.group_id) {
        return;
    }
    let Some(index) = resolved else {
        return;
    };
    if matches!(target.match_spec, Match::Discovery { .. })
        && hits.iter().any(|h| {
            !matches!(
                plan.targets[h.target_index].match_spec,
                Match::Discovery { .. }
            ) && (path_eq(&row.child.path, &h.path) || is_proper_ancestor(&row.child.path, &h.path))
        })
    {
        return;
    }
    if row.child.kind == Kind::Dir
        && matches!(plan.targets[primary].match_spec, Match::Marker { .. })
    {
        suppressed.push(row.child.path.clone());
    }
    push_measured_hit(
        engine,
        hits,
        target,
        index,
        &row.child.path,
        row.child.kind,
        row.measured,
    );
}

fn potential_children(children: &[Child]) -> HashSet<String> {
    let mut counts = HashMap::<(bool, String), usize>::new();
    for c in children {
        if c.kind == Kind::Reparse || c.name.eq_ignore_ascii_case(".git") {
            continue;
        }
        if let Some(t) = name_template(&c.name, c.kind == Kind::File)
            && !(c.kind == Kind::File && t.contains("<d>"))
        {
            *counts.entry((c.kind == Kind::File, t)).or_default() += 1;
        }
    }
    let templates: Vec<_> = counts
        .into_iter()
        .filter(|(_, n)| *n >= 10)
        .map(|(t, _)| t)
        .collect();
    children
        .iter()
        .filter(|c| {
            templates.iter().any(|(file, t)| {
                (*file
                    && c.kind == Kind::File
                    && (name_template(&c.name, true).as_ref() == Some(t) || hash_extra(t, &c.name)))
                    || (!file
                        && c.kind == Kind::Dir
                        && name_template(&c.name, false).as_ref() == Some(t))
            })
        })
        .map(|c| c.path.to_ascii_lowercase())
        .collect()
}

pub(super) fn walk(
    engine: &mut Engine<'_>,
    plan: &ScanPlan,
    fallbacks: &[Option<usize>],
) -> (Vec<Hit>, Vec<ClusterSearch>) {
    let mut hits = vec![];
    let mut searches = vec![];
    let mut suppressed = vec![];
    let mut roots = BTreeMap::<String, (String, Vec<usize>)>::new();
    for (i, target) in plan.targets.iter().enumerate() {
        if matches!(
            target.match_spec,
            Match::Marker {
                fallback_for: Some(_),
                ..
            }
        ) {
            continue;
        }
        let root = match &target.match_spec {
            Match::Pattern { base, .. } => base,
            Match::Marker { root, .. } | Match::Discovery { root } => root,
        };
        if ignored_root(root) || file_name(root).eq_ignore_ascii_case(".git") {
            continue;
        }
        roots
            .entry(root.to_ascii_lowercase())
            .or_insert_with(|| (root.clone(), vec![]))
            .1
            .push(i);
    }
    let mut roots = roots.into_values().collect::<Vec<_>>();
    roots.sort_by_key(|(path, ids)| (path.matches('\\').count(), ids[0]));
    let mut visited_roots = HashSet::new();
    let mut pending = HashMap::<usize, Measurement>::new();
    let mut serial = 0;
    for (root, indices) in &roots {
        if visited_roots.contains(&root.to_ascii_lowercase()) {
            continue;
        }
        let group = &plan.targets[indices[0]].group_id;
        if engine.stopped(group) {
            break;
        }
        let child = engine.child(root);
        if child.as_ref().is_none_or(|c| c.kind != Kind::Dir) {
            let code = if child.as_ref().is_some_and(|c| c.kind == Kind::Reparse) {
                "base_reparse"
            } else {
                "base_missing"
            };
            if let Some(c) = &child {
                engine.index_junction(c);
            }
            for i in indices {
                engine.add_issue(code, &plan.targets[*i].group_id);
                if let Some(f) = fallbacks[*i] {
                    engine.add_issue(code, &plan.targets[f].group_id);
                }
            }
            continue;
        }
        let mut stack = vec![Frame::Enter {
            child: child.unwrap(),
            closed: vec![],
            measuring: vec![],
            starts: vec![],
            parent: None,
        }];
        while let Some(frame) = stack.pop() {
            match frame {
                Frame::Exit {
                    path,
                    starts,
                    search,
                    group,
                } => {
                    for id in starts {
                        if let Some(row) = pending.remove(&id) {
                            finish(engine, &mut hits, row, plan, fallbacks, &mut suppressed);
                        }
                    }
                    if let Some(search) = search {
                        let members = search
                            .candidates
                            .into_iter()
                            .filter_map(|id| pending.remove(&id))
                            .filter(|row| !row.declared)
                            .map(|row| (row.child, row.measured))
                            .collect::<Vec<_>>();
                        if !engine.stopped(&search.group) && !members.is_empty() {
                            searches.push(ClusterSearch {
                                parent: search.parent,
                                category: search.category,
                                discovery: search.discovery,
                                group: search.group,
                                members,
                            });
                        }
                    }
                    engine.cache.remove(&path.to_ascii_lowercase());
                    engine.emit(
                        &group,
                        plan.targets
                            .iter()
                            .find(|t| t.group_id == group)
                            .map_or(Category::Custom, |t| t.category),
                        false,
                    );
                }
                Frame::Enter {
                    child: dir,
                    mut closed,
                    mut measuring,
                    mut starts,
                    parent,
                } => {
                    let mut markers = vec![];
                    let mut patterns = vec![];
                    let mut discoveries = vec![];
                    let mut custom = None;
                    for (index, target) in plan.targets.iter().enumerate() {
                        match &target.match_spec {
                            Match::Marker {
                                root,
                                max_depth,
                                fallback_for: None,
                                ..
                            } => {
                                if let Some(parts) = relative(root, &dir.path)
                                    && parts.iter().all(|p| !p.eq_ignore_ascii_case(".git"))
                                    && parts.len() <= *max_depth as usize
                                {
                                    if target.category == Category::Custom {
                                        custom.get_or_insert((index, parts.len()));
                                    }
                                    if !closed.contains(&index) {
                                        markers.push((index, parts.len()));
                                    }
                                }
                            }
                            Match::Pattern { base, segments, .. } => {
                                if let Some(parts) = relative(base, &dir.path)
                                    && parts.len() < segments.len()
                                    && parts.iter().zip(segments).all(|(p, s)| match_segment(s, p))
                                {
                                    patterns.push((index, parts.len()));
                                }
                            }
                            Match::Discovery { root } if path_eq(root, &dir.path) => {
                                discoveries.push(index)
                            }
                            _ => {}
                        }
                    }
                    for (root, _) in &roots {
                        if path_eq(root, &dir.path) {
                            visited_roots.insert(root.to_ascii_lowercase());
                        }
                    }
                    let primary = patterns
                        .first()
                        .map(|(i, _)| *i)
                        .or_else(|| markers.first().map(|(i, _)| *i))
                        .or_else(|| discoveries.first().copied())
                        .or_else(|| custom.map(|(i, _)| i))
                        .or_else(|| {
                            measuring.iter().find_map(|id| match pending[id].purpose {
                                Purpose::Target(i) => Some(i),
                                _ => None,
                            })
                        })
                        .unwrap_or(indices[0]);
                    let group = plan.targets[primary].group_id.clone();
                    if engine.stopped(&group) {
                        break;
                    }
                    let needs_listing = !measuring.is_empty()
                        || !patterns.is_empty()
                        || !discoveries.is_empty()
                        || markers.iter().any(|(i, depth)| {
                            let Match::Marker {
                                marker, max_depth, ..
                            } = &plan.targets[*i].match_spec
                            else {
                                return false;
                            };
                            *depth < *max_depth as usize
                                || !matches!(
                                    marker,
                                    Marker::DirName { .. }
                                        | Marker::DirNamePattern { .. }
                                        | Marker::FileName { .. }
                                )
                                || marker_matches(&dir.path, marker, &[])
                        });
                    if !needs_listing {
                        continue;
                    }
                    let listing = engine.listing(&dir.path, &group);
                    let declared = listing.children.iter().any(|c| {
                        c.kind == Kind::File && c.name.eq_ignore_ascii_case(".water-cleanup.json")
                    });
                    for id in &measuring {
                        let row = pending.get_mut(id).unwrap();
                        if path_eq(&row.child.path, &dir.path) {
                            row.declared = declared;
                        }
                        row.issues.extend(&listing.issues);
                    }
                    let mut descend_marker = false;
                    for (index, depth) in markers {
                        let target = &plan.targets[index];
                        let Match::Marker {
                            marker, max_depth, ..
                        } = &target.match_spec
                        else {
                            unreachable!()
                        };
                        if matches!(marker, Marker::FileName { .. }) {
                            if depth < *max_depth as usize {
                                engine.listing_issues(&listing, &target.group_id);
                                descend_marker = true;
                            }
                        } else if marker_matches(&dir.path, marker, &listing.children) {
                            let id = start(
                                &mut pending,
                                &mut serial,
                                &dir,
                                Purpose::Target(index),
                                parent.clone(),
                            );
                            let row = pending.get_mut(&id).unwrap();
                            row.declared = declared;
                            row.issues.extend(&listing.issues);
                            if !matches!(
                                marker,
                                Marker::DirName { .. } | Marker::DirNamePattern { .. }
                            ) {
                                row.issues.extend(&listing.issues);
                            }
                            starts.push(id);
                            measuring.push(id);
                            closed.push(index);
                        } else {
                            if !matches!(
                                marker,
                                Marker::DirName { .. } | Marker::DirNamePattern { .. }
                            ) {
                                engine.listing_issues(&listing, &target.group_id);
                                if depth < *max_depth as usize {
                                    engine.listing_issues(&listing, &target.group_id);
                                }
                            }
                            if depth < *max_depth as usize {
                                descend_marker = true;
                            }
                        }
                    }
                    let blocked = measuring.iter().any(|id| {
                        let row = &pending[id];
                        let Purpose::Target(i) = row.purpose else {
                            return false;
                        };
                        if let Match::Marker { conditions, .. } = &plan.targets[i].match_spec {
                            // Length-dependent ownership is resolved at exit; keep
                            // provisional extras until the actual hit is known.
                            !conditions
                                .iter()
                                .any(|c| matches!(c, Condition::SiblingWithoutSuffix { .. }))
                                && owner(row, plan, fallbacks).is_some()
                        } else {
                            false
                        }
                    });
                    let custom_search = custom.filter(|_| !blocked);
                    if let Some((_, depth)) = custom_search
                        && engine.extras_options.declarations
                        && depth <= 4
                        && declared
                    {
                        for c in &listing.children {
                            if c.kind == Kind::File
                                && c.name.eq_ignore_ascii_case(".water-cleanup.json")
                            {
                                engine.extras.declarations.push(c.path.clone());
                            }
                        }
                    }
                    let potential = if engine.extras_options.leak_clusters
                        && (!discoveries.is_empty() || custom_search.is_some())
                    {
                        potential_children(&listing.children)
                    } else {
                        HashSet::new()
                    };
                    let mut search = (!potential.is_empty()).then(|| {
                        let i = discoveries
                            .first()
                            .copied()
                            .unwrap_or_else(|| custom_search.unwrap().0);
                        Search {
                            parent: dir.path.clone(),
                            category: plan.targets[i].category,
                            discovery: !discoveries.is_empty(),
                            group: plan.targets[i].group_id.clone(),
                            candidates: vec![],
                        }
                    });
                    let parent_entries = Arc::new(
                        listing
                            .children
                            .iter()
                            .map(|c| ParentEntry {
                                name: c.name.clone(),
                                kind: c.kind,
                                len: c.len,
                            })
                            .collect::<Vec<_>>(),
                    );
                    for (index, depth) in &patterns {
                        let target = &plan.targets[*index];
                        let Match::Pattern { segments, .. } = &target.match_spec else {
                            unreachable!()
                        };
                        if !listing
                            .children
                            .iter()
                            .any(|c| match_segment(&segments[*depth], &c.name))
                        {
                            engine.emit(&target.group_id, target.category, true);
                        }
                    }
                    let mut child_frames = vec![];
                    for child in &listing.children {
                        if engine.stopped(&group) {
                            break;
                        }
                        for id in &measuring {
                            let row = pending.get_mut(id).unwrap();
                            accumulate(
                                &mut row.measured,
                                child,
                                &row.child.path,
                                &mut row.identities,
                            );
                        }
                        let mut child_starts = vec![];
                        let mut needs_pattern = false;
                        for (index, depth) in &patterns {
                            let target = &plan.targets[*index];
                            let Match::Pattern { segments, .. } = &target.match_spec else {
                                unreachable!()
                            };
                            if child.name.eq_ignore_ascii_case(".git")
                                || !match_segment(&segments[*depth], &child.name)
                            {
                                continue;
                            }
                            if child.kind == Kind::Reparse {
                                engine.add_issue("reparse_skipped", &target.group_id);
                                continue;
                            }
                            if depth + 1 == segments.len() {
                                if accept_type(target.item_type, child.kind) {
                                    child_starts.push(start(
                                        &mut pending,
                                        &mut serial,
                                        child,
                                        Purpose::Target(*index),
                                        Some(parent_entries.clone()),
                                    ));
                                }
                            } else if child.kind == Kind::Dir {
                                needs_pattern = true;
                            }
                        }
                        for (index, target) in plan.targets.iter().enumerate() {
                            if closed.contains(&index) {
                                continue;
                            }
                            if let Match::Marker {
                                root,
                                marker: Marker::FileName { patterns },
                                max_depth,
                                fallback_for: None,
                                ..
                            } = &target.match_spec
                                && let Some(parts) = relative(root, &dir.path)
                                && parts.len() < *max_depth as usize
                                && parts.iter().all(|p| !p.eq_ignore_ascii_case(".git"))
                                && child.kind == Kind::File
                                && !child.name.eq_ignore_ascii_case(".git")
                                && patterns.iter().any(|p| match_segment(p, &child.name))
                            {
                                child_starts.push(start(
                                    &mut pending,
                                    &mut serial,
                                    child,
                                    Purpose::Target(index),
                                    Some(parent_entries.clone()),
                                ));
                            }
                        }
                        for index in &discoveries {
                            let target = &plan.targets[*index];
                            if child.kind == Kind::Reparse {
                                engine.add_issue("reparse_skipped", &target.group_id);
                            } else if !child.name.eq_ignore_ascii_case(".git")
                                && accept_type(target.item_type, child.kind)
                            {
                                child_starts.push(start(
                                    &mut pending,
                                    &mut serial,
                                    child,
                                    Purpose::Target(*index),
                                    Some(parent_entries.clone()),
                                ));
                            }
                        }
                        if potential.contains(&child.path.to_ascii_lowercase()) {
                            let id = start(
                                &mut pending,
                                &mut serial,
                                child,
                                Purpose::Candidate,
                                Some(parent_entries.clone()),
                            );
                            search.as_mut().unwrap().candidates.push(id);
                            child_starts.push(id);
                        }
                        if child.kind == Kind::File {
                            for id in child_starts {
                                let row = pending.get_mut(&id).unwrap();
                                accumulate(
                                    &mut row.measured,
                                    child,
                                    &child.path,
                                    &mut row.identities,
                                );
                                if matches!(row.purpose, Purpose::Target(_)) {
                                    let row = pending.remove(&id).unwrap();
                                    finish(
                                        engine,
                                        &mut hits,
                                        row,
                                        plan,
                                        fallbacks,
                                        &mut suppressed,
                                    );
                                }
                            }
                        } else if child.kind == Kind::Dir {
                            let next = !child.name.eq_ignore_ascii_case(".git") && (descend_marker || needs_pattern
                                || custom_search.is_some_and(|(i, depth)| matches!(&plan.targets[i].match_spec, Match::Marker { max_depth, .. } if depth < *max_depth as usize)));
                            let nested_root =
                                roots.iter().any(|(root, _)| path_eq(root, &child.path));
                            if next
                                || !measuring.is_empty()
                                || !child_starts.is_empty()
                                || nested_root
                            {
                                let mut child_measuring = measuring.clone();
                                child_measuring.extend(&child_starts);
                                let target_starts = child_starts
                                    .into_iter()
                                    .filter(|id| matches!(pending[id].purpose, Purpose::Target(_)))
                                    .collect();
                                child_frames.push(Frame::Enter {
                                    child: child.clone(),
                                    closed: closed.clone(),
                                    measuring: child_measuring,
                                    starts: target_starts,
                                    parent: Some(parent_entries.clone()),
                                });
                            }
                        } else if child.kind == Kind::Reparse {
                            for (i, target) in plan.targets.iter().enumerate() {
                                if !closed.contains(&i)
                                    && matches!(&target.match_spec, Match::Marker { root, fallback_for: None, max_depth, .. }
                                    if relative(root, &dir.path).is_some_and(|p| p.len() < *max_depth as usize))
                                {
                                    engine.add_issue("reparse_skipped", &target.group_id);
                                }
                            }
                        }
                    }
                    // Empty pattern results finish here too; observers may act at
                    // this boundary before a later logical target consumes names.
                    for (index, _) in &patterns {
                        engine.listing_issues(&listing, &plan.targets[*index].group_id);
                    }
                    for index in &discoveries {
                        engine.listing_issues(&listing, &plan.targets[*index].group_id);
                    }
                    if engine.stopped(&group) {
                        break;
                    }
                    stack.push(Frame::Exit {
                        path: dir.path,
                        starts,
                        search,
                        group,
                    });
                    stack.extend(child_frames.into_iter().rev());
                }
            }
        }
        if engine.status == "cancelled" {
            break;
        }
    }
    engine.cache.clear();
    // An actual marker hit suppresses searches inside it. Cluster ownership
    // never removes a declaration that was found by the search.
    engine
        .extras
        .declarations
        .retain(|path| !suppressed.iter().any(|p| is_proper_ancestor(p, path)));
    searches.retain(|s| {
        !suppressed
            .iter()
            .any(|p| path_eq(p, &s.parent) || is_proper_ancestor(p, &s.parent))
    });
    for target in &plan.targets {
        if !matches!(
            target.match_spec,
            Match::Marker {
                fallback_for: Some(_),
                ..
            }
        ) {
            engine.emit(&target.group_id, target.category, true);
        }
    }
    (hits, searches)
}
