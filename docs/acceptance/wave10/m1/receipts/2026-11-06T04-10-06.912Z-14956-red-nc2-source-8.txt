use crate::error::RulesError;
use crate::folders::{KnownFolders, validate_guard_path};
use crate::m35::{self, AgeBasis, KeepBy, PlanOptions};
use crate::rules::RuleSet;
use crate::types::{
    Category, Condition, InUse, ItemType, KNOWN_FOLDER_ORDER, Level, Locator, Marker, MatchSpec,
    as_str, bounded_u32, expect_keys, field, is_drive_root, object, parse_conditions,
    valid_abs_path, valid_created_at, valid_data_version, valid_root_id, valid_segment,
};
use serde::Serialize;
use serde_json::{Map, Value};
use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CustomRoot {
    pub root_id: String,
    pub path: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Target {
    pub target_id: String,
    pub group_id: String,
    pub rule_id: String,
    pub level: Level,
    pub category: Category,
    pub match_spec: MatchSpec,
    pub item_type: ItemType,
    pub min_age_hours: u32,
    pub keep_newest: u32,
    pub in_use: InUse,
    pub git_is_copy: bool,
    pub age_basis: AgeBasis,
    pub keep_by: KeepBy,
    pub unlocks_target_ids: Vec<String>,
    /// Project declaration conditions, serialized inside a pattern match.
    /// Kept on Target to preserve the existing MatchSpec variants and fields.
    pub pattern_conditions: Vec<Condition>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ScanPlan {
    pub plan_version: String,
    pub rules_data_version: String,
    pub created_at: String,
    pub targets: Vec<Target>,
    pub hard_guards: Vec<String>,
}

impl ScanPlan {
    pub fn from_json(input: &str) -> Result<Self, RulesError> {
        let value: Value =
            serde_json::from_str(input).map_err(|_| RulesError::new("RULES_INVALID_JSON", "/"))?;
        parse_plan(&value)
    }
}

#[derive(Serialize)]
struct PlanWire<'a> {
    plan_version: &'a str,
    rules_data_version: &'a str,
    created_at: &'a str,
    targets: Vec<TargetWire<'a>>,
    hard_guards: &'a [String],
}

#[derive(Serialize)]
struct TargetWire<'a> {
    target_id: &'a str,
    group_id: &'a str,
    rule_id: &'a str,
    level: &'a str,
    category: &'a str,
    #[serde(rename = "match")]
    match_spec: MatchWire<'a>,
    item_type: &'a str,
    min_age_hours: u32,
    keep_newest: u32,
    in_use: InUseWire<'a>,
    #[serde(skip_serializing_if = "std::ops::Not::not")]
    git_is_copy: bool,
    #[serde(skip_serializing_if = "Option::is_none")]
    age_basis: Option<&'a str>,
    #[serde(skip_serializing_if = "Option::is_none")]
    keep_by: Option<&'a str>,
    #[serde(skip_serializing_if = "<[String]>::is_empty")]
    unlocks_target_ids: &'a [String],
}

#[derive(Serialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
enum MatchWire<'a> {
    Pattern {
        base: &'a str,
        segments: &'a [String],
        #[serde(skip_serializing_if = "Vec::is_empty")]
        conditions: Vec<ConditionWire<'a>>,
    },
    Marker {
        root: &'a str,
        marker: MarkerWire<'a>,
        max_depth: u32,
        #[serde(skip_serializing_if = "Vec::is_empty")]
        conditions: Vec<ConditionWire<'a>>,
        #[serde(skip_serializing_if = "Option::is_none")]
        fallback_for: Option<&'a str>,
    },
    Discovery {
        root: &'a str,
    },
}

#[derive(Serialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
enum ConditionWire<'a> {
    ParentHasFile { name: &'a str },
    ParentHasOtherEntries { except: &'a [String] },
    NoAncestorSuffix { suffixes: &'a [String] },
    PathHasAny { names: &'a [String] },
    ParentHasOtherMatch,
    ParentHasFileMatching { pattern: &'a str },
    ParentLacksFileMatching { pattern: &'a str },
    SiblingWithoutSuffix { suffixes: &'a [String] },
}

#[derive(Serialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
enum MarkerWire<'a> {
    File {
        name: &'a str,
        min_bytes: u64,
        max_bytes: u64,
    },
    ChildrenAll {
        names: &'a [String],
    },
    DirName {
        names: &'a [String],
    },
    DirNamePattern {
        patterns: &'a [String],
    },
    FileName {
        patterns: &'a [String],
    },
}

#[derive(Serialize)]
struct InUseWire<'a> {
    mode: &'a str,
    processes: &'a [String],
    #[serde(skip_serializing_if = "Option::is_none")]
    root_levels: Option<u8>,
}

pub fn resolve_plan(
    rules: &RuleSet,
    folders: &KnownFolders,
    custom_roots: &[CustomRoot],
    created_at: &str,
) -> Result<ScanPlan, RulesError> {
    resolve_plan_with(
        rules,
        folders,
        custom_roots,
        created_at,
        &PlanOptions::default(),
    )
}

pub fn resolve_plan_with(
    rules: &RuleSet,
    folders: &KnownFolders,
    custom_roots: &[CustomRoot],
    created_at: &str,
    options: &PlanOptions,
) -> Result<ScanPlan, RulesError> {
    let mut enabled = HashSet::new();
    for (index, switch) in options.sessions.iter().enumerate() {
        if !m35::SESSION_TOOLS.iter().any(|(id, _)| *id == switch.tool)
            || !(1..=365).contains(&switch.min_age_days)
            || !enabled.insert(switch.tool.as_str())
        {
            return Err(RulesError::new(
                "RULES_INVALID_SESSIONS",
                format!("/sessions/{index}"),
            ));
        }
    }
    if !valid_created_at(created_at) {
        return Err(RulesError::new("RULES_INVALID_CREATED_AT", "/created_at"));
    }
    let mut seen_roots = HashSet::new();
    for (index, root) in custom_roots.iter().enumerate() {
        let path = format!("/custom_roots/{index}");
        if !valid_root_id(&root.root_id) {
            return Err(RulesError::new(
                "RULES_INVALID_ROOT_ID",
                format!("{path}/root_id"),
            ));
        }
        if !valid_abs_path(&root.path, true) {
            return Err(RulesError::new(
                "RULES_INVALID_PATH",
                format!("{path}/path"),
            ));
        }
        if !seen_roots.insert(root.root_id.clone()) {
            return Err(RulesError::new("RULES_DUPLICATE_ID", path));
        }
    }
    for (rule_index, rule) in rules.rules().iter().enumerate() {
        for (offset, locator) in rule.locate.iter().enumerate() {
            if let Locator::Discovery { base } = locator
                && is_drive_root(folders.get(*base))
            {
                return Err(RulesError::new(
                    "RULES_INVALID_LOCATE",
                    format!("/rules/{rule_index}/locate/{offset}"),
                ));
            }
        }
    }
    let mut targets = Vec::new();
    let normal = rules
        .rules()
        .iter()
        .filter(|rule| rule.session_tool.is_none());
    let sessions = rules.rules().iter().filter(|rule| {
        rule.session_tool
            .as_deref()
            .is_some_and(|tool| enabled.contains(tool))
    });
    for rule in normal.chain(sessions) {
        for (offset, locator) in rule.locate.iter().enumerate() {
            let index = offset + 1;
            match locator {
                Locator::Pattern { base, segments } => targets.push(make_target(
                    rule,
                    format!("{}#{index}", rule.rule_id),
                    rule.rule_id.clone(),
                    MatchSpec::Pattern {
                        base: folders.get(*base).to_string(),
                        segments: segments.clone(),
                    },
                )),
                Locator::Discovery { base } => targets.push(make_target(
                    rule,
                    format!("{}#{index}", rule.rule_id),
                    rule.rule_id.clone(),
                    MatchSpec::Discovery {
                        root: folders.get(*base).to_string(),
                    },
                )),
                Locator::Marker {
                    marker,
                    max_depth,
                    conditions,
                    fallback_for,
                } => {
                    for root in custom_roots {
                        let group_id = format!("custom:{}:{}", root.root_id, rule.rule_id);
                        targets.push(make_target(
                            rule,
                            format!("{group_id}#{index}"),
                            group_id,
                            MatchSpec::Marker {
                                root: root.path.clone(),
                                marker: marker.clone(),
                                max_depth: *max_depth,
                                conditions: conditions.clone(),
                                fallback_for: fallback_for
                                    .as_ref()
                                    .map(|id| format!("custom:{}:{id}#{index}", root.root_id)),
                            },
                        ));
                    }
                }
                Locator::CustomDiscovery { root_names } => {
                    for root in custom_roots {
                        let name = root.path.rsplit('\\').next().unwrap_or("");
                        if !root_names
                            .iter()
                            .any(|allowed| name.eq_ignore_ascii_case(allowed) || !allowed.is_empty())
                        {
                            continue;
                        }
                        let group_id = format!("custom:{}:{}", root.root_id, rule.rule_id);
                        targets.push(make_target(
                            rule,
                            format!("{group_id}#{index}"),
                            group_id,
                            MatchSpec::Discovery {
                                root: root.path.clone(),
                            },
                        ));
                    }
                }
            }
        }
    }
    let unlocks: Vec<Vec<String>> = targets
        .iter()
        .map(|target| {
            rules
                .rule(&target.rule_id)
                .and_then(|rule| rule.unlocks.as_ref())
                .map_or_else(Vec::new, |id| {
                    targets
                        .iter()
                        .filter(|t| &t.rule_id == id)
                        .map(|t| t.target_id.clone())
                        .collect()
                })
        })
        .collect();
    for (target, ids) in targets.iter_mut().zip(unlocks) {
        target.unlocks_target_ids = ids;
        if let Some(tool) = rules
            .rule(&target.rule_id)
            .and_then(|rule| rule.session_tool.as_deref())
        {
            let switch = options
                .sessions
                .iter()
                .find(|s| s.tool == tool)
                .ok_or_else(|| RulesError::new("RULES_INVALID_SESSIONS", "/sessions"))?;
            target.min_age_hours = u32::from(switch.min_age_days) * 24;
        }
    }
    Ok(ScanPlan {
        plan_version: "water.scan-plan.v1".to_string(),
        rules_data_version: rules.data_version.clone(),
        created_at: created_at.to_string(),
        targets,
        hard_guards: collect_hard_guards(folders, custom_roots),
    })
}

fn make_target(
    rule: &crate::rules::Rule,
    target_id: String,
    group_id: String,
    match_spec: MatchSpec,
) -> Target {
    Target {
        target_id,
        group_id,
        rule_id: rule.rule_id.clone(),
        level: rule.level,
        category: rule.category,
        match_spec,
        item_type: rule.item_type,
        min_age_hours: rule.min_age_hours,
        keep_newest: rule.keep_newest,
        in_use: rule.in_use.clone(),
        git_is_copy: rule.git_is_copy,
        age_basis: rule.age_basis,
        keep_by: rule.keep_by,
        unlocks_target_ids: vec![],
        pattern_conditions: vec![],
    }
}

fn collect_hard_guards(folders: &KnownFolders, custom_roots: &[CustomRoot]) -> Vec<String> {
    let mut guards = Vec::new();
    let mut seen = HashSet::new();
    let mut push = |path: String| {
        let key = path.to_ascii_lowercase();
        if seen.insert(key) {
            guards.push(path);
        }
    };
    for folder in KNOWN_FOLDER_ORDER {
        push(folders.get(folder).to_string());
    }
    for root in folders.drive_roots() {
        push(root);
    }
    for root in custom_roots {
        push(root.path.clone());
    }
    guards
}

pub fn plan_to_json(plan: &ScanPlan) -> String {
    let targets = plan
        .targets
        .iter()
        .map(|target| TargetWire {
            target_id: &target.target_id,
            group_id: &target.group_id,
            rule_id: &target.rule_id,
            level: target.level.as_str(),
            category: target.category.as_str(),
            match_spec: match_wire(&target.match_spec, &target.pattern_conditions),
            item_type: target.item_type.as_str(),
            min_age_hours: target.min_age_hours,
            keep_newest: target.keep_newest,
            in_use: InUseWire {
                mode: target.in_use.mode.as_str(),
                processes: &target.in_use.processes,
                root_levels: target.in_use.root_levels,
            },
            git_is_copy: target.git_is_copy,
            age_basis: (target.age_basis != AgeBasis::Modified)
                .then_some(target.age_basis.as_str()),
            keep_by: (target.keep_by != KeepBy::Activity).then_some(target.keep_by.as_str()),
            unlocks_target_ids: &target.unlocks_target_ids,
        })
        .collect();
    let wire = PlanWire {
        plan_version: &plan.plan_version,
        rules_data_version: &plan.rules_data_version,
        created_at: &plan.created_at,
        targets,
        hard_guards: &plan.hard_guards,
    };
    serde_json::to_string(&wire).expect("scan plan serialization")
}

fn match_wire<'a>(spec: &'a MatchSpec, pattern_conditions: &'a [Condition]) -> MatchWire<'a> {
    match spec {
        MatchSpec::Pattern { base, segments } => MatchWire::Pattern {
            base,
            segments,
            conditions: pattern_conditions.iter().map(condition_wire).collect(),
        },
        MatchSpec::Marker {
            root,
            marker,
            max_depth,
            conditions,
            fallback_for,
        } => MatchWire::Marker {
            root,
            marker: marker_wire(marker),
            max_depth: *max_depth,
            conditions: conditions.iter().map(condition_wire).collect(),
            fallback_for: fallback_for.as_deref(),
        },
        MatchSpec::Discovery { root } => MatchWire::Discovery { root },
    }
}

fn condition_wire(condition: &Condition) -> ConditionWire<'_> {
    match condition {
        Condition::ParentHasFile { name } => ConditionWire::ParentHasFile { name },
        Condition::ParentHasOtherEntries { except } => {
            ConditionWire::ParentHasOtherEntries { except }
        }
        Condition::NoAncestorSuffix { suffixes } => ConditionWire::NoAncestorSuffix { suffixes },
        Condition::PathHasAny { names } => ConditionWire::PathHasAny { names },
        Condition::ParentHasOtherMatch => ConditionWire::ParentHasOtherMatch,
        Condition::ParentHasFileMatching { pattern } => {
            ConditionWire::ParentHasFileMatching { pattern }
        }
        Condition::ParentLacksFileMatching { pattern } => {
            ConditionWire::ParentLacksFileMatching { pattern }
        }
        Condition::SiblingWithoutSuffix { suffixes } => {
            ConditionWire::SiblingWithoutSuffix { suffixes }
        }
    }
}

fn marker_wire(marker: &Marker) -> MarkerWire<'_> {
    match marker {
        Marker::File {
            name,
            min_bytes,
            max_bytes,
        } => MarkerWire::File {
            name,
            min_bytes: *min_bytes,
            max_bytes: *max_bytes,
        },
        Marker::ChildrenAll { names } => MarkerWire::ChildrenAll { names },
        Marker::DirName { names } => MarkerWire::DirName { names },
        Marker::DirNamePattern { patterns } => MarkerWire::DirNamePattern { patterns },
        Marker::FileName { patterns } => MarkerWire::FileName { patterns },
    }
}

fn parse_plan(value: &Value) -> Result<ScanPlan, RulesError> {
    let map = object(value, "/")?;
    expect_keys(
        map,
        &[
            "plan_version",
            "rules_data_version",
            "created_at",
            "targets",
            "hard_guards",
        ],
        "/",
    )?;
    if as_str(field(map, "plan_version", "/")?, "/plan_version")? != "water.scan-plan.v1" {
        return Err(RulesError::new("RULES_INVALID_VERSION", "/plan_version"));
    }
    let rules_data_version = as_str(
        field(map, "rules_data_version", "/")?,
        "/rules_data_version",
    )?;
    if !valid_data_version(rules_data_version) {
        return Err(RulesError::new(
            "RULES_INVALID_DATA_VERSION",
            "/rules_data_version",
        ));
    }
    let created_at = as_str(field(map, "created_at", "/")?, "/created_at")?;
    if !valid_created_at(created_at) {
        return Err(RulesError::new("RULES_INVALID_CREATED_AT", "/created_at"));
    }
    let targets = parse_targets(field(map, "targets", "/")?, "/targets")?;
    let hard_guards = parse_hard_guards(field(map, "hard_guards", "/")?, "/hard_guards")?;
    Ok(ScanPlan {
        plan_version: "water.scan-plan.v1".to_string(),
        rules_data_version: rules_data_version.to_string(),
        created_at: created_at.to_string(),
        targets,
        hard_guards,
    })
}

fn parse_targets(value: &Value, path: &str) -> Result<Vec<Target>, RulesError> {
    let items = value
        .as_array()
        .ok_or_else(|| RulesError::new("RULES_INVALID_SHAPE", path))?;
    let mut targets = Vec::with_capacity(items.len());
    for (index, item) in items.iter().enumerate() {
        targets.push(parse_target(item, &format!("{path}/{index}"))?);
    }
    check_plan_fallbacks(&targets)?;
    for (index, target) in targets.iter().enumerate() {
        let mut seen = HashSet::new();
        for id in &target.unlocks_target_ids {
            if !target.level.is_cleanable()
                || id == &target.target_id
                || !seen.insert(id)
                || !targets
                    .iter()
                    .any(|other| &other.target_id == id && other.level == Level::Protected)
            {
                return Err(RulesError::new(
                    "RULES_INVALID_UNLOCKS",
                    format!("/targets/{index}/unlocks_target_ids"),
                ));
            }
        }
    }
    Ok(targets)
}

fn check_plan_fallbacks(targets: &[Target]) -> Result<(), RulesError> {
    let mut seen = HashSet::new();
    for (index, target) in targets.iter().enumerate() {
        let MatchSpec::Marker {
            root,
            marker,
            max_depth,
            conditions,
            fallback_for: Some(id),
        } = &target.match_spec
        else {
            continue;
        };
        let invalid = || {
            RulesError::new(
                "RULES_INVALID_FALLBACK",
                format!("/targets/{index}/match/fallback_for"),
            )
        };
        if !conditions.is_empty() || id == &target.target_id || !seen.insert(id) {
            return Err(invalid());
        }
        let main = targets
            .iter()
            .find(|candidate| &candidate.target_id == id)
            .ok_or_else(invalid)?;
        let MatchSpec::Marker {
            root: main_root,
            marker: main_marker,
            max_depth: main_depth,
            conditions: main_conditions,
            fallback_for: main_fallback,
        } = &main.match_spec
        else {
            return Err(invalid());
        };
        if main_conditions.is_empty()
            || main_fallback.is_some()
            || !root.eq_ignore_ascii_case(main_root)
            || marker != main_marker
            || max_depth != main_depth
        {
            return Err(invalid());
        }
    }
    Ok(())
}

fn parse_target(value: &Value, path: &str) -> Result<Target, RulesError> {
    let map = object(value, path)?;
    expect_keys(
        map,
        &[
            "target_id",
            "group_id",
            "rule_id",
            "level",
            "category",
            "match",
            "item_type",
            "min_age_hours",
            "keep_newest",
            "in_use",
            "git_is_copy",
            "age_basis",
            "keep_by",
            "unlocks_target_ids",
        ],
        path,
    )?;
    let target_id = as_str(field(map, "target_id", path)?, &format!("{path}/target_id"))?;
    let group_id = as_str(field(map, "group_id", path)?, &format!("{path}/group_id"))?;
    let rule_id = as_str(field(map, "rule_id", path)?, &format!("{path}/rule_id"))?;
    if target_id.is_empty() || group_id.is_empty() || rule_id.is_empty() {
        return Err(RulesError::new("RULES_INVALID_ID", path));
    }
    let level = Level::parse(as_str(
        field(map, "level", path)?,
        &format!("{path}/level"),
    )?)
    .ok_or_else(|| RulesError::new("RULES_INVALID_LEVEL", format!("{path}/level")))?;
    let category = Category::parse(as_str(
        field(map, "category", path)?,
        &format!("{path}/category"),
    )?)
    .ok_or_else(|| RulesError::new("RULES_INVALID_CATEGORY", format!("{path}/category")))?;
    let item_type = ItemType::parse(as_str(
        field(map, "item_type", path)?,
        &format!("{path}/item_type"),
    )?)
    .ok_or_else(|| RulesError::new("RULES_INVALID_ITEM_TYPE", format!("{path}/item_type")))?;
    let match_value = field(map, "match", path)?;
    let pattern_conditions = if match_value.get("kind").and_then(Value::as_str) == Some("pattern") {
        match_value
            .get("conditions")
            .map(|v| parse_conditions(v, &format!("{path}/match/conditions")))
            .transpose()?
            .unwrap_or_default()
    } else {
        vec![]
    };
    if !pattern_conditions.is_empty()
        && (category != Category::Custom
            || !matches!(
                &pattern_conditions[..],
                [Condition::SiblingWithoutSuffix { .. }]
            ))
    {
        return Err(RulesError::new(
            "RULES_INVALID_CONDITIONS",
            format!("{path}/match/conditions"),
        ));
    }
    let target = Target {
        target_id: target_id.to_string(),
        group_id: group_id.to_string(),
        rule_id: rule_id.to_string(),
        level,
        category,
        match_spec: parse_match(match_value, &format!("{path}/match"))?,
        item_type,
        min_age_hours: bounded_u32(
            field(map, "min_age_hours", path)?,
            0,
            8760,
            &format!("{path}/min_age_hours"),
        )?,
        keep_newest: bounded_u32(
            field(map, "keep_newest", path)?,
            0,
            8,
            &format!("{path}/keep_newest"),
        )?,
        in_use: parse_plan_in_use(field(map, "in_use", path)?, &format!("{path}/in_use"))?,
        git_is_copy: m35::git_is_copy(map, path)?,
        age_basis: m35::age_basis(map, path)?,
        keep_by: m35::keep_by(map, path)?,
        unlocks_target_ids: map
            .get("unlocks_target_ids")
            .map(|v| {
                let invalid = || {
                    RulesError::new(
                        "RULES_INVALID_UNLOCKS",
                        format!("{path}/unlocks_target_ids"),
                    )
                };
                let ids = v.as_array().ok_or_else(invalid)?;
                if ids.is_empty() {
                    return Err(invalid());
                }
                ids.iter()
                    .map(|v| {
                        v.as_str()
                            .filter(|s| !s.is_empty())
                            .map(str::to_string)
                            .ok_or_else(invalid)
                    })
                    .collect::<Result<Vec<_>, _>>()
            })
            .transpose()?
            .unwrap_or_default(),
        pattern_conditions,
    };
    if (target.git_is_copy && !target.level.is_cleanable())
        || (target.keep_by == KeepBy::VersionName && target.keep_newest == 0)
    {
        return Err(RulesError::new("RULES_INVALID_FIELD", path));
    }
    if let MatchSpec::Marker {
        marker, conditions, ..
    } = &target.match_spec
    {
        m35::validate_marker(marker, conditions, item_type, path)?;
    }
    Ok(target)
}

fn parse_plan_in_use(value: &Value, path: &str) -> Result<InUse, RulesError> {
    crate::rules::parse_in_use(value, path)
}

fn parse_match(value: &Value, path: &str) -> Result<MatchSpec, RulesError> {
    let map = object(value, path)?;
    let kind = as_str(field(map, "kind", path)?, &format!("{path}/kind"))?;
    match kind {
        "pattern" => {
            expect_keys(map, &["kind", "base", "segments", "conditions"], path)?;
            let base = abs_path_field(map, "base", path, true)?;
            Ok(MatchSpec::Pattern {
                base,
                segments: parse_plan_segments(
                    field(map, "segments", path)?,
                    &format!("{path}/segments"),
                )?,
            })
        }
        "discovery" => {
            expect_keys(map, &["kind", "root"], path)?;
            Ok(MatchSpec::Discovery {
                root: abs_path_field(map, "root", path, false)?,
            })
        }
        "marker" => {
            expect_keys(
                map,
                &[
                    "kind",
                    "root",
                    "marker",
                    "max_depth",
                    "conditions",
                    "fallback_for",
                ],
                path,
            )?;
            Ok(MatchSpec::Marker {
                root: abs_path_field(map, "root", path, true)?,
                marker: parse_plan_marker(field(map, "marker", path)?, &format!("{path}/marker"))?,
                max_depth: bounded_u32(
                    field(map, "max_depth", path)?,
                    1,
                    16,
                    &format!("{path}/max_depth"),
                )?,
                conditions: map
                    .get("conditions")
                    .map(|value| parse_conditions(value, &format!("{path}/conditions")))
                    .transpose()?
                    .unwrap_or_default(),
                fallback_for: map
                    .get("fallback_for")
                    .map(|value| {
                        let id = as_str(value, &format!("{path}/fallback_for"))?;
                        if id.is_empty() {
                            return Err(RulesError::new(
                                "RULES_INVALID_FALLBACK",
                                format!("{path}/fallback_for"),
                            ));
                        }
                        Ok(id.to_string())
                    })
                    .transpose()?,
            })
        }
        _ => Err(RulesError::new(
            "RULES_INVALID_LOCATE",
            format!("{path}/kind"),
        )),
    }
}

fn abs_path_field(
    map: &Map<String, Value>,
    key: &str,
    parent: &str,
    allow_drive_root: bool,
) -> Result<String, RulesError> {
    let path = format!("{parent}/{key}");
    let text = as_str(field(map, key, parent)?, &path)?;
    if allow_drive_root {
        validate_guard_path(text, &path)?;
    } else if !valid_abs_path(text, false) {
        return Err(RulesError::new("RULES_INVALID_PATH", path));
    }
    Ok(text.to_string())
}

fn parse_plan_segments(value: &Value, path: &str) -> Result<Vec<String>, RulesError> {
    let items = value
        .as_array()
        .ok_or_else(|| RulesError::new("RULES_INVALID_LOCATE", path))?;
    if items.is_empty() || items.len() > 8 {
        return Err(RulesError::new("RULES_INVALID_LOCATE", path));
    }
    let mut segments = Vec::with_capacity(items.len());
    for (index, item) in items.iter().enumerate() {
        let item_path = format!("{path}/{index}");
        let segment = as_str(item, &item_path)?;
        if !valid_segment(segment) {
            return Err(RulesError::new("RULES_INVALID_LOCATE", item_path));
        }
        segments.push(segment.to_string());
    }
    Ok(segments)
}

fn parse_plan_marker(value: &Value, path: &str) -> Result<Marker, RulesError> {
    crate::rules::parse_marker(value, path)
}

fn parse_hard_guards(value: &Value, path: &str) -> Result<Vec<String>, RulesError> {
    let items = value
        .as_array()
        .ok_or_else(|| RulesError::new("RULES_INVALID_SHAPE", path))?;
    let mut guards = Vec::with_capacity(items.len());
    for (index, item) in items.iter().enumerate() {
        let item_path = format!("{path}/{index}");
        let text = as_str(item, &item_path)?;
        validate_guard_path(text, &item_path)?;
        guards.push(text.to_string());
    }
    Ok(guards)
}
