#![deny(unsafe_op_in_unsafe_fn)]

use crate::{App, RegValue, parse_entry};
use std::ptr::null_mut;

type Hkey = isize;
const KEY_READ: u32 = 0x20019;
const VIEW64: u32 = 0x100;
const VIEW32: u32 = 0x200;
const ROOT: &str = "SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Uninstall";

#[link(name = "advapi32")]
unsafe extern "system" {
    fn RegOpenKeyExW(
        key: Hkey,
        sub: *const u16,
        options: u32,
        access: u32,
        result: *mut Hkey,
    ) -> i32;
    fn RegEnumKeyExW(
        key: Hkey,
        index: u32,
        name: *mut u16,
        len: *mut u32,
        reserved: *mut u32,
        class: *mut u16,
        class_len: *mut u32,
        time: *mut u64,
    ) -> i32;
    fn RegQueryValueExW(
        key: Hkey,
        name: *const u16,
        reserved: *mut u32,
        kind: *mut u32,
        data: *mut u8,
        len: *mut u32,
    ) -> i32;
    fn RegCloseKey(key: Hkey) -> i32;
}

struct Key(Hkey);
impl Drop for Key {
    fn drop(&mut self) {
        // SAFETY: owned handle returned by a successful open; closed exactly once.
        unsafe {
            RegCloseKey(self.0);
        }
    }
}
fn wide(s: &str) -> Vec<u16> {
    s.encode_utf16().chain(Some(0)).collect()
}
fn open(key: Hkey, sub: &[u16], view: u32) -> Option<Key> {
    let mut result = 0;
    // SAFETY: sub is NUL-terminated, result is writable and no handle is retained on failure.
    (unsafe { RegOpenKeyExW(key, sub.as_ptr(), 0, KEY_READ | view, &mut result) } == 0)
        .then(|| Key(result))
}
fn value(key: &Key, name: &str) -> RegValue {
    // 1024 Unicode scalar values can require 2048 UTF-16 code units, plus NUL.
    let mut data = [0_u16; 2050];
    let mut bytes = (data.len() * 2) as u32;
    let mut kind = 0;
    let name = wide(name);
    // SAFETY: aligned buffer with its exact writable capacity, valid pointers for the call.
    let status = unsafe {
        RegQueryValueExW(
            key.0,
            name.as_ptr(),
            null_mut(),
            &mut kind,
            data.as_mut_ptr().cast(),
            &mut bytes,
        )
    };
    if status != 0 || !bytes.is_multiple_of(2) || bytes as usize > data.len() * 2 {
        return RegValue::Other;
    }
    let Ok(s) = String::from_utf16(&data[..bytes as usize / 2]) else {
        return RegValue::Other;
    };
    match kind {
        1 => RegValue::Sz(s),
        2 => RegValue::ExpandSz(s),
        _ => RegValue::Other,
    }
}

pub(crate) fn collect() -> (Vec<App>, bool, u64) {
    let mut apps = Vec::new();
    let mut unreadable = 0;
    for (scope, root, view) in [
        ("machine", -2147483646_isize, VIEW64),
        ("machine_32", -2147483646_isize, VIEW32),
        ("user", -2147483647_isize, 0),
    ] {
        if scope == "machine_32" { continue; }
        let Some(root) = open(root, &wide(ROOT), view) else {
            continue;
        };
        let mut index = 0;
        loop {
            let mut name = [0_u16; 256];
            let mut len = name.len() as u32;
            // SAFETY: all outputs are valid buffers or optional null pointers.
            let status = unsafe {
                RegEnumKeyExW(
                    root.0,
                    index,
                    name.as_mut_ptr(),
                    &mut len,
                    null_mut(),
                    null_mut(),
                    null_mut(),
                    null_mut(),
                )
            };
            if status == 259 {
                break;
            }
            if status != 0 {
                unreadable += 1;
                break;
            }
            index += 1;
            let Ok(key) = String::from_utf16(&name[..len as usize]) else {
                unreadable += 1;
                continue;
            };
            let Some(child) = open(root.0, &wide(&key), view) else {
                unreadable += 1;
                continue;
            };
            let values = [
                ("DisplayName".into(), value(&child, "DisplayName")),
                ("DisplayVersion".into(), value(&child, "DisplayVersion")),
                ("InstallLocation".into(), value(&child, "InstallLocation")),
            ];
            if let Some(app) = parse_entry(scope, &key, &values) {
                if apps.len() == 5000 {
                    return (apps, true, unreadable);
                }
                apps.push(app);
            }
        }
    }
    (apps, false, unreadable)
}
