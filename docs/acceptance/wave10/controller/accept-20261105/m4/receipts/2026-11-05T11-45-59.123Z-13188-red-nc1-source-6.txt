    struct Handle(*mut c_void);
    impl Drop for Handle {
        fn drop(&mut self) {
            // SAFETY: this wrapper owns exactly one live handle.
            unsafe {
                CloseHandle(self.0);
            }
        }
    }

    // A traversal owns parents before children. Even an early return must close the
    // innermost handle first, just as unwinding the former recursive calls did.
    struct FrameStack<T>(Vec<T>);
    impl<T> FrameStack<T> {
        fn new() -> Self {
            Self(Vec::new())
        }
    }
    impl<T> std::ops::Deref for FrameStack<T> {
        type Target = Vec<T>;
        fn deref(&self) -> &Self::Target {
            &self.0
        }
    }
    impl<T> std::ops::DerefMut for FrameStack<T> {
        fn deref_mut(&mut self) -> &mut Self::Target {
            &mut self.0
        }
    }
    impl<T> Drop for FrameStack<T> {
        fn drop(&mut self) {
            while self.0.pop().is_some() {}
        }
    }

    fn wide(s: &str) -> Vec<u16> {
        s.encode_utf16().chain(Some(0)).collect()
    }

    // The drive root, identity_of and guard resolution still open by name; long names need
    // the \\?\ prefix so they are not capped at MAX_PATH.
    fn prefixed(s: &str) -> Vec<u16> {
        if s.len() >= 248 && !s.starts_with(r"\\") {
            wide(&format!(r"\\?\{s}"))
        } else {
            wide(s)
        }
    }

    // Convert a FILETIME tick count to SystemTime with checked arithmetic; any overflow yields
    // None so the caller treats the item as `changed` rather than wrapping to a bogus time.
    fn filetime(ticks: i64) -> Option<SystemTime> {
        let delta = ticks.checked_sub(FILETIME_UNIX)?;
        let nanos = delta.unsigned_abs().checked_mul(100)?;
        let dur = Duration::from_nanos(nanos);
        if delta >= 0 {
            SystemTime::UNIX_EPOCH.checked_add(dur)
        } else {
            SystemTime::UNIX_EPOCH.checked_sub(dur)
        }
    }

    fn open_root(drive: &str) -> Option<Handle> {
        let name = wide(drive);
        // SAFETY: name is NUL-terminated; only read/list access with backup semantics.
        let raw = unsafe {
            CreateFileW(
                name.as_ptr(),
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            None
        } else {
            Some(Handle(raw))
        }
    }

    // Open a single-segment child relative to the parent handle. Returns the Win32 error on
    // failure so callers can tell "missing" from "changed".
    fn nt_open(parent: &Handle, seg: &str, access: u32, share: u32) -> Result<Handle, u32> {
        nt_open_at(parent.0, seg, access, share)
    }

    fn nt_open_at(parent: *mut c_void, seg: &str, access: u32, share: u32) -> Result<Handle, u32> {
        nt_open_options(parent, seg, access, share, OPTS)
    }

    fn nt_open_options(
        parent: *mut c_void,
        seg: &str,
        access: u32,
        share: u32,
        options: u32,
    ) -> Result<Handle, u32> {
        let mut name: Vec<u16> = seg.encode_utf16().collect();
        // A real NTFS path component is 1..=255 UTF-16 units. Reject an empty name (length 0
        // would open the parent directory itself instead of a child) and anything longer than 255
        // (so the UNICODE_STRING byte length can never overflow u16 and wrap).
        let byte_len = match name.len().checked_mul(2) {
            Some(n) if (1..=255).contains(&name.len()) => n as u16,
            _ => return Err(ERR_INVALID_NAME),
        };
        let mut us = UnicodeString {
            length: byte_len,
            maximum_length: byte_len,
            buffer: name.as_mut_ptr(),
        };
        let oa = ObjectAttributes {
            length: size_of::<ObjectAttributes>() as u32,
            root_directory: parent,
            object_name: &mut us,
            attributes: OBJ_CASE_INSENSITIVE,
            security_descriptor: ptr::null_mut(),
            security_qos: ptr::null_mut(),
        };
        let mut io = IoStatusBlock {
            status_or_pointer: 0,
            information: 0,
        };
        let mut handle: *mut c_void = ptr::null_mut();
        // SAFETY: oa/io/handle are valid and the single-segment name lives across the call.
        let status = unsafe { NtOpenFile(&mut handle, access, &oa, &mut io, share, options) };
        // Own any returned handle immediately so RAII closes it even on an unusual non-zero
        // status; only status 0 is a success.
        let owned = if handle.is_null() {
            None
        } else {
            Some(Handle(handle))
        };
        if status != 0 {
            // SAFETY: pure status-to-DOS-error translation, no pointers. `owned` drops here,
            // closing any handle returned alongside the failure.
            return Err(unsafe { RtlNtStatusToDosError(status) });
        }
        owned.ok_or(ERR_INVALID_NAME)
    }

    fn info<T>(handle: &Handle, class: i32, value: &mut T) -> bool {
        // SAFETY: value is an initialized, aligned buffer of exactly size_of::<T>() bytes.
        0 != unsafe {
            GetFileInformationByHandleEx(
                handle.0,
                class,
                (value as *mut T).cast(),
                size_of::<T>() as u32,
            )
        }
    }

    fn attr_tag(handle: &Handle) -> Option<(u32, u32)> {
        let mut info_buf = FileAttributeTagInfo {
            file_attributes: 0,
            reparse_tag: 0,
        };
        info(handle, 9, &mut info_buf).then_some((info_buf.file_attributes, info_buf.reparse_tag))
    }

    fn file_id(handle: &Handle) -> Option<[u8; 16]> {
        let mut id = FileIdInfo {
            volume_serial: 0,
            file_id: [0; 16],
        };
        info(handle, 18, &mut id).then_some(id.file_id)
    }

    fn write_time(handle: &Handle) -> Option<i64> {
        let mut basic = FileBasicInfo {
            creation: 0,
            last_access: 0,
            last_write: 0,
            change: 0,
            attributes: 0,
        };
        info(handle, 0, &mut basic).then_some(basic.last_write)
    }

    fn standard_info(handle: &Handle) -> Option<(u64, u32)> {
        let mut std_info = FileStandardInfo {
            allocation: 0,
            end_of_file: 0,
            links: 0,
            delete_pending: 0,
            directory: 0,
        };
        info(handle, 1, &mut std_info)
            .then_some((std_info.end_of_file.max(0) as u64, std_info.links))
    }

    fn end_of_file(handle: &Handle) -> Option<u64> {
        standard_info(handle).map(|(size, _)| size)
    }

    fn volume_serial(handle: &Handle) -> Option<u32> {
        let mut serial = 0u32;
        // SAFETY: handle is live; optional outputs are null with zero sizes.
        let ok = unsafe {
            GetVolumeInformationByHandleW(
                handle.0,
                ptr::null_mut(),
                0,
                &mut serial,
                ptr::null_mut(),
                ptr::null_mut(),
                ptr::null_mut(),
                0,
            )
        };
        (ok != 0).then_some(serial)
    }

    fn identity_from(handle: &Handle) -> Option<Identity> {
        let id = file_id(handle)?;
        let serial = volume_serial(handle)?;
        Some(Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: id.iter().map(|b| format!("{b:02x}")).collect(),
        })
    }

    #[derive(Clone)]
    struct Raw {
        name: String,
        name_ok: bool,
        attr: u32,
        tag: u32,
        file_id: [u8; 16],
        size: u64,
        write: i64,
        created: i64,
    }

    // Enumerate one directory handle. Class 20 restarts, class 19 continues; the listing ends
    // only when GetFileInformationByHandleEx fails with ERROR_NO_MORE_FILES. Every other
    // failure is an error, never a silent "no entries".
    fn enumerate(handle: &Handle) -> Result<Vec<Raw>, ()> {
        let mut out = Vec::new();
        let mut buf = vec![0u64; 4096];
        let byte_len = buf.len() * 8;
        let mut class = 20i32;
        loop {
            // SAFETY: buf is 8-byte aligned (Vec<u64>) and its full byte length is passed.
            let ok = unsafe {
                GetFileInformationByHandleEx(
                    handle.0,
                    class,
                    buf.as_mut_ptr().cast(),
                    byte_len as u32,
                )
            };
            if ok == 0 {
                // last_os_error reads GetLastError immediately after the failing call.
                return if std::io::Error::last_os_error().raw_os_error() == Some(ERR_NO_MORE_FILES)
                {
                    Ok(out)
                } else {
                    Err(())
                };
            }
            class = 19;
            let base = buf.as_ptr() as *const u8;
            let mut off = 0usize;
            loop {
                // The fixed 88-byte header must lie fully inside the filled buffer.
                if off.checked_add(88).is_none_or(|end| end > byte_len) {
                    return Err(());
                }
                // SAFETY: [off, off+88) is in-bounds (checked just above); each fixed field is
                // read unaligned at its SPEC offset from the 8-byte-aligned buffer.
                let (next, write, size, attr, name_len, tag, file_id, created) = unsafe {
                    let p = base.add(off);
                    let next = ptr::read_unaligned(p as *const u32) as usize;
                    let write = ptr::read_unaligned(p.add(24) as *const i64);
                    let size = ptr::read_unaligned(p.add(40) as *const i64).max(0) as u64;
                    let attr = ptr::read_unaligned(p.add(56) as *const u32);
                    let name_len = ptr::read_unaligned(p.add(60) as *const u32) as usize;
                    let tag = ptr::read_unaligned(p.add(68) as *const u32);
                    let mut file_id = [0u8; 16];
                    ptr::copy_nonoverlapping(p.add(72), file_id.as_mut_ptr(), 16);
                    let created = ptr::read_unaligned(p.add(8) as *const i64);
                    (next, write, size, attr, name_len, tag, file_id, created)
                };
                // The variable-length name must also lie fully inside the buffer.
                if off
                    .checked_add(88)
                    .and_then(|h| h.checked_add(name_len))
                    .is_none_or(|end| end > byte_len)
                {
                    return Err(());
                }
                // SAFETY: [off+88, off+88+name_len) is in-bounds (checked just above); the
                // pointer is u16-aligned because `off` is always a multiple of 8 (it starts at 0
                // and every NextEntryOffset advance is checked to be a multiple of 8) and 88 is
                // even, so base+off+88 is 2-byte aligned; NTFS names are UTF-16 so name_len bytes
                // decode as name_len/2 u16 units.
                let (name, name_ok) = unsafe {
                    let units =
                        std::slice::from_raw_parts(base.add(off + 88) as *const u16, name_len / 2);
                    match String::from_utf16(units) {
                        Ok(s) => (s, true),
                        // NUL cannot occur in an actual component. This private snapshot
                        // key is injective even for names with identical lossy spellings,
                        // and is never passed to an open call (name_ok is false).
                        Err(_) => (
                            format!(
                                "\0{}",
                                units.iter().map(|u| format!("{u:04x}")).collect::<String>()
                            ),
                            false,
                        ),
                    }
                };
                if name != "." && name != ".." {
                    out.push(Raw {
                        name,
                        name_ok,
                        attr,
                        tag,
                        file_id,
                        size,
                        write,
                        created,
                    });
                }
                if next == 0 {
                    break;
                }
                // NextEntryOffset must stay 8-byte aligned and strictly advance forward.
                if next % 8 != 0 {
                    return Err(());
                }
                off = off.checked_add(next).ok_or(())?;
            }
        }
    }

    fn set_delete(handle: &Handle) -> bool {
        let ex = FileDispositionInfoEx {
            flags: 0x1 | 0x2 | 0x10,
        };
        // SAFETY: class 21 reads a 4-byte flags field.
        let ok = unsafe {
            SetFileInformationByHandle(
                handle.0,
                21,
                (&ex as *const FileDispositionInfoEx).cast(),
                size_of::<FileDispositionInfoEx>() as u32,
            )
        };
        if ok != 0 {
            return true;
        }
        let plain = FileDispositionInfo { delete_file: 1 };
        // SAFETY: class 4 reads a 1-byte flag.
        0 != unsafe {
            SetFileInformationByHandle(
                handle.0,
                4,
                (&plain as *const FileDispositionInfo).cast(),
                size_of::<FileDispositionInfo>() as u32,
            )
        }
    }

    fn single_name(seg: &str) -> Result<Vec<u16>, u32> {
        let name: Vec<u16> = seg.encode_utf16().collect();
        if !(1..=255).contains(&name.len())
            || seg.contains(['\\', '/', ':', '\0'])
            || matches!(seg, "." | "..")
        {
            return Err(ERR_INVALID_NAME);
        }
        Ok(name)
    }

    fn create_dir_at(parent: &Handle, seg: &str) -> Result<Handle, u32> {
        let mut name = single_name(seg)?;
        let mut us = UnicodeString {
            length: (name.len() * 2) as u16,
            maximum_length: (name.len() * 2) as u16,
            buffer: name.as_mut_ptr(),
        };
        let oa = ObjectAttributes {
            length: size_of::<ObjectAttributes>() as u32,
            root_directory: parent.0,
            object_name: &mut us,
            attributes: OBJ_CASE_INSENSITIVE,
            security_descriptor: ptr::null_mut(),
            security_qos: ptr::null_mut(),
        };
        let mut io = IoStatusBlock {
            status_or_pointer: 0,
            information: 0,
        };
        let mut raw = ptr::null_mut();
        // SAFETY: initialized ABI structures, live parent and single-segment name. FILE_CREATE
        // never opens an existing object; only a directory can be created.
        let status = unsafe {
            NtCreateFile(
                &mut raw,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                &oa,
                &mut io,
                ptr::null(),
                DIRECTORY,
                SHARE_RW,
                2,
                OPTS | 1,
                ptr::null(),
                0,
            )
        };
        let owned = (!raw.is_null()).then(|| Handle(raw));
        if status != 0 {
            return Err(unsafe { RtlNtStatusToDosError(status) });
        }
        owned.ok_or(ERR_INVALID_NAME)
    }

    fn rename_at(handle: &Handle, parent: &Handle, seg: &str) -> Result<(), u32> {
        let name = single_name(seg)?;
        // FILE_RENAME_INFO: BOOLEAN at 0, HANDLE at 8, DWORD length at 16, WCHARs at 20.
        // Include the fixed structure's trailing WCHAR/padding and a NUL terminator.
        let size = 24 + name.len() * 2;
        let mut buf = vec![0u64; size.div_ceil(8)];
        let base = buf.as_mut_ptr().cast::<u8>();
        // SAFETY: 8-byte aligned initialized buffer, with room for the variable UTF-16 tail.
        unsafe {
            ptr::write(base, 0u8); // ReplaceIfExists: never overwrite.
            ptr::write(base.add(8).cast::<*mut c_void>(), parent.0);
            ptr::write(base.add(16).cast::<u32>(), (name.len() * 2) as u32);
            ptr::copy_nonoverlapping(name.as_ptr(), base.add(20).cast::<u16>(), name.len());
        }
        let mut io = IoStatusBlock {
            status_or_pointer: 0,
            information: 0,
        };
        // SAFETY: class 10 reads the aligned FILE_RENAME_INFORMATION buffer above.
        // Source and target-parent handles are live across this synchronous operation.
        let status =
            unsafe { NtSetInformationFile(handle.0, &mut io, base.cast(), size as u32, 10) };
        if status == 0 {
            Ok(())
        } else {
            // SAFETY: pure translation of the returned NT status.
            Err(unsafe { RtlNtStatusToDosError(status) })
        }
    }

    fn last_error() -> u32 {
        std::io::Error::last_os_error().raw_os_error().unwrap_or(1) as u32
    }

    #[derive(Debug)]
    enum ChainError {
        Missing,
        Reparse,
        Other(u32),
    }
    fn chain_error(code: u32) -> ChainError {
        if matches!(code, 2 | 3) {
            ChainError::Missing
        } else {
            ChainError::Other(code)
        }
    }
    fn plain(handle: &Handle) -> Result<(), ChainError> {
        let (attr, _) = attr_tag(handle).ok_or(ChainError::Other(last_error()))?;
        if attr & REPARSE != 0 {
            Err(ChainError::Reparse)
        } else if attr & DIRECTORY == 0 {
            Err(ChainError::Other(267))
        } else {
            Ok(())
        }
    }

    // Each ancestor remains open without delete sharing until its descendants are closed.
    fn open_chain(name: &str, delete_last: bool) -> Result<FrameStack<Handle>, ChainError> {
        directory_chain(name, None, delete_last)
    }
    fn directory_chain(
        name: &str,
        create_from: Option<usize>,
        delete_last: bool,
    ) -> Result<FrameStack<Handle>, ChainError> {
        let mut handles = FrameStack::new();
        let root = open_root(&name[..3]).ok_or_else(|| ChainError::Other(last_error()))?;
        plain(&root)?;
        handles.push(root);
        if name.len() == 3 {
            return Ok(handles);
        }
        let parts: Vec<&str> = name[3..].split('\\').collect();
        for (i, part) in parts.iter().enumerate() {
            let access = FILE_LIST_DIRECTORY
                | FILE_READ_ATTRIBUTES
                | 0x20 // FILE_TRAVERSE: relative rename target directories.
                | SYNCHRONIZE
                | if delete_last && i + 1 == parts.len() {
                    DELETE
                } else {
                    0
                };
            let parent = handles.last().unwrap();
            let next = match nt_open(parent, part, access, SHARE_RW) {
                Ok(h) => h,
                Err(2 | 3) if create_from.is_some_and(|start| i >= start) => {
                    match create_dir_at(parent, part) {
                        Ok(h) => h,
                        // A concurrent creator is allowed only after a fresh no-follow open.
                        Err(80 | 183) => {
                            nt_open(parent, part, access, SHARE_RW).map_err(chain_error)?
                        }
                        Err(e) => return Err(chain_error(e)),
                    }
                }
                Err(e) => return Err(chain_error(e)),
            };
            plain(&next)?;
            handles.push(next);
        }
        Ok(handles)
    }

    fn move_error(code: u32) -> MoveError {
        match code {
            17 => MoveError::NotSameVolume,
            80 => MoveError::DestinationExists,
            5 | 32 | 33 => MoveError::Locked,
            _ => MoveError::Other,
        }
    }

    #[allow(clippy::too_many_arguments)]
    pub(crate) fn move_to_quarantine(
        plan: &CleanPlan,
        item: &PlanItem,
        root: &str,
        id: &str,
        before: &Snapshot,
        guard: &dyn ItemGuard,
        guard_ids: &[Identity],
        protected_ids: &[(String, Identity)],
        journal: &mut dyn std::io::Write,
        stop: &StopFlag,
        now: SystemTime,
    ) -> Result<MoveReport, CleanError> {
        let mut result = MoveReport {
            item_id: item.item_id.clone(),
            status: MoveStatus::Failed,
            skip_reason: None,
            move_error: None,
            error_code: None,
            destination: None,
            identity: None,
            bytes: before.bytes.to_string(),
        };
        let dest = format!("{root}\\{id}\\{}", &item.path[3..]);
        let (dest_parent, dest_name) = dest.rsplit_once('\\').ok_or(CleanError::InvalidPlan)?;
        let start = root[3..].split('\\').count() - 1;
        let qroot_chain = match directory_chain(root, Some(start), false) {
            Ok(h) => h,
            Err(e) => {
                result.move_error = Some(MoveError::QuarantineUnavailable);
                result.error_code = match e {
                    ChainError::Other(c) => Some(c),
                    _ => None,
                };
                return Ok(result);
            }
        };
        match identity_from(qroot_chain.last().unwrap()) {
            Some(actual) if actual.volume_serial == item.identity.volume_serial => {}
            Some(_) => {
                result.move_error = Some(MoveError::NotSameVolume);
                return Ok(result);
            }
            None => {
                result.move_error = Some(MoveError::QuarantineUnavailable);
                result.error_code = Some(last_error());
                return Ok(result);
            }
        }
        let dest_chain = match directory_chain(dest_parent, Some(start), false) {
            Ok(h) => h,
            Err(e) => {
                result.move_error = Some(MoveError::QuarantineUnavailable);
                result.error_code = match e {
                    ChainError::Other(c) => Some(c),
                    _ => None,
                };
                return Ok(result);
            }
        };
        let source_parent = item.path.rsplit_once('\\').unwrap().0;
        let source_chain = match open_chain(
            if source_parent.len() == 2 {
                &item.path[..3]
            } else {
                source_parent
            },
            false,
        ) {
            Ok(h) => h,
            Err(_) => {
                result.status = MoveStatus::Skipped;
                result.skip_reason = Some(SkipReason::Changed);
                return Ok(result);
            }
        };
        crate::journal_line(
            journal,
            serde_json::json!({
                "journal_version":"water.quarantine-journal.v1", "event":"intent", "quarantine_id":id,
                "item_id":item.item_id, "path":item.path, "destination":dest, "at": time::format(SystemTime::now())?
            }),
        )?;
        // Revalidate the snapshot and all guards after the external journal/guard callbacks.
        match crate::preflight(plan, item, guard, now, guard_ids, protected_ids) {
            Ok(fresh) if &fresh == before => {}
            other => {
                result.status = MoveStatus::Skipped;
                result.skip_reason = Some(other.err().unwrap_or(SkipReason::Changed));
            }
        }
        if stop.is_stopped() {
            result.status = MoveStatus::Skipped;
            result.skip_reason = Some(SkipReason::Cancelled);
        }
        if result.skip_reason.is_none() {
            let moved = nt_open(
                source_chain.last().unwrap(),
                last_segment(&item.path),
                DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            )
            .and_then(|handle| {
                if identity_from(&handle).as_ref() != Some(&item.identity) {
                    return Ok(None);
                }
                if attr_tag(&handle).is_none_or(|(attr, _)| {
                    attr & REPARSE != 0 || (attr & DIRECTORY != 0) != (item.kind == Kind::Dir)
                }) {
                    return Ok(None);
                }
                rename_at(&handle, dest_chain.last().unwrap(), dest_name)?;
                // A rename preserves this live handle's identity. Read it again after the move;
                // an attribute-query failure must never turn an actual move into a failed move.
                Ok(Some(
                    identity_from(&handle).unwrap_or_else(|| item.identity.clone()),
                ))
            });
            match moved {
                Ok(Some(actual)) => {
                    result.status = MoveStatus::Moved;
                    result.identity = Some(actual);
                    result.destination = Some(dest);
                }
                Ok(None) => {
                    result.status = MoveStatus::Skipped;
                    result.skip_reason = Some(SkipReason::Changed);
                }
                Err(code) => {
                    result.move_error = Some(move_error(code));
                    result.error_code = Some(code);
                }
            }
        }
        crate::journal_line(
            journal,
            serde_json::json!({
                "journal_version":"water.quarantine-journal.v1", "event": if result.status == MoveStatus::Moved { "moved" } else { "failed" },
                "quarantine_id": id, "item_id":item.item_id, "path":item.path, "move_error":result.move_error,
                "at":time::format(SystemTime::now())?
            }),
        )?;
        Ok(result)
    }

    // Only empty skeleton directories are removed, never their contents. Stop at the qroot.
    fn remove_empty_up(start: &str, dir: &str, root: &str) -> (bool, bool) {
        let mut next = start.to_owned();
        let (mut dir_removed, mut root_removed) = (false, false);
        while path::contains(root, &next) {
            let chain = match open_chain(&next, true) {
                Ok(h) => h,
                Err(_) => break,
            };
            let handle = chain.last().unwrap();
            if !enumerate(handle).is_ok_and(|kids| kids.is_empty()) || !set_delete(handle) {
                break;
            }
            dir_removed |= next.eq_ignore_ascii_case(dir);
            root_removed |= next.eq_ignore_ascii_case(root);
            drop(chain);
            if next.eq_ignore_ascii_case(root) {
                break;
            }
            next = next.rsplit_once('\\').unwrap().0.to_owned();
        }
        (dir_removed, root_removed)
    }

    fn restore_error(code: u32) -> RestoreError {
        match code {
            17 => RestoreError::NotSameVolume,
            80 | 183 => RestoreError::DestinationExists,
            5 | 32 | 33 => RestoreError::Locked,
            _ => RestoreError::Other,
        }
    }
    pub(crate) fn restore_quarantine(
        dir: &str,
        requests: &[RestoreRequest],
        stop: &StopFlag,
        now: SystemTime,
    ) -> Result<RestoreReport, CleanError> {
        let root = crate::quarantine::validate_dir(dir)?;
        let mut report = RestoreReport {
            report_version: "water.restore-report.v1".into(),
            quarantine_dir: dir.into(),
            started_at: time::format(now)?,
            ended_at: String::new(),
            status: RunStatus::Completed,
            items: Vec::new(),
            dir_removed: false,
            root_removed: false,
        };
        for request in requests {
            let mut result = RestoreItem {
                rel: request.rel.clone(),
                status: RestoreStatus::Failed,
                reason: None,
                error_code: None,
            };
            if stop.is_stopped() {
                result.status = RestoreStatus::Skipped;
                result.reason = Some(RestoreError::Cancelled);
                report.status = RunStatus::Stopped;
            } else {
                let source = format!("{dir}\\{}", request.rel);
                let source_parent = source.rsplit_once('\\').unwrap().0;
                let destination = request
                    .to
                    .clone()
                    .unwrap_or_else(|| format!("{}{}", path::volume(dir), request.rel));
                let parent = destination.rsplit_once('\\').unwrap().0;
                let operation = (|| -> Result<(), (RestoreError, Option<u32>)> {
                    let chain = open_chain(source_parent, false).map_err(|e| match e {
                        ChainError::Missing => (RestoreError::Gone, None),
                        ChainError::Reparse => (RestoreError::ParentReparse, None),
                        ChainError::Other(c) => (restore_error(c), Some(c)),
                    })?;
                    let source_handle = nt_open(
                        chain.last().unwrap(),
                        last_segment(&source),
                        DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                        SHARE_RWD,
                    )
                    .map_err(|c| {
                        (
                            if matches!(c, 2 | 3) {
                                RestoreError::Gone
                            } else {
                                restore_error(c)
                            },
                            Some(c),
                        )
                    })?;
                    if identity_from(&source_handle).as_ref() != Some(&request.identity) {
                        return Err((RestoreError::Changed, None));
                    }
                    let parents = directory_chain(
                        if parent.len() == 2 {
                            &destination[..3]
                        } else {
                            parent
                        },
                        Some(0),
                        false,
                    )
                    .map_err(|e| match e {
                        ChainError::Reparse => (RestoreError::ParentReparse, None),
                        ChainError::Other(c) => (restore_error(c), Some(c)),
                        ChainError::Missing => (RestoreError::Other, None),
                    })?;
                    if stop.is_stopped() {
                        return Err((RestoreError::Cancelled, None));
                    }
                    rename_at(
                        &source_handle,
                        parents.last().unwrap(),
                        last_segment(&destination),
                    )
                    .map_err(|c| (restore_error(c), Some(c)))
                })();
                match operation {
                    Ok(()) => {
                        result.status = RestoreStatus::Restored;
                        let (d, r) = remove_empty_up(source_parent, dir, root);
                        report.dir_removed |= d;
                        report.root_removed |= r;
                    }
                    Err((reason, code)) => {
                        if matches!(
                            reason,
                            RestoreError::Gone | RestoreError::Changed | RestoreError::Cancelled
                        ) {
                            result.status = RestoreStatus::Skipped;
                        }
                        if reason == RestoreError::Cancelled {
                            report.status = RunStatus::Stopped;
                        }
                        result.reason = Some(reason);
                        result.error_code = code;
                    }
                }
            }
            report.items.push(result);
        }
        report.ended_at = time::format(SystemTime::now())?;
        Ok(report)
    }

    pub(crate) fn list_quarantines(root: &str) -> Result<QuarantineList, CleanError> {
        let mut list = QuarantineList {
            list_version: "water.quarantine-list.v1".into(),
            root: root.into(),
            state: "present".into(),
            quarantines: Vec::new(),
            stray_entries: 0,
        };
        let chain = match open_chain(root, false) {
            Ok(h) => h,
            Err(ChainError::Missing) => {
                list.state = "absent".into();
                return Ok(list);
            }
            Err(_) => {
                list.state = "unavailable".into();
                return Ok(list);
            }
        };
        let kids = match enumerate(chain.last().unwrap()) {
            Ok(k) => k,
            Err(_) => {
                list.state = "unavailable".into();
                return Ok(list);
            }
        };
        for child in kids {
            if child.name_ok
                && valid_id(&child.name)
                && child.attr & DIRECTORY != 0
                && child.attr & REPARSE == 0
            {
                list.quarantines.push(QuarantineSummary {
                    dir: format!("{root}\\{}", child.name),
                    quarantine_id: child.name,
                    created_at: filetime(child.created).and_then(|t| time::format(t).ok()),
                });
            } else {
                list.stray_entries += 1;
            }
        }
        list.quarantines
            .sort_by(|a, b| a.quarantine_id.cmp(&b.quarantine_id));
        Ok(list)
    }

    fn odd_name(name: &str) -> bool {
        // path::validate deliberately rejects all DOS device spellings and trailing dot/space.
        path::validate(&format!("X:\\{name}")).is_err()
    }
    struct ReadFrame {
        handle: Handle,
        rel: String,
        kids: Vec<Raw>,
        index: usize,
    }
    fn add_hint(info: &mut QuarantineInfo, entry: HintEntry) {
        if info.hard_entries.len() < 200 {
            info.hard_entries.push(entry);
        } else {
            info.hard_truncated = true;
        }
    }
    pub(crate) fn inspect_quarantine(
        dir: &str,
        stop: &StopFlag,
    ) -> Result<QuarantineInfo, CleanError> {
        let mut result = QuarantineInfo {
            info_version: "water.quarantine-info.v1".into(),
            dir: dir.into(),
            quarantine_id: last_segment(dir).into(),
            exists: true,
            total_bytes: "0".into(),
            disk_bytes: "0".into(),
            file_count: 0,
            dir_count: 0,
            hard_entries: Vec::new(),
            hard_truncated: false,
        };
        let mut chain = match open_chain(dir, false) {
            Ok(h) => h,
            Err(ChainError::Missing) => {
                result.exists = false;
                return Ok(result);
            }
            Err(ChainError::Reparse) => return Err(CleanError::InvalidPlan),
            Err(ChainError::Other(_)) => {
                add_hint(
                    &mut result,
                    HintEntry {
                        rel: String::new(),
                        kind: Kind::Dir,
                        bytes: "0".into(),
                        hints: vec![LeftoverHint::AccessDenied],
                    },
                );
                return Ok(result);
            }
        };
        let mut stack: FrameStack<ReadFrame> = FrameStack::new();
        let mut bytes = 0u64;
        let mut disk = 0u64;
        let initial = enumerate(chain.last().unwrap());
        if initial.is_err() {
            add_hint(
                &mut result,
                HintEntry {
                    rel: String::new(),
                    kind: Kind::Dir,
                    bytes: "0".into(),
                    hints: vec![LeftoverHint::AccessDenied],
                },
            );
        }
        let mut kids = initial.unwrap_or_default();
        kids.sort_by(|a, b| a.name.cmp(&b.name));
        stack.push(ReadFrame {
            handle: chain.pop().unwrap(),
            rel: String::new(),
            kids,
            index: 0,
        });
        while !stop.is_stopped() {
            let (base, raw, parent) = if let Some(frame) = stack.last_mut() {
                if frame.index == frame.kids.len() {
                    stack.pop();
                    continue;
                }
                let raw = frame.kids[frame.index].clone();
                frame.index += 1;
                (frame.rel.clone(), raw, &frame.handle)
            } else {
                break;
            };
            let rel = if base.is_empty() {
                raw.name.clone()
            } else {
                format!("{base}\\{}", raw.name)
            };
            let is_dir = raw.attr & DIRECTORY != 0;
            if is_dir {
                result.dir_count += 1;
            } else {
                result.file_count += 1;
            }
            let size = if raw.attr & REPARSE == 0 && !is_dir {
                raw.size
            } else {
                0
            };
            bytes = bytes.checked_add(size).ok_or(CleanError::Overflow)?;
            let mut hints = Vec::new();
            if !raw.name_ok || odd_name(&raw.name) {
                hints.push(LeftoverHint::OddName);
            }
            if dir.encode_utf16().count() + 1 + rel.encode_utf16().count() > 259 {
                hints.push(LeftoverHint::LongPath);
            }
            if raw.attr & 1 != 0 {
                hints.push(LeftoverHint::ReadOnly);
            }
            if raw.attr & REPARSE != 0 && raw.tag & NAME_SURROGATE == 0 {
                hints.push(LeftoverHint::Reparse);
            }
            let mut descend = None;
            if raw.name_ok && raw.attr & REPARSE == 0 {
                let access = FILE_READ_ATTRIBUTES
                    | SYNCHRONIZE
                    | if is_dir { FILE_LIST_DIRECTORY } else { 0 };
                match nt_open(parent, &raw.name, access, SHARE_RWD) {
                    Ok(handle) => {
                        if file_id(&handle) != Some(raw.file_id)
                            || attr_tag(&handle).is_none_or(|(a, _)| a & REPARSE != 0)
                        {
                            hints.push(LeftoverHint::AccessDenied);
                        } else if is_dir {
                            match enumerate(&handle) {
                                Ok(mut children) => {
                                    children.sort_by(|a, b| a.name.cmp(&b.name));
                                    descend = Some(ReadFrame {
                                        handle,
                                        rel: rel.clone(),
                                        kids: children,
                                        index: 0,
                                    });
                                }
                                Err(_) => hints.push(LeftoverHint::AccessDenied),
                            }
                        } else {
                            let mut standard = FileStandardInfo {
                                allocation: 0,
                                end_of_file: 0,
                                links: 0,
                                delete_pending: 0,
                                directory: 0,
                            };
                            if info(&handle, 1, &mut standard) {
                                disk = disk
                                    .checked_add(standard.allocation.max(0) as u64)
                                    .ok_or(CleanError::Overflow)?;
                            } else {
                                hints.push(LeftoverHint::AccessDenied);
                            }
                        }
                    }
                    Err(_) => hints.push(LeftoverHint::AccessDenied),
                }
            }
            if !hints.is_empty() {
                add_hint(
                    &mut result,
                    HintEntry {
                        rel,
                        kind: if is_dir { Kind::Dir } else { Kind::File },
                        bytes: size.to_string(),
                        hints,
                    },
                );
            }
            if let Some(frame) = descend {
                stack.push(frame);
            }
        }
        result.total_bytes = bytes.to_string();
        result.disk_bytes = disk.to_string();
        result.hard_entries.sort_by(|a, b| a.rel.cmp(&b.rel));
        Ok(result)
    }

    fn purge_open(parent: &Handle, seg: &str, directory: bool) -> Result<Handle, u32> {
        nt_open_options(
            parent.0,
            seg,
            DELETE
                | FILE_READ_ATTRIBUTES
                | SYNCHRONIZE
                | if directory { FILE_LIST_DIRECTORY } else { 0 },
            SHARE_RWD,
            OPTS,
        )
    }
    struct PurgeFrame {
        handle: Handle,
        rel: String,
        kids: Vec<Raw>,
        index: usize,
        blocked: bool,
        delete_error: Option<u32>,
    }
    struct Purging<'a> {
        report: &'a mut PurgeReport,
        observer: &'a dyn PurgeObserver,
        freed: u64,
        leftovers_count: u64,
    }
    impl Purging<'_> {
        fn leftover(
            &mut self,
            rel: String,
            kind: Kind,
            size: u64,
            reason: LeftoverReason,
            code: Option<u32>,
        ) {
            self.leftovers_count += 1;
            let value = Leftover {
                rel,
                kind,
                bytes: size.to_string(),
                reason,
                error_code: code,
            };
            // Keep the first 200 in lexicographic order even when enumeration isn't sorted.
            self.report.leftovers.push(value);
            self.report.leftovers.sort_by(|a, b| a.rel.cmp(&b.rel));
            if self.report.leftovers.len() > 200 {
                self.report.leftovers.pop();
                self.report.leftovers_truncated = true;
            }
        }
        fn error(&mut self, rel: String, kind: Kind, size: u64, code: u32) {
            let reason = match code {
                32 | 33 => LeftoverReason::InUse,
                5 => LeftoverReason::AccessDenied,
                _ => LeftoverReason::Other,
            };
            self.leftover(rel, kind, size, reason, Some(code));
        }
        fn removed(&mut self, size: u64) -> Result<(), CleanError> {
            self.report.removed_entries = self
                .report
                .removed_entries
                .checked_add(1)
                .ok_or(CleanError::Overflow)?;
            self.freed = self.freed.checked_add(size).ok_or(CleanError::Overflow)?;
            if self.report.removed_entries.is_multiple_of(256) {
                self.observer
                    .on_progress(self.report.removed_entries, self.freed);
            }
            Ok(())
        }
    }
    pub(crate) fn purge_quarantine(
        dir: &str,
        journal: &mut dyn std::io::Write,
        stop: &StopFlag,
        observer: &dyn PurgeObserver,
        now: SystemTime,
    ) -> Result<PurgeReport, CleanError> {
        let root = crate::quarantine::validate_dir(dir)?;
        // Reject every reparse ancestor before writing intent or deleting any object.
        let mut chain = open_chain(dir, false).map_err(|_| CleanError::InvalidPlan)?;
        let probe = chain.pop().ok_or(CleanError::InvalidPlan)?;
        let expected = identity_from(&probe).ok_or(CleanError::Native)?;
        drop(probe);
        let parent = chain.last().ok_or(CleanError::InvalidPlan)?;
        let (handle, delete_error) = match purge_open(parent, last_segment(dir), true) {
            Ok(h) => (h, None),
            Err(code) => (
                nt_open(
                    parent,
                    last_segment(dir),
                    FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                    SHARE_RW,
                )
                .map_err(|_| CleanError::Native)?,
                Some(code),
            ),
        };
        plain(&handle).map_err(|_| CleanError::InvalidPlan)?;
        if identity_from(&handle).as_ref() != Some(&expected) {
            return Err(CleanError::InvalidPlan);
        }
        let drive = path::volume(dir);
        let mut report = PurgeReport {
            report_version: "water.purge-report.v1".into(),
            quarantine_dir: dir.into(),
            started_at: time::format(now)?,
            ended_at: String::new(),
            status: RunStatus::Completed,
            volume: VolumeReport {
                root: drive.clone(),
                free_before: volume_free(&drive)?.to_string(),
                free_after: String::new(),
            },
            freed_bytes: "0".into(),
            removed_entries: 0,
            remaining_entries: 0,
            remaining_bytes: "0".into(),
            leftovers: Vec::new(),
            leftovers_truncated: false,
            dir_removed: false,
            root_removed: false,
        };
        crate::journal_line(
            journal,
            serde_json::json!({"journal_version":"water.purge-journal.v1", "event":"intent", "quarantine_dir":dir, "at":time::format(now)?}),
        )?;
        let mut stack: FrameStack<PurgeFrame> = FrameStack::new();
        let mut ctx = Purging {
            report: &mut report,
            observer,
            freed: 0,
            leftovers_count: 0,
        };
        let kids = match enumerate(&handle) {
            Ok(k) => k,
            Err(_) => {
                ctx.error(String::new(), Kind::Dir, 0, last_error());
                Vec::new()
            }
        };
        stack.push(PurgeFrame {
            handle,
            rel: String::new(),
            kids,
            index: 0,
            blocked: ctx.leftovers_count != 0,
            delete_error,
        });
        while !stack.is_empty() {
            if stop.is_stopped() {
                ctx.report.status = RunStatus::Stopped;
                break;
            }
            let frame = stack.last_mut().unwrap();
            if frame.index == frame.kids.len() {
                let frame = stack.pop().unwrap();
                if !frame.blocked {
                    if let Some(code) = frame.delete_error {
                        ctx.error(frame.rel.clone(), Kind::Dir, 0, code);
                        if let Some(parent) = stack.last_mut() {
                            parent.blocked = true;
                        }
                    } else if set_delete(&frame.handle) {
                        if frame.rel.is_empty() {
                            ctx.report.dir_removed = true;
                        }
                        drop(frame);
                        ctx.removed(0)?;
                    } else {
                        ctx.error(frame.rel.clone(), Kind::Dir, 0, last_error());
                        if let Some(parent) = stack.last_mut() {
                            parent.blocked = true;
                        }
                    }
                } else if let Some(parent) = stack.last_mut() {
                    parent.blocked = true;
                }
                continue;
            }
            let raw = frame.kids[frame.index].clone();
            frame.index += 1;
            let rel = if frame.rel.is_empty() {
                raw.name.clone()
            } else {
                format!("{}\\{}", frame.rel, raw.name)
            };
            let kind = if raw.attr & DIRECTORY != 0 {
                Kind::Dir
            } else {
                Kind::File
            };
            let size = if raw.attr & (REPARSE | DIRECTORY) == 0 {
                raw.size
            } else {
                0
            };
            if !raw.name_ok {
                ctx.error(rel, kind, size, ERR_INVALID_NAME);
                frame.blocked = true;
                continue;
            }
            if raw.attr & REPARSE != 0 && raw.tag & NAME_SURROGATE == 0 {
                ctx.leftover(rel, kind, size, LeftoverReason::Reparse, None);
                frame.blocked = true;
                continue;
            }
            let handle = match purge_open(&frame.handle, &raw.name, raw.attr & DIRECTORY != 0) {
                Ok(h) => h,
                Err(c) => {
                    ctx.error(rel, kind, size, c);
                    frame.blocked = true;
                    continue;
                }
            };
            let (attr, tag) = match attr_tag(&handle) {
                Some(a) => a,
                None => {
                    ctx.error(rel, kind, size, last_error());
                    frame.blocked = true;
                    continue;
                }
            };
            if file_id(&handle) != Some(raw.file_id)
                || (attr & (DIRECTORY | REPARSE)) != (raw.attr & (DIRECTORY | REPARSE))
            {
                ctx.error(rel, kind, size, 1);
                frame.blocked = true;
                continue;
            }
            if attr & REPARSE != 0 && tag & NAME_SURROGATE == 0 {
                ctx.leftover(rel, kind, size, LeftoverReason::Reparse, None);
                frame.blocked = true;
                continue;
            }
            if attr & DIRECTORY != 0 && attr & REPARSE == 0 {
                match enumerate(&handle) {
                    Ok(kids) => stack.push(PurgeFrame {
                        handle,
                        rel,
                        kids,
                        index: 0,
                        blocked: false,
                        delete_error: None,
                    }),
                    Err(_) => {
                        ctx.error(rel, kind, size, last_error());
                        frame.blocked = true;
                    }
                }
            } else {
                let freed = if attr & REPARSE != 0 {
                    Some(0)
                } else {
                    standard_info(&handle).map(|(n, links)| if links == 1 { n } else { 0 })
                };
                match freed {
                    Some(n) if set_delete(&handle) => {
                        drop(handle);
                        ctx.removed(n)?;
                    }
                    _ => {
                        ctx.error(rel, kind, size, last_error());
                        frame.blocked = true;
                    }
                }
            }
        }
        drop(stack);
        drop(chain);
        let freed = ctx.freed;
        ctx.report.freed_bytes = freed.to_string();
        if stop.is_stopped() {
            ctx.report.status = RunStatus::Stopped;
        }
        if ctx.report.dir_removed && !stop.is_stopped() {
            let (_, removed) = remove_empty_up(root, dir, root);
            ctx.report.root_removed = removed;
        } else if !ctx.report.dir_removed {
            // Remaining totals are measured without stop filtering; stop only forbids mutation.
            let remaining = inspect_quarantine(dir, &StopFlag::default())?;
            ctx.report.remaining_entries = remaining
                .file_count
                .checked_add(remaining.dir_count)
                .ok_or(CleanError::Overflow)?;
            ctx.report.remaining_bytes = remaining.total_bytes;
            if ctx
                .report
                .leftovers
                .iter()
                .any(|entry| entry.rel.is_empty())
            {
                ctx.report.remaining_entries = ctx
                    .report
                    .remaining_entries
                    .checked_add(1)
                    .ok_or(CleanError::Overflow)?;
            }
        }
        ctx.report.volume.free_after = volume_free(&drive)?.to_string();
        ctx.report.ended_at = time::format(SystemTime::now())?;
        observer.on_progress(ctx.report.removed_entries, freed);
        crate::journal_line(
            journal,
            serde_json::json!({"journal_version":"water.purge-journal.v1", "event":"done", "quarantine_dir":dir,
            "removed_entries":ctx.report.removed_entries, "freed_bytes":ctx.report.freed_bytes, "remaining_entries":ctx.report.remaining_entries, "remaining_bytes":ctx.report.remaining_bytes,
            "at":ctx.report.ended_at}),
        )?;
        Ok(report)
    }

    pub(crate) fn identity(name: &str) -> Result<Identity, CleanError> {
        let wide_name = prefixed(name);
        // SAFETY: NUL-terminated name; attribute-only read of the reparse point itself.
        let raw = unsafe {
            CreateFileW(
                wide_name.as_ptr(),
                FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS | FLAG_OPEN_REPARSE_POINT,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            return Err(CleanError::Native);
        }
        let handle = Handle(raw);
        identity_from(&handle).ok_or(CleanError::Native)
    }

    // Follow links and read attributes only: used to resolve guard and protected identities.
    pub(crate) fn resolve_follow(name: &str) -> Option<Identity> {
        let wide_name = prefixed(name);
        // SAFETY: NUL-terminated name; attribute-only, following any link to its target.
        let raw = unsafe {
            CreateFileW(
                wide_name.as_ptr(),
                FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            return None;
        }
        identity_from(&Handle(raw))
    }

    pub(crate) fn volume_free(name: &str) -> Result<u64, CleanError> {
        let wide_name = wide(name);
        let mut available = 0u64;
        // SAFETY: NUL-terminated name; only the available-bytes output is written.
        let ok = unsafe {
            GetDiskFreeSpaceExW(
                wide_name.as_ptr(),
                &mut available,
                ptr::null_mut(),
                ptr::null_mut(),
            )
        };
        if ok == 0 {
            Err(CleanError::Native)
        } else {
            Ok(available)
        }
    }

    fn last_segment(name: &str) -> &str {
        name.rsplit('\\').next().unwrap_or(name)
    }

    struct Walked {
        handle: Handle,
        attr: u32,
        tag: u32,
        file_id: [u8; 16],
        serial: u32,
        write: i64,
        size: u64,
        is_dir: bool,
        ancestors: Vec<Identity>,
    }

    // Walk from the drive root to the item, opening each level relative to the previous one
    // and never following reparse points. The item's identity is verified from its own handle.
    fn walk_item(item: &PlanItem) -> Result<Walked, SkipReason> {
        if !path::strictly_contains(&item.root, &item.path) {
            return Err(SkipReason::OutsideRoot);
        }
        let root = open_root(&item.path[..3]).ok_or(SkipReason::Changed)?;
        let mut ancestors = Vec::new();
        // The drive root is an ancestor too, so a protected path aliased onto the volume root
        // is still caught by identity.
        ancestors.push(identity_from(&root).ok_or(SkipReason::Changed)?);
        let mut parent = root;
        let segs: Vec<&str> = item.path[3..].split('\\').collect();
        for seg in &segs[..segs.len() - 1] {
            let handle = nt_open(
                &parent,
                seg,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            )
            .map_err(gone_or_changed)?;
            let (attr, _) = attr_tag(&handle).ok_or(SkipReason::Changed)?;
            if attr & REPARSE != 0 {
                return Err(SkipReason::OutsideRoot);
            }
            if attr & DIRECTORY == 0 {
                return Err(SkipReason::Changed);
            }
            // An ancestor whose identity cannot be read is treated as `changed`, never silently
            // dropped, so a protected-path alias to it is never missed.
            ancestors.push(identity_from(&handle).ok_or(SkipReason::Changed)?);
            parent = handle;
        }
        let name = segs[segs.len() - 1];
        let probe = nt_open(&parent, name, FILE_READ_ATTRIBUTES | SYNCHRONIZE, SHARE_RWD)
            .map_err(gone_or_changed)?;
        let (attr, tag) = attr_tag(&probe).ok_or(SkipReason::Changed)?;
        if attr & REPARSE != 0 {
            return Err(SkipReason::Reparse);
        }
        let is_dir = attr & DIRECTORY != 0;
        match item.kind {
            Kind::Dir if !is_dir => return Err(SkipReason::Changed),
            Kind::File if is_dir => return Err(SkipReason::Changed),
            _ => {}
        }
        let item_fid = file_id(&probe).ok_or(SkipReason::Changed)?;
        let serial = volume_serial(&probe).ok_or(SkipReason::Changed)?;
        // Step 3: identity is read from this walked handle (not a by-name open) and compared to
        // the plan; a mismatch means the object was replaced.
        let actual = Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: item_fid.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if actual != item.identity {
            return Err(SkipReason::Changed);
        }
        let write = write_time(&probe).ok_or(SkipReason::Changed)?;
        let (handle, size) = if is_dir {
            drop(probe);
            let listing = nt_open(
                &parent,
                name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            )
            .map_err(gone_or_changed)?;
            match (attr_tag(&listing), file_id(&listing)) {
                (Some((a, _)), Some(id))
                    if a & REPARSE == 0 && a & DIRECTORY != 0 && id == item_fid => {}
                _ => return Err(SkipReason::Changed),
            }
            (listing, 0)
        } else {
            let len = end_of_file(&probe).ok_or(SkipReason::Changed)?;
            (probe, len)
        };
        Ok(Walked {
            handle,
            attr,
            tag,
            file_id: item_fid,
            serial,
            write,
            size,
            is_dir,
            ancestors,
        })
    }

    fn gone_or_changed(werr: u32) -> SkipReason {
        if werr == ERR_FILE_NOT_FOUND || werr == ERR_PATH_NOT_FOUND {
            SkipReason::Gone
        } else {
            SkipReason::Changed
        }
    }

    pub(crate) fn check_chain(item: &PlanItem) -> Result<(), SkipReason> {
        walk_item(item).map(|_| ())
    }

    pub(crate) fn snapshot(
        item: &PlanItem,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        ancestor_protected_ids: &[Identity],
        unlocked_guard_ids: &[Identity],
    ) -> Result<Snapshot, SkipReason> {
        let walked = walk_item(item)?;
        // An opened ancestor (including the drive root) sharing a protected identity means the
        // item sits inside protected data.
        for ancestor in &walked.ancestors {
            if unlocked_guard_ids.contains(ancestor) {
                return Err(SkipReason::Guarded);
            }
            if ancestor_protected_ids.contains(ancestor) {
                return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
            }
        }
        let item_id = Identity {
            volume_serial: format!("{:08X}", walked.serial),
            file_id: walked.file_id.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if guard_ids.contains(&item_id) {
            return Err(SkipReason::Guarded); // CLEAN_GUARD_ID
        }
        if protected_ids.contains(&item_id) {
            return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
        }
        let item_time = filetime(walked.write).ok_or(SkipReason::Changed)?;
        let mut snap = Snapshot {
            bytes: 0,
            newest: item_time,
            entries: vec![Entry {
                rel: String::new(),
                file_id: walked.file_id,
                size: walked.size,
                mtime: item_time,
                tag: walked.tag,
                attr: walked.attr,
                name_ok: true,
            }],
        };
        if walked.is_dir {
            walk_subtree(
                &walked.handle,
                "",
                walked.serial,
                guard_ids,
                protected_ids,
                &mut snap,
            )?;
        } else {
            snap.bytes = walked.size;
        }
        snap.entries.sort_by(|a, b| a.rel.cmp(&b.rel));
        Ok(snap)
    }

    struct SnapFrame {
        handle: Handle,
        base: String,
        kids: Vec<Raw>,
        i: usize,
    }

    // Record one enumerated object on the snapshot. Returns true when the caller should open
    // that directory and walk its children. Invalid UTF-16 files/links are stored but never
    // opened; invalid-name non-reparse directories reject the item to avoid an incomplete tree.
    fn consider_entry(
        raw: &Raw,
        rel: String,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<bool, SkipReason> {
        let id = Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: raw.file_id.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if guard_ids.contains(&id) {
            return Err(SkipReason::Guarded); // CLEAN_GUARD_ID
        }
        if protected_ids.contains(&id) {
            return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
        }
        let mtime = filetime(raw.write).ok_or(SkipReason::Changed)?;
        snap.newest = snap.newest.max(mtime);
        let reparse = raw.attr & REPARSE != 0;
        if reparse {
            // CLEAN_REPARSE_KIND: only name-surrogate links are treated as links; any other
            // reparse point in the subtree poisons the whole item.
            if raw.tag & NAME_SURROGATE == 0 {
                return Err(SkipReason::Reparse);
            }
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: 0,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(false)
        } else if raw.attr & DIRECTORY != 0 {
            if !raw.name_ok {
                // A hidden subtree could contain a veto (age, reparse kind, guard/protection).
                // Without raw-name opens, skip the whole item before deleting anything.
                return Err(SkipReason::Changed);
            }
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: 0,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(true)
        } else {
            snap.bytes = snap
                .bytes
                .checked_add(raw.size)
                .ok_or(SkipReason::Changed)?;
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: raw.size,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(false)
        }
    }

    fn push_snap_dir(
        parent: *mut c_void,
        name: &str,
        rel: String,
        stack: &mut FrameStack<SnapFrame>,
    ) -> Result<(), SkipReason> {
        let child = nt_open_at(
            parent,
            name,
            FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
            SHARE_RWD,
        )
        .map_err(|_| SkipReason::Changed)?;
        let kids = enumerate(&child).map_err(|()| SkipReason::Changed)?;
        stack.push(SnapFrame {
            handle: child,
            base: rel,
            kids,
            i: 0,
        });
        Ok(())
    }

    fn walk_subtree(
        dir: &Handle,
        base: &str,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<(), SkipReason> {
        let mut stack = FrameStack::new();
        for raw in enumerate(dir).map_err(|()| SkipReason::Changed)? {
            let rel = if base.is_empty() {
                raw.name.clone()
            } else {
                format!("{base}\\{}", raw.name)
            };
            if consider_entry(&raw, rel.clone(), serial, guard_ids, protected_ids, snap)? {
                push_snap_dir(dir.0, &raw.name, rel, &mut stack)?;
            }
            drain_snap(&mut stack, serial, guard_ids, protected_ids, snap)?;
        }
        Ok(())
    }

    fn drain_snap(
        stack: &mut FrameStack<SnapFrame>,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<(), SkipReason> {
        while !stack.is_empty() {
            let top = stack.len() - 1;
            if stack[top].i >= stack[top].kids.len() {
                stack.pop();
                continue;
            }
            let raw = stack[top].kids[stack[top].i].clone();
            stack[top].i += 1;
            let rel = if stack[top].base.is_empty() {
                raw.name.clone()
            } else {
                format!("{}\\{}", stack[top].base, raw.name)
            };
            let parent = stack[top].handle.0;
            if consider_entry(&raw, rel.clone(), serial, guard_ids, protected_ids, snap)? {
                push_snap_dir(parent, &raw.name, rel, stack)?;
            }
        }
        Ok(())
    }

    enum ParentErr {
        Gone,
        Other,
    }

    // Walk only to the item's parent, rejecting any reparse ancestor. Error 2 or 3 at any
    // level means the item is no longer under that name (CLEAN-M2 section 3).
    fn walk_parent(item: &PlanItem) -> Result<Handle, ParentErr> {
        let mut parent = open_root(&item.path[..3]).ok_or(ParentErr::Other)?;
        let segs: Vec<&str> = item.path[3..].split('\\').collect();
        for seg in &segs[..segs.len() - 1] {
            let handle = match nt_open(
                &parent,
                seg,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(h) => h,
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    return Err(ParentErr::Gone);
                }
                Err(_) => return Err(ParentErr::Other),
            };
            match attr_tag(&handle) {
                Some((a, _)) if a & REPARSE == 0 && a & DIRECTORY != 0 => {}
                _ => return Err(ParentErr::Other),
            }
            parent = handle;
        }
        Ok(parent)
    }

    // Use one Unicode simple-uppercase character per input character, without full-uppercase
    // expansions such as sharp-s -> SS. Greek iota-subscript letters have a simple mapping even
    // though char::to_uppercase expands them (UnicodeData.txt, Unicode 17.0.0).
    fn uppercase_key(rel: &str) -> String {
        rel.chars()
            .map(|c| match c {
                '\u{1f80}'..='\u{1f87}' | '\u{1f90}'..='\u{1f97}' | '\u{1fa0}'..='\u{1fa7}' => {
                    char::from_u32(c as u32 + 8).unwrap_or(c)
                }
                '\u{1fb3}' => '\u{1fbc}',
                '\u{1fc3}' => '\u{1fcc}',
                '\u{1ff3}' => '\u{1ffc}',
                _ => {
                    let mut upper = c.to_uppercase();
                    let first = upper.next().unwrap_or(c);
                    if upper.next().is_none() { first } else { c }
                }
            })
            .collect()
    }

    struct DelFrame {
        handle: Handle,
        rel: String,
        index: usize,
        kids: Vec<Raw>,
        kid_i: usize,
    }

    struct DelCtx<'a> {
        before: &'a Snapshot,
        stop: &'a StopFlag,
        folded: Vec<(String, usize)>,
        accounted: Vec<bool>,
        deleted: Vec<bool>,
        removed: u64,
        failed: u64,
        bytes_freed: u64,
        changed_stop: bool,
        initial_skip: Option<SkipReason>,
        // Set once the item itself (rel == "") has had its delete flag set this run. The residual
        // pass needs it: if a different object later bears the item's name, the item may only be
        // reported deleted when this run actually deleted it (addendum 02 change 4).
        self_deleted: bool,
    }

    impl DelCtx<'_> {
        // Prefer the exact spelling; only a miss uses the simple-uppercase index. Both searches
        // are logarithmic (M2's cache directories hold on the order of 1e5 entries).
        fn expected_index(&self, rel: &str) -> Option<usize> {
            self.before
                .entries
                .binary_search_by(|e| e.rel.as_str().cmp(rel))
                .ok()
                .or_else(|| {
                    let key = uppercase_key(rel);
                    self.folded
                        .binary_search_by(|(k, _)| k.cmp(&key))
                        .ok()
                        .map(|i| self.folded[i].1)
                })
        }

        fn fail(&mut self, index: usize) {
            if !self.accounted[index] {
                self.failed += 1;
                self.accounted[index] = true;
            }
        }

        // After a directory's own delete fails, report whether the failure is expected rather
        // than counted. At least one entry must remain, and each must be new or an already
        // accounted-for snapshot entry. Empty directories and enumeration errors still count.
        fn only_post_snapshot(&self, handle: &Handle, rel: &str) -> bool {
            match enumerate(handle) {
                Ok(rest) => {
                    !rest.is_empty()
                        && rest.iter().all(|k| {
                            let crel = if rel.is_empty() {
                                k.name.clone()
                            } else {
                                format!("{rel}\\{}", k.name)
                            };
                            self.expected_index(&crel)
                                .is_none_or(|i| self.accounted[i] || self.deleted[i])
                        })
                }
                Err(()) => false,
            }
        }

        fn note_removed(&mut self, index: usize, rel: &str) {
            self.removed += 1;
            self.deleted[index] = true;
            if rel.is_empty() {
                self.self_deleted = true;
            }
        }

        fn finish_dir(&mut self, handle: &Handle, rel: &str, index: usize) {
            if set_delete(handle) {
                self.note_removed(index, rel);
            } else if self.only_post_snapshot(handle, rel) {
                self.accounted[index] = true;
            } else {
                self.fail(index);
            }
        }

        // Open one snapshot node relative to its parent. Directories return a frame so the
        // caller can walk children on an explicit stack; files and links are finished here.
        fn enter(&mut self, parent: *mut c_void, name: &str, rel: &str) -> Option<DelFrame> {
            if self.changed_stop {
                return None;
            }
            let Some(index) = self.expected_index(rel) else {
                return None; // appeared after the snapshot: never touched, never counted
            };
            let exp = &self.before.entries[index];
            if !exp.name_ok {
                self.fail(index);
                return None;
            }
            let (exp_attr, exp_tag, exp_id) = (exp.attr, exp.tag, exp.file_id);
            if exp_attr & REPARSE != 0 {
                if exp_tag & NAME_SURROGATE != 0 {
                    // CLEAN_NO_FOLLOW / H1: re-verify from the link's own handle that it is still
                    // a name-surrogate reparse point with the snapshot FileId before deleting, so
                    // a concurrently swapped-in non-link reparse point (e.g. a cloud placeholder)
                    // is never removed.
                    match nt_open_at(
                        parent,
                        name,
                        DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                        SHARE_RWD,
                    ) {
                        Ok(handle) => match (attr_tag(&handle), file_id(&handle)) {
                            (Some((a, t)), Some(id))
                                if a & REPARSE != 0 && t & NAME_SURROGATE != 0 && id == exp_id =>
                            {
                                if set_delete(&handle) {
                                    self.note_removed(index, rel);
                                } else {
                                    self.fail(index);
                                }
                            }
                            _ => self.fail(index),
                        },
                        Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {}
                        Err(_) => self.fail(index),
                    }
                }
                return None;
            }
            if exp_attr & DIRECTORY != 0 {
                let handle = match nt_open_at(
                    parent,
                    name,
                    FILE_LIST_DIRECTORY | DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                    SHARE_RW,
                ) {
                    Ok(h) => h,
                    Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                        if rel.is_empty() {
                            self.initial_skip = Some(SkipReason::Gone);
                        }
                        return None;
                    }
                    Err(_) => {
                        self.fail(index);
                        return None;
                    }
                };
                let id = file_id(&handle);
                if rel.is_empty() && id.is_some_and(|id| id != exp_id) {
                    self.initial_skip = Some(SkipReason::Changed);
                    return None;
                }
                match (attr_tag(&handle), id) {
                    (Some((a, _)), Some(id))
                        if a & REPARSE == 0 && a & DIRECTORY != 0 && id == exp_id => {}
                    _ => {
                        // The directory was replaced under us: stop deleting this item.
                        self.fail(index);
                        self.changed_stop = true;
                        return None;
                    }
                }
                let kids = match enumerate(&handle) {
                    Ok(k) => k,
                    Err(()) => {
                        // Enumeration failed (not "no more files"): stop this item, count one.
                        self.fail(index);
                        self.changed_stop = true;
                        return None;
                    }
                };
                return Some(DelFrame {
                    handle,
                    rel: rel.to_owned(),
                    index,
                    kids,
                    kid_i: 0,
                });
            }
            match nt_open_at(
                parent,
                name,
                DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(handle) => {
                    let id = file_id(&handle);
                    if rel.is_empty() && id.is_some_and(|id| id != exp_id) {
                        self.initial_skip = Some(SkipReason::Changed);
                        return None;
                    }
                    match (attr_tag(&handle), id) {
                        (Some((a, _)), Some(id)) if a & REPARSE == 0 && id == exp_id => {
                            let Some((size, links)) = standard_info(&handle) else {
                                self.fail(index);
                                return None;
                            };
                            if set_delete(&handle) {
                                self.note_removed(index, rel);
                                if links <= 1 {
                                    self.bytes_freed = self.bytes_freed.saturating_add(size);
                                }
                            } else {
                                self.fail(index);
                            }
                        }
                        // became a link, FileId changed, or attributes unreadable: skip + count
                        _ => self.fail(index),
                    }
                }
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    if rel.is_empty() {
                        self.initial_skip = Some(SkipReason::Gone);
                    }
                }
                Err(_) => self.fail(index),
            }
            None
        }

        // Delete one node by handle, relative to its parent. Returns true if a stop cut it short.
        fn delete_from(&mut self, parent: &Handle, name: &str, rel: &str) -> bool {
            let mut stack = FrameStack::new();
            if let Some(frame) = self.enter(parent.0, name, rel) {
                stack.push(frame);
            }
            while !stack.is_empty() {
                if self.changed_stop {
                    return false;
                }
                let top = stack.len() - 1;
                if stack[top].kid_i < stack[top].kids.len() {
                    if self.stop.is_stopped() {
                        return true;
                    }
                    let kid = stack[top].kids[stack[top].kid_i].clone();
                    stack[top].kid_i += 1;
                    let child_rel = if stack[top].rel.is_empty() {
                        kid.name.clone()
                    } else {
                        format!("{}\\{}", stack[top].rel, kid.name)
                    };
                    let parent_ptr = stack[top].handle.0;
                    // Copy the snapshot FileId and kind out before any &mut self call so the
                    // immutable borrow of `before.entries` does not span enter().
                    let expected = self.expected_index(&child_rel).map(|i| {
                        let e = &self.before.entries[i];
                        (i, e.file_id, e.attr)
                    });
                    match expected {
                        None => {} // appeared after the snapshot: ignore, not a failure
                        Some((_, fid, _)) if fid == kid.file_id => {
                            if let Some(frame) = self.enter(parent_ptr, &kid.name, &child_rel) {
                                stack.push(frame);
                                // The recursive call used to finish this subtree before
                                // the parent's post-child stop checkpoint.
                                continue;
                            }
                            if self.changed_stop {
                                return false;
                            }
                        }
                        Some((i, _, snap_attr)) => {
                            // Name is in the snapshot but the object was replaced (FileId
                            // differs). Decide by the SNAPSHOT entry's kind (SPEC section 4 step
                            // 7), never by the live object that replaced it.
                            self.fail(i);
                            if snap_attr & DIRECTORY != 0 && snap_attr & REPARSE == 0 {
                                // The snapshot entry was a non-reparse directory: stop this item
                                // whatever replaced it (dir, file, link, or other reparse point).
                                self.changed_stop = true;
                                return false;
                            }
                            // The snapshot entry was a file or link: do not delete it, count one
                            // failure, and continue with the next sibling.
                        }
                    }
                    if self.stop.is_stopped() {
                        return true;
                    }
                } else {
                    let frame = stack.pop().unwrap();
                    if self.changed_stop {
                        return false;
                    }
                    self.finish_dir(&frame.handle, &frame.rel, frame.index);
                    drop(frame);
                    // After a child directory finishes (including its own delete), the parent
                    // checks the stop flag before the next sibling or its own delete — same
                    // checkpoints as the previous recursive node().
                    if !stack.is_empty() && self.stop.is_stopped() {
                        return true;
                    }
                }
            }
            false
        }
    }

    struct MeasFrame {
        handle: Handle,
        kids: Vec<Raw>,
        i: usize,
    }

    fn measure_kid(
        parent: *mut c_void,
        kid: &Raw,
        total: &mut u64,
        stack: &mut FrameStack<MeasFrame>,
    ) -> Option<()> {
        if kid.attr & REPARSE != 0 {
            return Some(());
        }
        if kid.attr & DIRECTORY != 0 {
            if !kid.name_ok {
                // Its subtree cannot be measured without opening the raw name.
                return None;
            }
            match nt_open_at(
                parent,
                &kid.name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(child) => {
                    let kids = enumerate(&child).ok()?;
                    stack.push(MeasFrame {
                        handle: child,
                        kids,
                        i: 0,
                    });
                    Some(())
                }
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => Some(()),
                Err(_) => None,
            }
        } else {
            *total = total.saturating_add(kid.size);
            Some(())
        }
    }

    // Residual bytes still under the item, measured by handle. None means the item is gone; the
    // second field is 1 when the residual could not be determined (never report deleted then).
    fn measure_tree(dir: &Handle) -> Option<u64> {
        let mut total = 0u64;
        let mut stack = FrameStack::new();
        for kid in enumerate(dir).ok()? {
            measure_kid(dir.0, &kid, &mut total, &mut stack)?;
            drain_measure(&mut total, &mut stack)?;
        }
        Some(total)
    }

    fn drain_measure(total: &mut u64, stack: &mut FrameStack<MeasFrame>) -> Option<()> {
        while !stack.is_empty() {
            let top = stack.len() - 1;
            if stack[top].i >= stack[top].kids.len() {
                stack.pop();
                continue;
            }
            let kid = stack[top].kids[stack[top].i].clone();
            stack[top].i += 1;
            let parent = stack[top].handle.0;
            measure_kid(parent, &kid, total, stack)?;
        }
        Some(())
    }

    fn measure_residual(
        item: &PlanItem,
        before: &Snapshot,
        self_deleted: bool,
    ) -> (Option<u64>, u64) {
        let before_bytes = before.bytes;
        // The original name may be missing or belong to a different FileId. Only this run's
        // successful delete flag on the item proves it was deleted; otherwise its residual is
        // unknown because someone else moved/replaced it. Keep bytes_before without an extra
        // failure (CLEAN-REPORT section 2, extending addendum 02 change 4).
        let replaced = |sd: bool| -> (Option<u64>, u64) {
            if sd {
                (None, 0)
            } else {
                (Some(before_bytes), 0)
            }
        };
        let Some(item_entry) = before.entries.iter().find(|e| e.rel.is_empty()) else {
            return (Some(before_bytes), 1);
        };
        let parent = match walk_parent(item) {
            Ok(p) => p,
            Err(ParentErr::Gone) => return replaced(self_deleted),
            Err(ParentErr::Other) => return (Some(before_bytes), 1),
        };
        let name = last_segment(&item.path);
        // Open the item itself relative to the parent handle (never following a link) and
        // identify it by FileId, instead of listing the parent and matching names.
        let handle = match nt_open(&parent, name, FILE_READ_ATTRIBUTES | SYNCHRONIZE, SHARE_RWD) {
            Ok(h) => h,
            Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                return replaced(self_deleted);
            }
            Err(_) => return (Some(before_bytes), 1),
        };
        let Some((attr, _)) = attr_tag(&handle) else {
            return (Some(before_bytes), 1);
        };
        let Some(id) = file_id(&handle) else {
            return (Some(before_bytes), 1);
        };
        if id != item_entry.file_id {
            return replaced(self_deleted);
        }
        if attr & DIRECTORY != 0 {
            match nt_open(
                &parent,
                name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                // addendum 02 change 2: the handle used to sum the residual must itself carry the
                // snapshot FileId, so a swap between the two opens cannot make a different
                // object's tree read as the item's residual.
                Ok(dir) => match file_id(&dir) {
                    Some(did) if did == item_entry.file_id => match measure_tree(&dir) {
                        Some(bytes) => (Some(bytes), 0),
                        None => (Some(before_bytes), 1),
                    },
                    Some(_) => replaced(self_deleted),
                    None => (Some(before_bytes), 1),
                },
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    replaced(self_deleted)
                }
                Err(_) => (Some(before_bytes), 1),
            }
        } else {
            match end_of_file(&handle) {
                Some(len) => (Some(len), 0),
                None => (Some(before_bytes), 1),
            }
        }
    }

    pub(crate) fn delete(
        item: &PlanItem,
        before: &Snapshot,
        stop: &StopFlag,
    ) -> (ItemReport, bool) {
        let mut folded: Vec<(String, usize)> = before
            .entries
            .iter()
            .enumerate()
            .map(|(i, e)| (uppercase_key(&e.rel), i))
            .collect();
        folded.sort_unstable();
        let mut ctx = DelCtx {
            before,
            stop,
            folded,
            accounted: vec![false; before.entries.len()],
            deleted: vec![false; before.entries.len()],
            removed: 0,
            failed: 0,
            bytes_freed: 0,
            changed_stop: false,
            initial_skip: None,
            self_deleted: false,
        };
        let mut interrupted = false;
        match walk_parent(item) {
            Ok(parent) => {
                interrupted = ctx.delete_from(&parent, last_segment(&item.path), "");
            }
            Err(ParentErr::Gone) => {
                let mut report = ItemReport::skipped(item, SkipReason::Gone);
                report.bytes_before = before.bytes.to_string();
                return (report, false);
            }
            Err(ParentErr::Other) => {}
        }
        if let Some(reason) = ctx.initial_skip {
            let mut report = ItemReport::skipped(item, reason);
            report.bytes_before = before.bytes.to_string();
            return (report, false);
        }
        let (residual, extra_failed) = measure_residual(item, before, ctx.self_deleted);
        ctx.failed += extra_failed;
        let status = if residual.is_none() {
            ItemStatus::Deleted
        } else if ctx.removed > 0 {
            ItemStatus::Partial
        } else if interrupted {
            ItemStatus::Skipped
        } else {
            ItemStatus::Failed
        };
        let cancelled = status == ItemStatus::Skipped;
        let report = ItemReport {
            item_id: item.item_id.clone(),
            status,
            skip_reason: cancelled.then_some(SkipReason::Cancelled),
            bytes_before: before.bytes.to_string(),
            bytes_freed: ctx.bytes_freed.to_string(),
            failed_entries: ctx.failed,
        };
        (report, interrupted)
    }
}
