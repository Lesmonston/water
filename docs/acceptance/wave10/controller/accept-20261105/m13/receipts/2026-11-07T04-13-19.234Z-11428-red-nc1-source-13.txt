#![forbid(unsafe_code)]

fn separator(b: u8) -> bool {
    matches!(b, b'-' | b'_' | b'.' | b' ')
}

/// Fixed M3.5 suffix grammar, independent of directory contents.
pub fn name_template(name: &str, is_file: bool) -> Option<String> {
    let (stem, extension) = if is_file {
        name.rfind('.')
            .filter(|i| *i > 0)
            .and_then(|i| {
                let ext = &name[i + 1..];
                ((1..=8).contains(&ext.len()) && ext.bytes().all(|b| b.is_ascii_alphanumeric()))
                    .then_some((&name[..i], Some(ext)))
            })
            .unwrap_or((name, None))
    } else {
        (name, None)
    };
    let b = stem.as_bytes();
    let mut found = None;
    if b.len() >= 36 {
        let start = b.len() - 36;
        if let Some(tail) = stem.get(start..)
            && tail.bytes().enumerate().all(|(i, c)| {
                if [8, 13, 18, 23].contains(&i) {
                    c == b'-'
                } else {
                    c.is_ascii_hexdigit()
                }
            })
            && (start == 0 || separator(b[start - 1]))
        {
            found = Some((start, "<u>"));
        }
    }
    if found.is_none() {
        let mut start = b.len();
        while start > 0 && (b[start - 1].is_ascii_digit() || separator(b[start - 1])) {
            start -= 1;
        }
        while start < b.len() && separator(b[start]) {
            start += 1;
        }
        if b[start..].iter().filter(|c| c.is_ascii_digit()).count() >= 8 {
            found = Some((start, "<d>"));
        }
    }
    if found.is_none()
        && b.len() == 11
        && b.starts_with(b"tmp")
        && b[3..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
        && b[3..].iter().any(|c| c.is_ascii_digit() || *c == b'_')
    {
        found = Some((3, "<p8>"));
    }
    if found.is_none() {
        let start = stem.rfind(['-', '_', '.', ' ']).map_or(0, |i| i + 1);
        let tail = &b[start..];
        if tail.len() >= 7
            && tail.iter().all(|c| c.is_ascii_hexdigit())
            && tail.iter().any(|c| c.is_ascii_digit())
            && tail
                .iter()
                .any(|c| matches!(c.to_ascii_lowercase(), b'a'..=b'f'))
        {
            found = Some((start, "<h>"));
        } else if tail.len() == 6
            && tail.iter().all(|c| c.is_ascii_alphanumeric())
            && tail
                .iter()
                .any(|c| c.is_ascii_digit() || c.is_ascii_lowercase())
        {
            found = Some((start, "<r6>"));
        }
    }
    let (start, token) = found?;
    let prefix = &stem[..start];
    if prefix.bytes().filter(|b| b.is_ascii_alphabetic()).count() < 2 {
        return None;
    }
    let mut result = String::new();
    let mut digit = false;
    for c in prefix.chars() {
        if c.is_ascii_digit() {
            if !digit {
                result.push('#');
            }
            digit = true;
        } else {
            digit = false;
            result.push(c.to_ascii_lowercase());
        }
    }
    result.push_str(token);
    if let Some(ext) = extension {
        result.push('.');
        result.push_str(&ext.to_ascii_lowercase());
    }
    Some(result)
}
