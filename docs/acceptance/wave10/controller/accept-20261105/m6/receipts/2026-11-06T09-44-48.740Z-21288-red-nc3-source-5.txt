const invalid = () => ({code:'invalid', message_zh:'所选内容无法清理，请重新选择。'});
function id(value) {
  if (typeof value !== 'string' || !/^[A-Za-z0-9][A-Za-z0-9:._-]{0,255}$/.test(value) || (/^[A-Za-z]:/.test(value) && !/^(m:[a-z0-9._-]+|r:[a-f0-9]{16})$/.test(value))) throw invalid();
  return value;
}
function token(value) {
  if (typeof value !== 'string' || !value.length || value.length > 512 || value.includes('\\')) throw invalid();
  return value;
}
async function invoke(command, build = () => ({})) {
  const host = (globalThis.window ?? globalThis).__TAURI__;
  if (typeof host?.core?.invoke !== 'function') throw {code:'unavailable'};
  return host.core.invoke(command, build());
}

export const clean_prepare = (groupIds, excludeItemIds) => invoke('clean_prepare', () => {
  if (groupIds && !Array.isArray(groupIds) && typeof groupIds === 'object') {
    if (hasKeys(groupIds, ['conclusionVersion', 'chosenMode'])) return {conclusionVersion:version(groupIds.conclusionVersion), chosenMode:oneOf(groupIds.chosenMode, ['quarantine', 'delete'])};
    if (hasKeys(groupIds, ['dedupeId', 'dedupeVersion'])) return {dedupeId:id(groupIds.dedupeId), dedupeVersion:version(groupIds.dedupeVersion)};
    throw invalid();
  }
  if (!Array.isArray(groupIds) || !groupIds.length) throw invalid();
  if (excludeItemIds === undefined) return {groupIds:groupIds.map(id)};
  if (!Array.isArray(excludeItemIds) || excludeItemIds.length > 5000 || new Set(excludeItemIds).size !== excludeItemIds.length) throw invalid();
  return {groupIds:groupIds.map(id), excludeItemIds:excludeItemIds.map(id)};
});

function hasKeys(value, keys) { return value && typeof value === 'object' && !Array.isArray(value) && Object.keys(value).length === keys.length && keys.every(k => Object.hasOwn(value, k)); }
function bool(value) { if (typeof value !== 'boolean') throw invalid(); return value; }
function oneOf(value, allowed) { if (!allowed.includes(value)) throw invalid(); return value; }
function version(value) { if (typeof value !== 'string' || !/^[A-Za-z0-9:._-]{1,64}$/.test(value)) throw invalid(); return value; }
function ids(values, max) { if (!Array.isArray(values) || !values.length || values.length > max || new Set(values).size !== values.length) throw invalid(); return values.map(id); }
function query(value) { if (value === null) return value; if (typeof value !== 'string' || !value.length || [...value].length > 64 || /[\\/:\u0000-\u001f\u007f]/.test(value)) throw invalid(); return value; }
function decision(value) {
  if (hasKeys(value, ['group', 'choice'])) return {group:id(value.group), choice:oneOf(value.choice, ['clean', 'keep', 'undecided'])};
  for (const kind of ['category', 'row', 'item']) if (hasKeys(value, [kind, 'include'])) return {[kind]:id(value[kind]), include:bool(value.include)};
  throw invalid();
}
