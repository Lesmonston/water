const REPARSE: u32 = 0x400;
const DIRECTORY: u32 = 0x10;

#[derive(Debug, Clone)]
pub struct ScanOptions {
    pub max_entries: u64,
    pub max_seconds: u64,
}

impl Default for ScanOptions {
    fn default() -> Self {
        Self {
            max_entries: 20_000_000,
            max_seconds: 900,
        }
    }
}

#[derive(Clone)]
pub struct CancelToken {
    flag: Arc<AtomicBool>,
}

impl CancelToken {
    pub fn new() -> Self {
        Self {
            flag: Arc::new(AtomicBool::new(false)),
        }
    }

    pub fn cancel(&self) {
        self.flag.store(true, Ordering::SeqCst);
    }

    pub fn is_cancelled(&self) -> bool {
        self.flag.load(Ordering::SeqCst)
    }
}

impl Default for CancelToken {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ScanProgress {
    pub group_id: String,
    pub category: String,
    pub visited_entries: u64,
    pub found_bytes: u64,
}

pub trait ScanObserver: Sync {
    fn on_progress(&self, progress: &ScanProgress);
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Kind {
    File,
    Dir,
    Reparse,
}

#[derive(Clone)]
struct Child {
    name: String,
    path: String,
    kind: Kind,
    len: u64,
    mtime: SystemTime,
    creation: SystemTime,
    allocation: Option<u64>,
    reparse_tag: u32,
    identity: Option<(u32, [u8; 16])>,
}

#[derive(Clone)]
struct Listing {
    children: Vec<Child>,
    issues: Vec<&'static str>,
}

#[derive(Clone)]
struct Hit {
    target_index: usize,
    target_id: String,
    group_id: String,
    level: Level,
    min_age_hours: u64,
    keep_newest: u64,
    path: String,
    kind: Kind,
    total_bytes: u64,
    unique_bytes: u64,
    file_count: u64,
    newest_activity: SystemTime,
    oldest_activity: SystemTime,
    /// `.git` check done on the entry's own listing, before any stop.
    has_git: bool,
    /// `None` means identity lookup failed (not that it was skipped).
    identity: Option<Identity>,
    disk_bytes: u64,
    git_is_copy: bool,
    keep_by: KeepBy,
    unlocks_target_ids: Vec<String>,
    valuable_count: u64,
    valuable_names: Vec<String>,
    odd_count: u64,
    odd_names: Vec<String>,
}

struct Engine<'a> {
    options: &'a ScanOptions,
    cancel: &'a CancelToken,
    observer: &'a dyn ScanObserver,
    now: SystemTime,
    started: Instant,
    visited: u64,
    found_bytes: u64,
    status: &'static str,
    issues: Vec<Issue>,
    last_progress: Instant,
    cache: HashMap<String, Arc<Listing>>,
    extras_options: &'a ExtrasOptions,
    extras: ScanExtras,
    include_quarantine: bool,
    search_dirs: BTreeMap<String, (String, Category, bool)>,
}

impl<'a> Engine<'a> {
    fn stopped(&mut self, group_id: &str) -> bool {
        if self.cancel.is_cancelled() {
            self.status = "cancelled";
            return true;
        }
        if self.visited > self.options.max_entries
            || self.started.elapsed() > Duration::from_secs(self.options.max_seconds)
        {
            // Recorded once per group, not once per call.
            if !self
                .issues
                .iter()
                .any(|issue| issue.code == "budget_exhausted" && issue.group_id == group_id)
            {
                self.add_issue("budget_exhausted", group_id);
            }
            if self.status != "cancelled" {
                self.status = "partial";
            }
            true
        } else {
            false
        }
    }

    fn add_issue(&mut self, code: &str, group_id: &str) {
        if let Some(issue) = self
            .issues
            .iter_mut()
            .find(|issue| issue.code == code && issue.group_id == group_id)
        {
            issue.count += 1;
        } else {
            self.issues.push(Issue {
                code: code.to_string(),
                group_id: group_id.to_string(),
                count: 1,
            });
        }
        if self.status == "completed"
            && matches!(
                code,
                "access_denied" | "unrepresentable_name" | "budget_exhausted"
            )
        {
            self.status = "partial";
        }
    }

    fn emit(&mut self, group_id: &str, category: Category, force: bool) {
        if !force && self.last_progress.elapsed() < Duration::from_millis(50) {
            return;
        }
        self.observer.on_progress(&ScanProgress {
            group_id: group_id.to_string(),
            category: category_name(category).to_string(),
            visited_entries: self.visited,
            found_bytes: self.found_bytes,
        });
        self.last_progress = Instant::now();
    }

    fn listing(&mut self, dir: &str, group_id: &str) -> Listing {
        let key = dir.to_ascii_lowercase();
        if let Some(cached) = self.cache.get(&key) {
            return cached.as_ref().clone();
        }
        let mut listing = Listing {
            children: Vec::new(),
            issues: Vec::new(),
        };
        if self.stopped(group_id) {
            return listing;
        }
        let result = native_windows::list_directory(dir, |entry| {
            self.visited = self.visited.saturating_add(1);
            if self.stopped(group_id) {
                return false;
            }
            let Some(name) = entry.name else {
                listing.issues.push("unrepresentable_name");
                return true;
            };
            let path = child_path(dir, &name);
            listing.children.push(Child {
                name,
                path,
                kind: kind_from_attributes(entry.attributes),
                len: entry.len,
                mtime: clamp_time(entry.mtime),
                creation: clamp_time(entry.creation),
                allocation: Some(entry.allocation),
                reparse_tag: entry.reparse_tag,
                identity: entry.identity,
            });
            true
        });
        if result.is_err() {
            listing.issues.push("access_denied");
        }
        for child in &listing.children {
            self.index_junction(child);
        }
        if !self.include_quarantine {
            listing.children.retain(|c| {
                !(c.kind == Kind::Dir && c.name.eq_ignore_ascii_case(".water-quarantine"))
            });
        }
        self.cache.insert(key, Arc::new(listing.clone()));
        listing
    }

    fn cached_child(&self, path: &str) -> Option<Child> {
        let (parent, _) = path.rsplit_once('\\')?;
        let parent = if parent.len() == 2 {
            format!("{parent}\\")
        } else {
            parent.to_string()
        };
        self.cache
            .get(&parent.to_ascii_lowercase())?
            .children
            .iter()
            .find(|c| path_eq(&c.path, path))
            .cloned()
    }

    fn child(&self, path: &str) -> Option<Child> {
        self.cached_child(path).or_else(|| classify_path(path))
    }

    fn index_junction(&mut self, child: &Child) {
        if !self.extras_options.junctions
            || child.kind != Kind::Reparse
            || self
                .extras
                .junctions
                .iter()
                .any(|j| path_eq(&j.path, &child.path))
        {
            return;
        }
        let supported = matches!(child.reparse_tag, 0xa0000003 | 0xa000000c)
            || (child.reparse_tag == 0
                && fs::symlink_metadata(&child.path).is_ok_and(|m| m.file_type().is_symlink()));
        if !supported {
            return;
        }
        if self.extras.junctions.len() >= 10_000 {
            self.extras.junctions_truncated = true;
            return;
        }
        let decoded = native_windows::reparse_data(&child.path)
            .ok()
            .and_then(|data| crate::extras::decode_reparse(&child.path, &data));
        let (tag, target) = decoded.unwrap_or_else(|| {
            (
                if child.reparse_tag == 0xa000000c {
                    "symlink".into()
                } else {
                    "junction".into()
                },
                None,
            )
        });
        self.extras.junctions.push(Junction {
            path: child.path.clone(),
            tag,
            target,
            final_target: None,
        });
    }

    fn listing_issues(&mut self, listing: &Listing, group_id: &str) {
        for code in &listing.issues {
            self.add_issue(code, group_id);
        }
    }

    fn list(&mut self, dir: &str, group_id: &str) -> Vec<Child> {
        let listing = self.listing(dir, group_id);
        self.listing_issues(&listing, group_id);
        listing.children
    }
}

fn kind_from_attributes(attrs: u32) -> Kind {
    if attrs & REPARSE != 0 {
        Kind::Reparse
    } else if attrs & DIRECTORY != 0 {
        Kind::Dir
    } else {
        Kind::File
    }
}

fn child_from_meta(name: String, path: String, meta: &Metadata) -> Child {
    #[cfg(windows)]
    let attrs = meta.file_attributes();
    #[cfg(not(windows))]
    let attrs = if meta.is_symlink() {
        REPARSE
    } else if meta.is_dir() {
        DIRECTORY
    } else {
        0
    };
    Child {
        name,
        path,
        kind: kind_from_attributes(attrs),
        len: meta.len(),
        mtime: clamp_time(meta.modified().unwrap_or(SystemTime::UNIX_EPOCH)),
        creation: clamp_time(meta.created().unwrap_or(SystemTime::UNIX_EPOCH)),
        allocation: None,
        reparse_tag: 0,
        identity: None,
    }
}

fn classify_path(path: &str) -> Option<Child> {
    let meta = fs::symlink_metadata(path).ok()?;
    Some(child_from_meta(
        file_name(path).to_string(),
        path.to_string(),
        &meta,
    ))
}

fn category_name(category: Category) -> &'static str {
    match category {
        Category::AgentTool => "agent_tool",
        Category::Browser => "browser",
        Category::SystemTemp => "system_temp",
        Category::AppLeftover => "app_leftover",
        Category::DevCache => "dev_cache",
        Category::CrashLog => "crash_log",
        Category::GpuSystem => "gpu_system",
        Category::Custom => "custom",
        Category::Unrecognized => "unrecognized",
    }
}

fn accept_type(item_type: ItemType, kind: Kind) -> bool {
    match (item_type, kind) {
        (_, Kind::Reparse) => false,
        (ItemType::Any, _) => true,
        (ItemType::Dir, Kind::Dir) => true,
        (ItemType::File, Kind::File) => true,
        _ => false,
    }
}

fn ignored_root(path: &str) -> bool {
    path.split('\\')
        .any(|p| p.eq_ignore_ascii_case(".water-quarantine"))
}

fn odd_name(name: &str) -> bool {
    let first = name.split('.').next().unwrap_or("").to_ascii_uppercase();
    name.ends_with(['.', ' '])
        || ["CON", "PRN", "AUX", "NUL"].contains(&first.as_str())
        || (first.len() == 4
            && (first.starts_with("COM") || first.starts_with("LPT"))
            && matches!(first.as_bytes()[3], b'1'..=b'9'))
}

fn valuable(child: &Child) -> bool {
    if child.kind != Kind::File {
        return false;
    }
    let path = child.path.to_ascii_lowercase();
    let parent = path.rsplit_once('\\').map_or("", |(p, _)| p);
    if parent
        .split('\\')
        .any(|p| ["node_modules", "site-packages", "dist-packages"].contains(&p))
    {
        return false;
    }
    let name = child.name.to_ascii_lowercase();
    if ["thumbs.db", "ehthumbs.db"].contains(&name.as_str()) {
        return false;
    }
    name.starts_with(".env")
        || name.starts_with("credentials")
        || name == "google-services.json"
        || [
            "*.jks",
            "*.keystore",
            "*.p12",
            "*.pfx",
            "*.p8",
            "*.pem",
            "*.key",
            "*.mobileprovision",
            "*.db",
            "*.sqlite*",
        ]
        .iter()
        .any(|p| match_segment(p, &name))
}

fn retain_name(names: &mut Vec<String>, value: String) {
    names.push(value);
    names.sort_by_key(|s| s.to_ascii_lowercase());
    names.truncate(8);
}

fn observe_name(measured: &mut Measured, child: &Child, root: &str) {
    let relative = if path_eq(root, &child.path) {
        child.name.clone()
    } else {
        child
            .path
            .get(root.len()..)
            .unwrap_or(&child.name)
            .trim_start_matches('\\')
            .to_string()
    };
    if valuable(child) {
        measured.valuable_count += 1;
        retain_name(&mut measured.valuable_names, relative.clone());
    }
    if odd_name(&child.name) {
        measured.odd_count += 1;
        retain_name(&mut measured.odd_names, relative);
    }
}

fn accumulate(
    measured: &mut Measured,
    child: &Child,
    root: &str,
    identities: &mut HashSet<(u32, [u8; 16])>,
) {
    measured.newest = measured.newest.max(child.mtime);
    measured.oldest = measured.oldest.min(child.mtime);
    let latest = child.mtime.max(child.creation);
    measured.latest_newest = measured.latest_newest.max(latest);
    measured.latest_oldest = measured.latest_oldest.min(latest);
    if child
        .path
        .rsplit_once('\\')
        .is_some_and(|(parent, _)| path_eq(parent, root))
        && child.name.eq_ignore_ascii_case(".git")
    {
        measured.has_git = true;
    }
    observe_name(measured, child, root);
    if child.kind == Kind::File {
        measured.bytes = measured.bytes.saturating_add(child.len);
        measured.files += 1;
        if child.identity.is_none_or(|id| identities.insert(id)) {
            measured.unique_bytes = measured.unique_bytes.saturating_add(child.len);
            measured.disk_bytes = measured
                .disk_bytes
                .saturating_add(child.allocation.unwrap_or(0));
        }
    }
}

fn measure(engine: &mut Engine<'_>, path: &str, kind: Kind, group_id: &str) -> Measured {
    let Some(mut self_child) = engine.child(path) else {
        engine.add_issue("access_denied", group_id);
        return Measured::empty(SystemTime::UNIX_EPOCH, SystemTime::UNIX_EPOCH);
    };
    let mut measured = Measured::empty(self_child.mtime, self_child.creation);
    let mut identities = HashSet::new();
    if kind == Kind::File {
        if self_child.allocation.is_none() {
            self_child.allocation =
                Some(native_windows::allocation_of(path).unwrap_or_else(|_| {
                    engine.add_issue("access_denied", group_id);
                    0
                }));
        }
        accumulate(&mut measured, &self_child, path, &mut identities);
        return measured;
    }
    observe_name(&mut measured, &self_child, path);
    let mut stack = vec![path.to_string()];
    while let Some(dir) = stack.pop() {
        if engine.stopped(group_id) {
            break;
        }
        for child in engine.list(&dir, group_id) {
            accumulate(&mut measured, &child, path, &mut identities);
            match child.kind {
                Kind::File => {}
                Kind::Dir => stack.push(child.path),
                Kind::Reparse => {}
            }
        }
    }
    measured
}

#[derive(Clone)]
struct Measured {
    bytes: u64,
    unique_bytes: u64,
    files: u64,
    newest: SystemTime,
    oldest: SystemTime,
    has_git: bool,
    disk_bytes: u64,
    latest_newest: SystemTime,
    latest_oldest: SystemTime,
    valuable_count: u64,
    valuable_names: Vec<String>,
    odd_count: u64,
    odd_names: Vec<String>,
}

impl Measured {
    fn empty(mtime: SystemTime, creation: SystemTime) -> Self {
        Self {
            bytes: 0,
            unique_bytes: 0,
            files: 0,
            newest: mtime,
            oldest: mtime,
            has_git: false,
            disk_bytes: 0,
            latest_newest: mtime.max(creation),
            latest_oldest: mtime.max(creation),
            valuable_count: 0,
            valuable_names: vec![],
            odd_count: 0,
            odd_names: vec![],
        }
    }
}

fn push_hit(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    target: &Target,
    index: usize,
    path: &str,
    kind: Kind,
) {
    if engine.stopped(&target.group_id) {
        return;
    }
    let measured = measure(engine, path, kind, &target.group_id);
    push_measured_hit(engine, hits, target, index, path, kind, measured);
}

#[allow(clippy::too_many_arguments)]
fn push_measured_hit(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    target: &Target,
    index: usize,
    path: &str,
    kind: Kind,
    measured: Measured,
) {
    // A stop during the walk leaves size, newest_activity and the .git check
    // incomplete: the entry is not an obtained result and is not output.
    if engine.stopped(&target.group_id) {
        return;
    }
    let identity = if matches!(
        &target.match_spec,
        Match::Marker {
            marker: Marker::FileName { .. },
            ..
        }
    ) {
        engine.cached_child(path).and_then(|child| {
            child.identity.map(|(serial, id)| Identity {
                volume_serial: format!("{serial:08X}"),
                file_id: id.iter().map(|b| format!("{b:02x}")).collect(),
            })
        })
    } else {
        finish_identity(path).ok()
    };
    if engine.stopped(&target.group_id) {
        return;
    }
    push_complete_hit(engine, hits, target, index, path, kind, measured, identity);
}

#[allow(clippy::too_many_arguments)]
fn push_complete_hit(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    target: &Target,
    index: usize,
    path: &str,
    kind: Kind,
    measured: Measured,
    identity: Option<Identity>,
) {
    engine.found_bytes = engine.found_bytes.saturating_add(measured.bytes);
    hits.push(Hit {
        target_index: index,
        target_id: target.target_id.clone(),
        group_id: target.group_id.clone(),
        level: target.level,
        min_age_hours: target.min_age_hours,
        keep_newest: target.keep_newest,
        path: path.to_string(),
        kind,
        total_bytes: measured.bytes,
        unique_bytes: measured.unique_bytes,
        file_count: measured.files,
        newest_activity: if target.age_basis == AgeBasis::Modified {
            measured.newest
        } else {
            measured.latest_newest
        },
        oldest_activity: if target.age_basis == AgeBasis::Modified {
            measured.oldest
        } else {
            measured.latest_oldest
        },
        has_git: measured.has_git,
        identity,
        disk_bytes: measured.disk_bytes,
        git_is_copy: target.git_is_copy,
        keep_by: target.keep_by,
        unlocks_target_ids: target.unlocks_target_ids.clone(),
        valuable_count: measured.valuable_count,
        valuable_names: measured.valuable_names,
        odd_count: measured.odd_count,
        odd_names: measured.odd_names,
    });
}

fn walk_pattern(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    target: &Target,
    index: usize,
    current: &str,
    segments: &[String],
) {
    if engine.stopped(&target.group_id) || segments.is_empty() {
        return;
    }
    let last = segments.len() == 1;
    let children = engine.list(current, &target.group_id);
    let parent: Vec<_> = children
        .iter()
        .map(|c| ParentEntry {
            name: c.name.clone(),
            kind: c.kind,
            len: c.len,
        })
        .collect();
    for child in children {
        if child.name.eq_ignore_ascii_case(".git") {
            continue;
        }
        if !match_segment(&segments[0], &child.name) {
            continue;
        }
        if child.kind == Kind::Reparse {
            engine.add_issue("reparse_skipped", &target.group_id);
            continue;
        }
        if last {
            if accept_type(target.item_type, child.kind) {
                let measured = measure(engine, &child.path, child.kind, &target.group_id);
                let conditions = match &target.match_spec {
                    Match::Pattern { conditions, .. } => conditions.as_slice(),
                    _ => &[],
                };
                if conditions_match(
                    &child.path,
                    conditions,
                    Some(&parent),
                    None,
                    Some(measured.bytes),
                ) {
                    push_measured_hit(
                        engine,
                        hits,
                        target,
                        index,
                        &child.path,
                        child.kind,
                        measured,
                    );
                }
            }
        } else if child.kind == Kind::Dir {
            walk_pattern(engine, hits, target, index, &child.path, &segments[1..]);
        }
    }
}

fn marker_matches(dir: &str, marker: &Marker, children: &[Child]) -> bool {
    match marker {
        Marker::DirName { names } => names
            .iter()
            .any(|want| file_name(dir).eq_ignore_ascii_case(want)),
        Marker::DirNamePattern { patterns } => {
            patterns.iter().any(|p| match_segment(p, file_name(dir)))
        }
        Marker::FileName { .. } => false,
        Marker::File {
            name,
            min_bytes,
            max_bytes,
        } => children.iter().any(|child| {
            child.kind == Kind::File
                && child.name.eq_ignore_ascii_case(name)
                && child.len >= *min_bytes
                && child.len <= *max_bytes
        }),
        Marker::ChildrenAll { names } => names.iter().all(|want| {
            children
                .iter()
                .any(|child| child.name.eq_ignore_ascii_case(want))
        }),
    }
}

struct ParentEntry {
    name: String,
    kind: Kind,
    len: u64,
}

fn conditions_match(
    dir: &str,
    conditions: &[Condition],
    parent: Option<&[ParentEntry]>,
    marker: Option<&Marker>,
    bytes: Option<u64>,
) -> bool {
    if conditions.is_empty() {
        return true;
    }
    let components: Vec<_> = dir
        .split('\\')
        .skip(1)
        .filter(|part| !part.is_empty())
        .collect();
    conditions.iter().all(|condition| match condition {
        Condition::ParentHasFile { name } => parent.is_some_and(|entries| {
            entries
                .iter()
                .any(|entry| entry.kind == Kind::File && entry.name.eq_ignore_ascii_case(name))
        }),
        Condition::ParentHasOtherEntries { except } => parent.is_some_and(|entries| {
            entries.iter().any(|entry| {
                !entry.name.eq_ignore_ascii_case(file_name(dir))
                    && !except
                        .iter()
                        .any(|name| entry.name.eq_ignore_ascii_case(name))
            })
        }),
        Condition::ParentHasOtherMatch => parent.is_some_and(|entries| {
            entries.iter().any(|entry| {
                !entry.name.eq_ignore_ascii_case(file_name(dir))
                    && match marker {
                        Some(Marker::DirName { names }) => {
                            entry.kind == Kind::Dir
                                && names.iter().any(|n| entry.name.eq_ignore_ascii_case(n))
                        }
                        Some(Marker::DirNamePattern { patterns }) => {
                            entry.kind == Kind::Dir
                                && patterns.iter().any(|p| match_segment(p, &entry.name))
                        }
                        Some(Marker::FileName { patterns }) => {
                            entry.kind == Kind::File
                                && patterns.iter().any(|p| match_segment(p, &entry.name))
                        }
                        _ => false,
                    }
            })
        }),
        Condition::ParentHasFileMatching { pattern } => parent.is_some_and(|entries| {
            entries
                .iter()
                .any(|e| e.kind == Kind::File && match_segment(pattern, &e.name))
        }),
        Condition::ParentLacksFileMatching { pattern } => parent.is_some_and(|entries| {
            !entries
                .iter()
                .any(|e| e.kind == Kind::File && match_segment(pattern, &e.name))
        }),
        Condition::SiblingWithoutSuffix { suffixes } => parent.is_some_and(|entries| {
            let name = file_name(dir);
            suffixes.iter().any(|suffix| {
                name.to_ascii_lowercase()
                    .ends_with(&suffix.to_ascii_lowercase())
                    && name
                        .get(..name.len().saturating_sub(suffix.len()))
                        .is_some_and(|plain| {
                            entries.iter().any(|e| {
                                e.kind == Kind::File
                                    && e.name.eq_ignore_ascii_case(plain)
                                    && bytes.is_none_or(|size| e.len >= size)
                            })
                        })
            })
        }),
        Condition::NoAncestorSuffix { suffixes } => components
            [..components.len().saturating_sub(1)]
            .iter()
            .all(|part| {
                !suffixes.iter().any(|suffix| {
                    part.to_ascii_lowercase()
                        .ends_with(&suffix.to_ascii_lowercase())
                })
            }),
        Condition::PathHasAny { names } => components.iter().any(|part| {
            names.iter().any(|name| {
                part.to_ascii_lowercase()
                    .contains(&name.to_ascii_lowercase())
            })
        }),
    })
}

struct Pending {
    // The primary owns traversal, listing issues and budget accounting even
    // when its hit is routed to a fallback (or discarded).
    index: usize,
    hit_index: Option<usize>,
    path: String,
    measured: Measured,
    identities: HashSet<(u32, [u8; 16])>,
    parent: Option<Arc<Vec<ParentEntry>>>,
    issue_group: String,
}

enum MarkerFrame {
    Enter {
        dir: Child,
        depth: u32,
        searching: Vec<usize>,
        measuring: Vec<usize>,
        parent: Option<Arc<Vec<ParentEntry>>>,
    },
    Exit(Vec<usize>),
}

/// A single physical DFS serves all marker targets on this root. Each target
/// stops searching independently; open measurements keep accumulating through
/// the subtree (including .git), without listing any directory a second time.
fn walk_markers(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    plan: &ScanPlan,
    indices: Vec<usize>,
    fallbacks: &[Option<usize>],
) {
    let Match::Marker { root, .. } = &plan.targets[indices[0]].match_spec else {
        return;
    };
    if ignored_root(root) || file_name(root).eq_ignore_ascii_case(".git") {
        return;
    }
    let dir = match engine.child(root) {
        Some(dir) if dir.kind == Kind::Dir => dir,
        other => {
            if let Some(child) = &other {
                engine.index_junction(child);
            }
            let code = if other.is_some_and(|dir| dir.kind == Kind::Reparse) {
                "base_reparse"
            } else {
                "base_missing"
            };
            for index in indices {
                engine.add_issue(code, &plan.targets[index].group_id);
                if let Some(fallback) = fallbacks[index] {
                    engine.add_issue(code, &plan.targets[fallback].group_id);
                }
            }
            return;
        }
    };
    let mut pending: Vec<Option<Pending>> = Vec::new();
    let mut stack = vec![MarkerFrame::Enter {
        dir,
        depth: 0,
        searching: indices,
        measuring: Vec::new(),
        parent: None,
    }];
    while let Some(frame) = stack.pop() {
        match frame {
            MarkerFrame::Exit(ids) => {
                for id in ids {
                    let row = pending[id].take().expect("active measurement");
                    let index = final_marker_owner(&row, plan, fallbacks);
                    route_issues(
                        engine,
                        &row.issue_group,
                        &plan.targets[index.unwrap_or(row.index)].group_id,
                    );
                    let Some(index) = index else {
                        continue;
                    };
                    let target = &plan.targets[index];
                    push_measured_hit(
                        engine,
                        hits,
                        target,
                        index,
                        &row.path,
                        Kind::Dir,
                        row.measured,
                    );
                }
            }
            MarkerFrame::Enter {
                dir,
                depth,
                searching,
                mut measuring,
                parent,
            } => {
                let primary_group = searching
                    .first()
                    .map(|i| plan.targets[*i].group_id.clone())
                    .unwrap_or_else(|| {
                        pending[measuring[0]]
                            .as_ref()
                            .expect("active measurement")
                            .issue_group
                            .clone()
                    });
                if engine.stopped(&primary_group) {
                    break;
                }
                let needs_listing = !measuring.is_empty()
                    || searching.iter().any(|index| {
                        let Match::Marker {
                            marker, max_depth, ..
                        } = &plan.targets[*index].match_spec
                        else {
                            return false;
                        };
                        if matches!(marker, Marker::FileName { .. }) {
                            depth < *max_depth
                        } else {
                            !matches!(
                                marker,
                                Marker::DirName { .. } | Marker::DirNamePattern { .. }
                            ) || depth < *max_depth
                                || marker_matches(&dir.path, marker, &[])
                        }
                    });
                let listing_group = format!("@listing:{}", dir.path);
                let listing = if needs_listing {
                    engine.listing(&dir.path, &listing_group)
                } else {
                    Listing {
                        children: Vec::new(),
                        issues: Vec::new(),
                    }
                };
                let listing_budget = engine
                    .issues
                    .iter()
                    .any(|i| i.group_id == listing_group && i.code == "budget_exhausted");
                engine.issues.retain(|i| i.group_id != listing_group);
                // Replay the old logical listing issues per target, while visited
                // counts only the one actual enumeration above.
                for id in &measuring {
                    let row = pending[*id].as_ref().expect("active measurement");
                    engine.listing_issues(&listing, &row.issue_group);
                }
                let mut searching_next = Vec::new();
                let mut started_here = Vec::new();
                for index in searching {
                    let target = &plan.targets[index];
                    let Match::Marker {
                        marker,
                        max_depth,
                        conditions,
                        ..
                    } = &target.match_spec
                    else {
                        continue;
                    };
                    if matches!(marker, Marker::FileName { .. }) {
                        if depth < *max_depth {
                            engine.listing_issues(&listing, &target.group_id);
                            searching_next.push(index);
                        }
                        continue;
                    }
                    if marker_matches(&dir.path, marker, &listing.children) {
                        let id = pending.len();
                        let issue_group = format!("@measure:{id}");
                        engine.listing_issues(&listing, &issue_group);
                        if !matches!(
                            marker,
                            Marker::DirName { .. } | Marker::DirNamePattern { .. }
                        ) {
                            engine.listing_issues(&listing, &issue_group);
                        }
                        let mut measured = Measured::empty(dir.mtime, dir.creation);
                        observe_name(&mut measured, &dir, &dir.path);
                        pending.push(Some(Pending {
                            index,
                            hit_index: if conditions_match(
                                &dir.path,
                                conditions,
                                parent.as_deref().map(Vec::as_slice),
                                Some(marker),
                                None,
                            ) {
                                Some(index)
                            } else {
                                fallbacks[index]
                            },
                            path: dir.path.clone(),
                            identities: HashSet::new(),
                            measured,
                            parent: parent.clone(),
                            issue_group,
                        }));
                        started_here.push(id);
                        measuring.push(id);
                    } else if depth < *max_depth {
                        if !matches!(
                            marker,
                            Marker::DirName { .. } | Marker::DirNamePattern { .. }
                        ) {
                            engine.listing_issues(&listing, &target.group_id);
                        }
                        engine.listing_issues(&listing, &target.group_id);
                        searching_next.push(index);
                    } else if !matches!(
                        marker,
                        Marker::DirName { .. } | Marker::DirNamePattern { .. }
                    ) {
                        engine.listing_issues(&listing, &target.group_id);
                    }
                }
                let mut stopped = false;
                if listing_budget && searching_next.is_empty() && measuring.is_empty() {
                    engine.add_issue("budget_exhausted", &primary_group);
                }
                for index in searching_next.iter().copied().chain(
                    measuring
                        .iter()
                        .map(|id| pending[*id].as_ref().expect("active measurement").index),
                ) {
                    let group = measuring
                        .iter()
                        .filter_map(|id| pending[*id].as_ref())
                        .find(|row| row.index == index)
                        .map_or_else(
                            || plan.targets[index].group_id.clone(),
                            |row| row.issue_group.clone(),
                        );
                    stopped |= engine.stopped(&group);
                }
                if stopped {
                    break;
                } // Discard every unfinished measurement.
                stack.push(MarkerFrame::Exit(started_here));
                if measuring.is_empty()
                    && searching_next
                        .iter()
                        .any(|i| plan.targets[*i].category == Category::Custom)
                {
                    engine
                        .search_dirs
                        .entry(dir.path.to_ascii_lowercase())
                        .or_insert((dir.path.clone(), Category::Custom, false));
                    if engine.extras_options.declarations && depth <= 4 {
                        for c in &listing.children {
                            if c.kind == Kind::File
                                && c.name.eq_ignore_ascii_case(".water-cleanup.json")
                            {
                                engine.extras.declarations.push(c.path.clone());
                            }
                        }
                    }
                }
                let file_parent: Vec<_> = listing
                    .children
                    .iter()
                    .map(|c| ParentEntry {
                        name: c.name.clone(),
                        kind: c.kind,
                        len: c.len,
                    })
                    .collect();
                for index in &searching_next {
                    let target = &plan.targets[*index];
                    if let Match::Marker {
                        marker: Marker::FileName { patterns },
                        conditions,
                        ..
                    } = &target.match_spec
                    {
                        for c in &listing.children {
                            if c.kind == Kind::File
                                && !c.name.eq_ignore_ascii_case(".git")
                                && patterns.iter().any(|p| match_segment(p, &c.name))
                                && conditions_match(
                                    &c.path,
                                    conditions,
                                    Some(&file_parent),
                                    None,
                                    Some(c.len),
                                )
                            {
                                push_hit(engine, hits, target, *index, &c.path, Kind::File);
                            } else if c.kind == Kind::File
                                && !c.name.eq_ignore_ascii_case(".git")
                                && patterns.iter().any(|p| match_segment(p, &c.name))
                                && let Some(fallback) = fallbacks[*index]
                            {
                                push_hit(
                                    engine,
                                    hits,
                                    &plan.targets[fallback],
                                    fallback,
                                    &c.path,
                                    Kind::File,
                                );
                            }
                        }
                    }
                }
                // Keep only the parent names/kinds needed by children. This
                // reuses the listing above; conditions never enumerate again.
                let parent_entries = Some(Arc::new(file_parent));
                for child in listing.children {
                    for id in &measuring {
                        let row = pending[*id].as_mut().expect("active measurement");
                        accumulate(&mut row.measured, &child, &row.path, &mut row.identities);
                    }
                    if child.kind == Kind::Reparse {
                        for index in &searching_next {
                            engine.add_issue("reparse_skipped", &plan.targets[*index].group_id);
                        }
                    } else if child.kind == Kind::Dir {
                        let next = if child.name.eq_ignore_ascii_case(".git") {
                            Vec::new()
                        } else {
                            searching_next.clone()
                        };
                        if !next.is_empty() || !measuring.is_empty() {
                            stack.push(MarkerFrame::Enter {
                                dir: child,
                                depth: depth + 1,
                                searching: next,
                                measuring: measuring.clone(),
                                parent: parent_entries.clone(),
                            });
                        }
                    }
                }
            }
        }
    }
    for row in pending.into_iter().flatten() {
        let index = final_marker_owner(&row, plan, fallbacks).unwrap_or(row.index);
        route_issues(engine, &row.issue_group, &plan.targets[index].group_id);
    }
}

fn final_marker_owner(
    row: &Pending,
    plan: &ScanPlan,
    fallbacks: &[Option<usize>],
) -> Option<usize> {
    let Match::Marker {
        conditions, marker, ..
    } = &plan.targets[row.index].match_spec
    else {
        return row.hit_index;
    };
    if conditions_match(
        &row.path,
        conditions,
        row.parent.as_deref().map(Vec::as_slice),
        Some(marker),
        Some(row.measured.bytes),
    ) {
        Some(row.index)
    } else {
        fallbacks[row.index]
    }
}

fn route_issues(engine: &mut Engine<'_>, from: &str, to: &str) {
    let issues: Vec<_> = engine
        .issues
        .iter()
        .filter(|i| i.group_id == from)
        .cloned()
        .collect();
    engine.issues.retain(|i| i.group_id != from);
    for issue in issues {
        if issue.code == "budget_exhausted" {
            if !engine
                .issues
                .iter()
                .any(|i| i.code == issue.code && i.group_id == to)
            {
                engine.add_issue(&issue.code, to);
            }
            continue;
        }
        for _ in 0..issue.count {
            engine.add_issue(&issue.code, to);
        }
    }
}

fn run_target(engine: &mut Engine<'_>, hits: &mut Vec<Hit>, target: &Target, index: usize) {
    if let Match::Pattern { base, .. } = &target.match_spec
        && (ignored_root(base) || file_name(base).eq_ignore_ascii_case(".git"))
    {
        return;
    }
    match &target.match_spec {
        Match::Pattern { base, segments, .. } => match if ignored_root(base) {
            None
        } else {
            engine.child(base)
        } {
            None => engine.add_issue("base_missing", &target.group_id),
            Some(child) if child.kind == Kind::Reparse => {
                engine.index_junction(&child);
                engine.add_issue("base_reparse", &target.group_id);
            }
            Some(child) if child.kind != Kind::Dir => {
                engine.add_issue("base_missing", &target.group_id);
            }
            Some(_) => walk_pattern(engine, hits, target, index, base, segments),
        },
        Match::Marker { .. } => {}
        Match::Discovery { .. } => {}
    }
}

fn run_discovery(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    target: &Target,
    index: usize,
    others: &[Hit],
) {
    let Match::Discovery { root } = &target.match_spec else {
        return;
    };
    if ignored_root(root) || file_name(root).eq_ignore_ascii_case(".git") {
        return;
    }
    match engine.child(root) {
        None => {
            engine.add_issue("base_missing", &target.group_id);
            return;
        }
        Some(child) if child.kind == Kind::Reparse => {
            engine.index_junction(&child);
            engine.add_issue("base_reparse", &target.group_id);
            return;
        }
        Some(child) if child.kind != Kind::Dir => {
            engine.add_issue("base_missing", &target.group_id);
            return;
        }
        Some(_) => {}
    }
    let protected: Vec<&str> = others
        .iter()
        .filter(|hit| hit.level == Level::Protected)
        .map(|hit| hit.path.as_str())
        .collect();
    let other_paths: Vec<&str> = others.iter().map(|hit| hit.path.as_str()).collect();
    engine.search_dirs.insert(
        root.to_ascii_lowercase(),
        (root.clone(), target.category, true),
    );
    for child in engine.list(root, &target.group_id) {
        if child.name.eq_ignore_ascii_case(".git") {
            continue;
        }
        if child.kind == Kind::Reparse {
            engine.add_issue("reparse_skipped", &target.group_id);
            continue;
        }
        if !accept_type(target.item_type, child.kind) {
            continue;
        }
        // SCAN §3: skip a child that equals another entry or is its ancestor.
        let covered = other_paths
            .iter()
            .any(|path| path_eq(path, &child.path) || is_proper_ancestor(&child.path, path));
        let guarded = protected
            .iter()
            .any(|path| path_eq(path, &child.path) || is_proper_ancestor(&child.path, path));
        if covered || guarded {
            continue;
        }
        push_hit(engine, hits, target, index, &child.path, child.kind);
    }
}

fn merge_hits(mut hits: Vec<Hit>) -> Vec<Hit> {
    hits.sort_by(|a, b| {
        a.path
            .to_ascii_lowercase()
            .cmp(&b.path.to_ascii_lowercase())
            .then(a.target_index.cmp(&b.target_index))
    });
    let mut out = Vec::new();
    for hit in hits {
        if let Some(existing) = out
            .iter_mut()
            .find(|item: &&mut Hit| path_eq(&item.path, &hit.path))
        {
            let take = hit.level.rank() > existing.level.rank()
                || (hit.level.rank() == existing.level.rank()
                    && hit.target_index < existing.target_index);
            if take {
                *existing = hit;
            }
        } else {
            out.push(hit);
        }
    }
    out
}

fn too_recent(now: SystemTime, newest: SystemTime, min_age_hours: u64) -> bool {
    match now.duration_since(newest) {
        Ok(age) => age < Duration::from_secs(min_age_hours.saturating_mul(3600)),
        Err(_) => true,
    }
}

fn classify_reason(
    hit: &Hit,
    engine: &Engine<'_>,
    all: &[Hit],
    hard_guards: &[String],
) -> Option<String> {
    if !hit.level.cleanable() {
        return Some("level_not_cleanable".to_string());
    }
    if hard_guards
        .iter()
        .any(|guard| path_eq(&hit.path, guard) || is_proper_ancestor(&hit.path, guard))
    {
        return Some("hard_guard".to_string());
    }
    let protected: Vec<&Hit> = all
        .iter()
        .filter(|item| item.level == Level::Protected)
        .collect();
    if protected
        .iter()
        .any(|item| is_proper_ancestor(&hit.path, &item.path))
    {
        return Some("contains_protected".to_string());
    }
    if protected.iter().any(|item| {
        is_proper_ancestor(&item.path, &hit.path)
            && true
    }) {
        return Some("inside_protected".to_string());
    }
    if hit.kind == Kind::Dir && hit.has_git && !hit.git_is_copy {
        return Some("repository".to_string());
    }
    if too_recent(engine.now, hit.newest_activity, hit.min_age_hours) {
        return Some("too_recent".to_string());
    }
    None
}

fn apply_keep_newest(rows: &mut [(Hit, Option<String>)]) {
    let mut groups: Vec<(usize, u64, Option<String>)> = Vec::new();
    for (hit, reason) in rows.iter() {
        let parent = (hit.keep_by == KeepBy::VersionName).then(|| {
            hit.path
                .rsplit_once('\\')
                .map_or("", |(p, _)| p)
                .to_ascii_lowercase()
        });
        if reason.is_none()
            && hit.keep_newest > 0
            && !groups
                .iter()
                .any(|(i, _, p)| *i == hit.target_index && *p == parent)
        {
            groups.push((hit.target_index, hit.keep_newest, parent));
        }
    }
    for (target_index, keep, parent) in groups {
        let mut idxs: Vec<usize> = rows
            .iter()
            .enumerate()
            .filter(|(_, (hit, reason))| {
                hit.target_index == target_index
                    && reason.is_none()
                    && parent.as_ref().is_none_or(|p| {
                        hit.path
                            .rsplit_once('\\')
                            .is_some_and(|(value, _)| value.eq_ignore_ascii_case(p))
                    })
            })
            .map(|(i, _)| i)
            .collect();
        idxs.sort_by(|a, b| {
            if parent.is_some() {
                return version_cmp(file_name(&rows[*b].0.path), file_name(&rows[*a].0.path));
            }
            rows[*b]
                .0
                .newest_activity
                .cmp(&rows[*a].0.newest_activity)
                .then(
                    rows[*a]
                        .0
                        .path
                        .to_ascii_lowercase()
                        .cmp(&rows[*b].0.path.to_ascii_lowercase()),
                )
        });
        for idx in idxs.into_iter().take(keep as usize) {
            rows[idx].1 = Some("keep_newest".to_string());
        }
    }
}

fn version_cmp(a: &str, b: &str) -> std::cmp::Ordering {
    let numbers = |name: &str| {
        name.split(|c: char| !c.is_ascii_digit())
            .filter(|p| !p.is_empty())
            .map(|p| p.trim_start_matches('0').to_string())
            .collect::<Vec<_>>()
    };
    let left = numbers(a);
    let right = numbers(b);
    for (x, y) in left.iter().zip(&right) {
        let order = x.len().cmp(&y.len()).then(x.cmp(y));
        if !order.is_eq() {
            return order;
        }
    }
    left.len()
        .cmp(&right.len())
        .then(a.to_ascii_lowercase().cmp(&b.to_ascii_lowercase()))
}

fn top_keys(items: &[&Hit]) -> Vec<TopKey> {
    let mut map: Vec<(String, u64, u64, SystemTime)> = Vec::new();
    for item in items {
        let key = normalize_key(file_name(&item.path));
        if let Some(row) = map.iter_mut().find(|(k, _, _, _)| k == &key) {
            row.1 += 1;
            row.2 = row.2.saturating_add(item.total_bytes);
            if item.newest_activity > row.3 {
                row.3 = item.newest_activity;
            }
        } else {
            map.push((key, 1, item.total_bytes, item.newest_activity));
        }
    }
    map.sort_by(|a, b| b.2.cmp(&a.2).then(a.0.cmp(&b.0)));
    map.into_iter()
        .take(10)
        .map(|(key, count, bytes, newest)| TopKey {
            key,
            count,
            bytes: format_bytes(bytes),
            newest_activity: format_rfc3339(newest),
        })
        .collect()
}

fn finish_identity(path: &str) -> Result<Identity, ScanError> {
    native_windows::identity_of(path)
}

#[allow(clippy::too_many_arguments)]
pub fn scan(
    plan: &ScanPlan,
    scan_id: &str,
    options: &ScanOptions,
    cancel: &CancelToken,
    observer: &dyn ScanObserver,
    now: SystemTime,
) -> Result<ScanResult, ScanError> {
    scan_full(
        plan,
        scan_id,
        options,
        &ExtrasOptions::default(),
        cancel,
        observer,
        now,
    )
    .map(|pair| pair.0)
}

pub fn scan_full(
    plan: &ScanPlan,
    scan_id: &str,
    options: &ScanOptions,
    extras: &ExtrasOptions,
    cancel: &CancelToken,
    observer: &dyn ScanObserver,
    now: SystemTime,
) -> Result<(ScanResult, ScanExtras), ScanError> {
    let mut engine = Engine {
        options,
        cancel,
        observer,
        now,
        started: Instant::now(),
        visited: 0,
        found_bytes: 0,
        status: "completed",
        issues: Vec::new(),
        last_progress: Instant::now() - Duration::from_millis(50),
        cache: HashMap::new(),
        extras_options: extras,
        include_quarantine: false,
        extras: ScanExtras {
            extras_version: "water.scan-extras.v1".into(),
            scan_id: scan_id.into(),
            junctions: vec![],
            junctions_truncated: false,
            declarations: vec![],
        },
        search_dirs: BTreeMap::new(),
    };
    let mut hits = Vec::new();
    let mut marker_roots = HashSet::new();
    let mut fallbacks = vec![None; plan.targets.len()];
    for (index, target) in plan.targets.iter().enumerate() {
        if let Match::Marker {
            fallback_for: Some(reference),
            ..
        } = &target.match_spec
            && let Some(main) = plan.targets.iter().position(|t| &t.target_id == reference)
        {
            fallbacks[main] = Some(index);
        }
    }
    if !engine.stopped("") {
        for (index, target) in plan.targets.iter().enumerate() {
            if matches!(
                &target.match_spec,
                Match::Marker {
                    fallback_for: Some(_),
                    ..
                }
            ) {
                continue;
            }
            if !matches!(target.match_spec, Match::Discovery { .. }) {
                if let Match::Marker { root, .. } = &target.match_spec {
                    if marker_roots.insert(root.to_ascii_lowercase()) {
                        let indices = plan.targets.iter().enumerate().filter_map(|(i, t)| {
                            matches!(&t.match_spec, Match::Marker { root: other, fallback_for: None, .. } if path_eq(root, other)).then_some(i)
                        }).collect();
                        walk_markers(&mut engine, &mut hits, plan, indices, &fallbacks);
                    }
                } else {
                    run_target(&mut engine, &mut hits, target, index);
                }
                engine.emit(&target.group_id, target.category, true);
            }
            if engine.status == "cancelled" {
                break;
            }
        }
    }
    if engine.status != "cancelled" {
        for (index, target) in plan.targets.iter().enumerate() {
            if matches!(target.match_spec, Match::Discovery { .. }) {
                let others = hits.clone();
                run_discovery(&mut engine, &mut hits, target, index, &others);
                engine.emit(&target.group_id, target.category, true);
            }
            if engine.status == "cancelled" {
                break;
            }
        }
    }
    let guard_hits = hits.clone();
    let mut expanded = plan.clone();
    let clusters = if extras.leak_clusters {
        build_clusters(&mut engine, &mut hits, &mut expanded)
    } else {
        BTreeMap::new()
    };
    let plan = &expanded;
    crate::extras::resolve_junctions(&mut engine.extras);
    engine
        .extras
        .declarations
        .sort_by_key(|p| p.to_ascii_lowercase());
    engine.extras.declarations.dedup_by(|a, b| path_eq(a, b));
    engine.extras.declarations.truncate(256);
    let merged = merge_hits(hits);
    let mut guards = merged.clone();
    guards.extend(
        guard_hits
            .into_iter()
            .filter(|h| h.level == Level::Protected),
    );
    let mut rows: Vec<(Hit, Option<String>)> = merged
        .iter()
        .map(|hit| {
            let reason = classify_reason(hit, &engine, &guards, &plan.hard_guards);
            (hit.clone(), reason)
        })
        .collect();
    apply_keep_newest(&mut rows);
    let mut items = Vec::new();
    for (offset, (hit, mut reason)) in rows.into_iter().enumerate() {
        let identity = match hit.identity.clone() {
            Some(id) => Some(id),
            None => {
                if hit.level.cleanable() {
                    engine.add_issue("identity_unavailable", &hit.group_id);
                }
                if reason.is_none() {
                    reason = Some("identity_unavailable".to_string());
                }
                None
            }
        };
        if identity.is_none()
            && reason.as_deref() != Some("identity_unavailable")
            && reason.is_none()
        {
            reason = Some("identity_unavailable".to_string());
        }
        items.push((
            hit,
            Item {
                item_id: format!("i-{:06}", offset + 1),
                group_id: String::new(),
                path: String::new(),
                kind: String::new(),
                total_bytes: String::new(),
                unique_bytes: String::new(),
                file_count: 0,
                newest_activity: String::new(),
                identity,
                eligible: reason.is_none(),
                ineligible_reason: reason,
                disk_bytes: None,
                template: None,
                valuable_count: 0,
                valuable_names: vec![],
                odd_count: 0,
                odd_names: vec![],
                referenced_count: 0,
                referenced_by: vec![],
            },
        ));
    }
    items.sort_by(|a, b| {
        let ia = plan
            .targets
            .iter()
            .position(|t| t.group_id == a.0.group_id)
            .unwrap_or(usize::MAX);
        let ib = plan
            .targets
            .iter()
            .position(|t| t.group_id == b.0.group_id)
            .unwrap_or(usize::MAX);
        ia.cmp(&ib).then(
            a.0.path
                .to_ascii_lowercase()
                .cmp(&b.0.path.to_ascii_lowercase()),
        )
    });
    let mut out_items = Vec::new();
    for (seq, (hit, mut item)) in items.into_iter().enumerate() {
        item.item_id = format!("i-{:06}", seq + 1);
        item.group_id = hit.group_id.clone();
        item.path = hit.path.clone();
        item.kind = match hit.kind {
            Kind::File => "file",
            _ => "dir",
        }
        .to_string();
        item.total_bytes = format_bytes(hit.total_bytes);
        item.unique_bytes = format_bytes(hit.unique_bytes);
        item.file_count = hit.file_count;
        item.newest_activity = format_rfc3339(hit.newest_activity);
        item.disk_bytes = Some(format_bytes(hit.disk_bytes));
        item.template = name_template(file_name(&hit.path), hit.kind == Kind::File);
        item.valuable_count = hit.valuable_count;
        item.valuable_names = hit.valuable_names.clone();
        item.odd_count = hit.odd_count;
        item.odd_names = hit.odd_names.clone();
        for junction in &engine.extras.junctions {
            if is_proper_ancestor(&item.path, &junction.path) || path_eq(&item.path, &junction.path)
            {
                continue;
            }
            if junction.final_target.as_ref().is_some_and(|p| {
                path_eq(p, &item.path)
                    || is_proper_ancestor(p, &item.path)
                    || is_proper_ancestor(&item.path, p)
            }) {
                item.referenced_count += 1;
                if item.referenced_by.len() < 8 {
                    item.referenced_by.push(junction.path.clone());
                }
            }
        }
        out_items.push((hit, item));
    }
    let mut groups = Vec::new();
    let mut seen = Vec::new();
    for target in &plan.targets {
        if seen.iter().any(|id: &String| id == &target.group_id) {
            continue;
        }
        seen.push(target.group_id.clone());
        let members: Vec<&Hit> = out_items
            .iter()
            .filter(|(hit, _)| hit.group_id == target.group_id)
            .map(|(hit, _)| hit)
            .collect();
        let eligible: Vec<&Hit> = out_items
            .iter()
            .filter(|(hit, item)| hit.group_id == target.group_id && item.eligible)
            .map(|(hit, _)| hit)
            .collect();
        let total_bytes: u64 = members.iter().map(|hit| hit.total_bytes).sum();
        let eligible_bytes: u64 = eligible.iter().map(|hit| hit.total_bytes).sum();
        let file_count: u64 = members.iter().map(|hit| hit.file_count).sum();
        let oldest = members.iter().map(|hit| hit.oldest_activity).min();
        let newest = members.iter().map(|hit| hit.newest_activity).max();
        let keys = if target.discovery_group(plan) && members.len() >= 5 {
            top_keys(&members)
        } else {
            Vec::new()
        };
        groups.push(Group {
            group_id: target.group_id.clone(),
            rule_id: target.rule_id.clone(),
            level: target.level,
            category: target.category,
            item_count: members.len() as u64,
            eligible_count: eligible.len() as u64,
            total_bytes: format_bytes(total_bytes),
            eligible_bytes: format_bytes(eligible_bytes),
            unique_bytes: format_bytes(members.iter().map(|hit| hit.unique_bytes).sum()),
            eligible_unique_bytes: format_bytes(eligible.iter().map(|hit| hit.unique_bytes).sum()),
            file_count,
            oldest_activity: oldest.map(format_rfc3339),
            newest_activity: newest.map(format_rfc3339),
            in_use: target.in_use.clone(),
            top_keys: keys,
            disk_bytes: Some(format_bytes(members.iter().map(|hit| hit.disk_bytes).sum())),
            eligible_disk_bytes: Some(format_bytes(
                eligible.iter().map(|hit| hit.disk_bytes).sum(),
            )),
            cluster: clusters.get(&target.group_id).cloned(),
        });
    }
    engine
        .issues
        .sort_by(|a, b| a.group_id.cmp(&b.group_id).then(a.code.cmp(&b.code)));
    Ok((
        ScanResult {
            result_version: "water.scan-result.v1".to_string(),
            scan_id: scan_id.to_string(),
            started_at: format_rfc3339(now),
            ended_at: format_rfc3339(now + engine.started.elapsed()),
            status: engine.status.to_string(),
            visited_entries: engine.visited,
            groups,
            items: out_items.into_iter().map(|(_, item)| item).collect(),
            issues: engine.issues,
        },
        engine.extras,
    ))
}

fn build_clusters(
    engine: &mut Engine<'_>,
    hits: &mut Vec<Hit>,
    plan: &mut ScanPlan,
) -> BTreeMap<String, ClusterInfo> {
    let mut roots = engine.search_dirs.values().cloned().collect::<Vec<_>>();
    roots.sort_by(|a, b| {
        a.0.matches('\\')
            .count()
            .cmp(&b.0.matches('\\').count())
            .then(a.0.to_ascii_lowercase().cmp(&b.0.to_ascii_lowercase()))
    });
    let mut infos = BTreeMap::new();
    let mut member_paths: Vec<String> = Vec::new();
    let mut pending_groups = Vec::new();
    for (parent, category, discovery) in roots {
        if member_paths
            .iter()
            .any(|p| path_eq(p, &parent) || is_proper_ancestor(p, &parent))
        {
            continue;
        }
        let Some(list) = engine.cache.get(&parent.to_ascii_lowercase()).cloned() else {
            continue;
        };
        let is_candidate = |c: &Child| {
            if c.kind == Kind::Reparse || c.name.eq_ignore_ascii_case(".git") {
                return false;
            }
            if discovery {
                hits.iter().any(|h| {
                    path_eq(&h.path, &c.path)
                        && matches!(
                            plan.targets[h.target_index].match_spec,
                            Match::Discovery { .. }
                        )
                }) && !hits.iter().any(|h| {
                    !matches!(
                        plan.targets[h.target_index].match_spec,
                        Match::Discovery { .. }
                    ) && (path_eq(&h.path, &c.path) || is_proper_ancestor(&c.path, &h.path))
                })
            } else {
                !hits.iter().any(|h| path_eq(&h.path, &c.path))
            }
        };
        let mut templates: BTreeMap<(String, String), Vec<Child>> = BTreeMap::new();
        for child in &list.children {
            if !is_candidate(child) {
                continue;
            }
            if let Some(template) = name_template(&child.name, child.kind == Kind::File) {
                if child.kind == Kind::File && template.contains("<d>") {
                    continue;
                }
                templates
                    .entry((
                        if child.kind == Kind::File {
                            "file"
                        } else {
                            "dir"
                        }
                        .into(),
                        template,
                    ))
                    .or_default()
                    .push(child.clone());
            }
        }
        for ((kind, template), mut members) in templates {
            members.retain(|c| !member_paths.iter().any(|p| path_eq(p, &c.path)));
            if members.len() < 10 {
                continue;
            }
            let mut all_members = members.clone();
            if kind == "file" && template.contains("<h>") {
                let (prefix, extension) = template.split_once("<h>").expect("hash template");
                for child in &list.children {
                    if child.kind == Kind::File
                        && is_candidate(child)
                        && !member_paths.iter().any(|p| path_eq(p, &child.path))
                        && !members.iter().any(|m| path_eq(&m.path, &child.path))
                        && match_segment(&format!("{prefix}*{extension}"), &child.name)
                    {
                        all_members.push(child.clone());
                    }
                }
            }
            let id = cluster_id(&parent, &kind, &template);
            let mut measurements = Vec::new();
            for member in &all_members {
                if engine.stopped(&id) {
                    break;
                }
                measurements.push((
                    member.clone(),
                    measure(engine, &member.path, member.kind, &id),
                ));
            }
            if engine.stopped(&id) || measurements.len() != all_members.len() {
                continue;
            }
            let mut sizes = measurements[..members.len()]
                .iter()
                .map(|(_, m)| m.disk_bytes)
                .collect::<Vec<_>>();
            sizes.sort_unstable();
            let median = sizes[(sizes.len() - 1) / 2] as u128;
            let similar = sizes
                .iter()
                .filter(|size| {
                    if median == 0 {
                        **size == 0
                    } else {
                        (**size as u128) * 4 >= median && (**size as u128) <= median * 4
                    }
                })
                .count()
                * 5
                >= members.len() * 4;
            let mut days = BTreeMap::<String, usize>::new();
            for child in &members {
                *days
                    .entry(format_rfc3339(child.creation)[..10].to_string())
                    .or_default() += 1;
            }
            let burst_day = days.into_iter().find(|(_, n)| *n >= 5).map(|(d, _)| d);
            let info = ClusterInfo {
                parent: parent.clone(),
                kind: kind.clone(),
                template: template.clone(),
                member_count: members.len() as u64,
                similar_size: similar,
                burst: burst_day.is_some(),
                burst_day,
                keep_newest: if template.contains("<h>") { 1 } else { 0 },
            };
            let target = Target {
                target_id: id.clone(),
                group_id: id.clone(),
                rule_id: "leak-cluster".into(),
                level: if info.similar_size || info.burst {
                    Level::Confirm
                } else {
                    Level::Advisory
                },
                category,
                match_spec: Match::Discovery {
                    root: parent.clone(),
                },
                item_type: ItemType::Any,
                min_age_hours: 24,
                keep_newest: info.keep_newest,
                in_use: crate::model::InUse {
                    mode: "runs_from_or_references_item".into(),
                    processes: vec![],
                    root_levels: None,
                },
                git_is_copy: false,
                age_basis: AgeBasis::LatestCreatedOrModified,
                keep_by: KeepBy::Activity,
                unlocks_target_ids: vec![],
            };
            for child in &all_members {
                member_paths.push(child.path.clone());
            }
            pending_groups.push((id, target, info, measurements));
        }
    }
    hits.retain(|hit| {
        !member_paths
            .iter()
            .any(|p| path_eq(p, &hit.path) || is_proper_ancestor(p, &hit.path))
    });
    engine
        .extras
        .declarations
        .retain(|path| !member_paths.iter().any(|p| is_proper_ancestor(p, path)));
    pending_groups.sort_by(|a, b| a.0.cmp(&b.0));
    for (id, target, info, members) in pending_groups {
        let index = plan.targets.len();
        plan.targets.push(target.clone());
        for (member, measured) in members {
            let identity = member.identity.map(|(serial, id)| Identity {
                volume_serial: format!("{serial:08X}"),
                file_id: id.iter().map(|b| format!("{b:02x}")).collect(),
            });
            push_complete_hit(
                engine,
                hits,
                &target,
                index,
                &member.path,
                member.kind,
                measured,
                identity,
            );
        }
        engine.emit(&id, target.category, true);
        infos.insert(id, info);
    }
    infos
}

fn cluster_id(parent: &str, kind: &str, template: &str) -> String {
    let text = format!("{}\n{kind}\n{template}", parent.to_ascii_lowercase());
    let mut hash = 0xcbf29ce484222325u64;
    for byte in text.bytes() {
        hash = (hash ^ u64::from(byte)).wrapping_mul(0x100000001b3);
    }
    format!("leak:{hash:016x}")
}

pub fn space_overview(
    root: &str,
    scan_id: &str,
    options: &ScanOptions,
    cancel: &CancelToken,
    observer: &dyn ScanObserver,
    now: SystemTime,
) -> Result<SpaceOverview, ScanError> {
    crate::util::validate_abs_path(root)?;
    let Some(dir) = classify_path(root) else {
        return Err(ScanError::Native);
    };
    if dir.kind != Kind::Dir {
        return Err(ScanError::Native);
    }
    let extra_options = ExtrasOptions::default();
    let mut engine = Engine {
        options,
        cancel,
        observer,
        now,
        started: Instant::now(),
        visited: 0,
        found_bytes: 0,
        status: "completed",
        issues: vec![],
        last_progress: Instant::now() - Duration::from_millis(50),
        cache: HashMap::new(),
        extras_options: &extra_options,
        include_quarantine: true,
        extras: ScanExtras {
            extras_version: "water.scan-extras.v1".into(),
            scan_id: scan_id.into(),
            junctions: vec![],
            junctions_truncated: false,
            declarations: vec![],
        },
        search_dirs: BTreeMap::new(),
    };
    let mut entries = Vec::new();
    for child in engine.list(root, "space_overview") {
        if engine.stopped("space_overview") {
            break;
        }
        let measured = if child.kind == Kind::Reparse {
            Measured::empty(child.mtime, child.creation)
        } else {
            measure(&mut engine, &child.path, child.kind, "space_overview")
        };
        if engine.stopped("space_overview") {
            break;
        }
        engine.found_bytes = engine.found_bytes.saturating_add(measured.bytes);
        if engine.last_progress.elapsed() >= Duration::from_millis(50) {
            observer.on_progress(&ScanProgress {
                group_id: "space_overview".into(),
                category: "space".into(),
                visited_entries: engine.visited,
                found_bytes: engine.found_bytes,
            });
            engine.last_progress = Instant::now();
        }
        entries.push(SpaceEntry {
            name: child.name,
            path: child.path,
            kind: match child.kind {
                Kind::File => "file",
                Kind::Dir => "dir",
                Kind::Reparse => "reparse",
            }
            .into(),
            disk_bytes: format_bytes(measured.disk_bytes),
            total_bytes: format_bytes(measured.bytes),
            file_count: measured.files,
            newest_activity: Some(format_rfc3339(measured.newest)),
            has_git: measured.has_git,
        });
    }
    entries.sort_by(|a, b| {
        a.disk_bytes
            .parse::<u64>()
            .unwrap_or(0)
            .cmp(&b.disk_bytes.parse().unwrap_or(0))
            .reverse()
            .then(
                a.name
                    .to_ascii_lowercase()
                    .cmp(&b.name.to_ascii_lowercase()),
            )
    });
    let total = entries
        .iter()
        .map(|e| e.total_bytes.parse::<u64>().unwrap_or(0))
        .sum::<u64>();
    let disk = entries
        .iter()
        .map(|e| e.disk_bytes.parse::<u64>().unwrap_or(0))
        .sum::<u64>();
    let files = entries.iter().map(|e| e.file_count).sum();
    let others_count = entries.len().saturating_sub(10) as u64;
    let others_disk = entries
        .iter()
        .skip(10)
        .map(|e| e.disk_bytes.parse::<u64>().unwrap_or(0))
        .sum();
    entries.truncate(10);
    observer.on_progress(&ScanProgress {
        group_id: "space_overview".into(),
        category: "space".into(),
        visited_entries: engine.visited,
        found_bytes: engine.found_bytes,
    });
    Ok(SpaceOverview {
        overview_version: "water.space-overview.v1".into(),
        scan_id: scan_id.into(),
        root: root.into(),
        started_at: format_rfc3339(now),
        ended_at: format_rfc3339(now + engine.started.elapsed()),
        status: engine.status.into(),
        visited_entries: engine.visited,
        disk_bytes: format_bytes(disk),
        total_bytes: format_bytes(total),
        file_count: files,
        top: entries,
        others_count,
        others_disk_bytes: format_bytes(others_disk),
        issues: engine.issues,
    })
}

trait DiscoveryFlag {
    fn discovery_group(&self, plan: &ScanPlan) -> bool;
}

impl DiscoveryFlag for Target {
    fn discovery_group(&self, plan: &ScanPlan) -> bool {
        plan.targets
            .iter()
            .find(|t| t.group_id == self.group_id)
            .is_some_and(|t| matches!(t.match_spec, Match::Discovery { .. }))
    }
}

pub fn result_to_json(result: &ScanResult) -> String {
    crate::model::to_json(result).unwrap_or_else(|_| "{}".to_string())
}

pub fn result_from_json(text: &str) -> Result<ScanResult, ScanError> {
    let mut value: serde_json::Value = crate::model::from_json(text)?;
    for (array, fields) in [
        ("items", &[("unique_bytes", "total_bytes")][..]),
        (
            "groups",
            &[
                ("unique_bytes", "total_bytes"),
                ("eligible_unique_bytes", "eligible_bytes"),
            ][..],
        ),
    ] {
        if let Some(rows) = value
            .get_mut(array)
            .and_then(serde_json::Value::as_array_mut)
        {
            for row in rows {
                for (field, fallback) in fields {
                    // Only absent fields receive legacy defaults; explicit null
                    // or a wrong type still fails the strict typed parser.
                    if row.get(*field).is_none()
                        && let Some(old) = row.get(*fallback).cloned()
                    {
                        row[*field] = old;
                    }
                }
            }
        }
    }
    let result: ScanResult = serde_json::from_value(value).map_err(|_| ScanError::InvalidResult)?;
    if result.result_version != "water.scan-result.v1" {
        return Err(ScanError::InvalidResult);
    }
    Ok(result)
}
