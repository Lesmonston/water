function archive(source, destination) {
  const from = join(root, source), to = join(root, output, destination);
  assert(lstatSync(from).isFile() && !lstatSync(from).isSymbolicLink());
  assert(!relative(root, to).startsWith('..'));
  const bytes = readFileSync(from);
  if (from !== to) {
    mkdirSync(resolve(to,'..'),{recursive:true});
    copyFileSync(from,to,constants.COPYFILE_EXCL);
  }
  assert.equal(sha(readFileSync(to)), sha(bytes));
  entries.push({source, archive:output+'/'+destination, bytes:bytes.length, sha256:sha(bytes)});
}

  if (file.startsWith('2026-10-20T') && /\.(json|log|txt)$/.test(file)) {
    archive(path,'gates/'+file);
    if (file.endsWith('-result.json')) receipts.push({file,...json(path)});
  } else if (/^type-probes-\d+$/.test(file) && statSync(join(root,path)).mtimeMs >= start) {
    assert(lstatSync(join(root,path)).isDirectory() && !lstatSync(join(root,path)).isSymbolicLink());
    for (const sub of readdirSync(join(root,path)).sort()) {
      if (/\.(txt|log)$/.test(sub)) archive(path+'/'+sub,'type-probes/'+file+'/'+sub);
    }
  }
}
