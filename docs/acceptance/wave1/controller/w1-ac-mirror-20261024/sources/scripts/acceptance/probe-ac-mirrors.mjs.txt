const root=resolve('.'),out=join(root,'.local/checks/controller-ac-mirror-20261024');mkdirSync(out,{recursive:true});

const record={at:new Date().toISOString(),authorization:'User requested domestic mirror downloads on 2026-10-24',systemWindows:0,buildOrProductTestsRun:false,globalConfigurationChanged:false,mirrors:[]};

for(const mirror of[{id:'rsproxy',index:'https://rsproxy.cn/index/'},{id:'ustc',index:'https://mirrors.ustc.edu.cn/crates.io-index/'}]){

  const results=await Promise.allSettled([request(mirror.id+'-wry',mirror.index+'3/w/wry'),request(mirror.id+'-tauri',mirror.index+'ta/ur/tauri',8388608)]);for(const r of results){assert.equal(r.status,'fulfilled');entry.requests.push(r.value.record);assert.equal(r.value.record.exit.code,0);assert(r.value.record.stdout.startsWith('200 0 '));assert(r.value.bytes.toString('utf8').trim().split('\n').every(line=>JSON.parse(line).vers));if(r.value.record.url.endsWith('/ta/ur/tauri'))entry.pinnedTauri=JSON.parse(r.value.bytes.toString('utf8').trim().split('\n').find(line=>JSON.parse(line).vers==='2.12.5'));}
  assert(entry.pinnedTauri&&!entry.pinnedTauri.yanked);const dl=entry.config.dl;const crateUrl=dl.includes('{crate}')?dl.replaceAll('{crate}','tauri').replaceAll('{version}','2.12.5').replaceAll('{prefix}','ta/ur').replaceAll('{lowerprefix}','ta/ur'):dl.replace(/\/$/,'')+'/tauri/2.12.5/download';const crate=await request(mirror.id+'-tauri-crate',crateUrl,8388608);entry.requests.push(crate.record);assert.equal(crate.record.exit.code,0);assert(crate.record.stdout.startsWith('200 0 '));assert.equal(hash(crate.bytes),entry.pinnedTauri.cksum);entry.packageChecksumMatchesIndex=true;entry.status='passed';
 }catch(e){entry.status='failed';entry.error=e.stack;}
