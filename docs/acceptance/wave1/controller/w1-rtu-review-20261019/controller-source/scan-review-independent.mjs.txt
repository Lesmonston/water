assert(root.endsWith('controller-rtu-accept-20261019'));
const model=await import(pathToFileURL(join(root,'prototypes/scan-review/model.mjs')));

function caller(label){const b={bundle_version:source.bundle_version,snapshots:[0,1,9].map(i=>structuredClone(source.snapshots[i]))};b.snapshots.forEach((s,i)=>{s.source_kind='synthetic_memory';s.view.scan_id=label+'-'+i;const ids=new Map();s.view.scopes.forEach((x,j)=>{const next=label+'-scope-'+i+'-'+j;ids.set(x.scope_id,next);x.scope_id=next;});s.view.findings.forEach((x,j)=>{x.scope_id=ids.get(x.scope_id);x.id=label+'-finding-'+i+'-'+j;});s.view.issues.forEach(x=>{x.scope_id=ids.get(x.scope_id);});});return b;}
function raw(url,path,method='GET',host){const u=new URL(url);return new Promise((ok,bad)=>{const req=request({hostname:'127.0.0.1',port:u.port,path,method,headers:{Host:host??u.host}},res=>{const chunks=[];res.on('data',b=>chunks.push(b));res.on('end',()=>ok({status:res.statusCode,headers:res.headers,body:Buffer.concat(chunks)}));});req.on('error',bad);req.end();});}
test('controller_model_associations_and_raw_q',()=>{const b=caller('controller-model');assert.equal(model.assertReviewBundle(b),b);for(const s of b.snapshots)assert.equal(model.exportText(b,s.view.scan_id),s.redacted_json);assert.equal(model.selectSnapshot(b,'absent'),null);for(const edit of [x=>{const q=JSON.parse(x.snapshots[0].redacted_json);q.findings[0].scope_ref='scope-999';x.snapshots[0].redacted_json=JSON.stringify(q);},x=>{x.snapshots[1].summary.counts.findings++;},x=>{x.snapshots[2].view.totals.logical_bytes='0';},x=>{x.snapshots[0].redacted_json+=' ';},x=>{x.snapshots[1].redacted_json=x.snapshots[0].redacted_json;}]){const bad=structuredClone(b);edit(bad);assert.throws(()=>model.assertReviewBundle(bad),{message:'Unsupported scan review.'});}let read=0;const getter=structuredClone(b);Object.defineProperty(getter,'bundle_version',{enumerable:true,get(){read++;return source.bundle_version;}});assert.throws(()=>model.assertReviewBundle(getter),{message:'Unsupported scan review.'});assert.equal(read,0);});
test('controller_http_capture_and_precise_routes',async()=>{for(const label of ['controller-http-a','controller-http-b']){const input=caller(label),expected=structuredClone(input),opening=startServer({bundle:input});input.snapshots[0].redacted_json='mutated after synchronous capture';input.snapshots[0].view.scan_id='mutated';input.snapshots.pop();const host=await opening;try{assert.deepEqual(JSON.parse((await raw(host.url,'/reports.json')).body),expected);for(let n=1;n<=3;n++){const response=await raw(host.url,'/exports/'+n+'.json');assert.equal(response.status,200);assert.deepEqual(response.body,Buffer.from(expected.snapshots[n-1].redacted_json));assert.equal(response.headers['content-disposition'],'attachment; filename="water-report-'+n+'.json"');assert.equal(response.headers['content-type'],'application/json;charset=utf-8');assert.equal(response.headers['access-control-allow-origin'],undefined);const head=await raw(host.url,'/exports/'+n+'.json','HEAD');assert.equal(head.body.length,0);assert.equal(Number(head.headers['content-length']),response.body.length);}for(const p of ['/exports/0.json','/exports/01.json','/exports/4.json','/exports/1.json?raw=1','/exports/1.json#fragment','//exports/1.json','/exports/%31.json','/exports/../1.json','/reports.json?extra=1'])assert.equal((await raw(host.url,p)).status,404);assert.equal((await raw(host.url,'/exports/1.json','POST')).status,405);assert.equal((await raw(host.url,'/exports/1.json','GET','localhost')).status,403);}finally{await host.close();}}});
