// Evidence-only launcher. Never changes the frozen checker or its assertions.
const fs = require('node:fs');
const cp = require('node:child_process');
const path = require('node:path');
const assert = require('node:assert/strict');
const root = 'F:/new/smaltalk/water/.worktrees/ac';
assert.equal(process.cwd().replaceAll('\\', '/').toLowerCase(), root.toLowerCase());
const mode = process.argv[2], final = process.argv[3] === '--final';
assert(['red-behavior','test','red-nc1','red-nc2','check','native'].includes(mode));
assert(process.argv.length === (final ? 4 : 3));
const stamp = new Date().toISOString().replaceAll(':','-') + '-launch-' + process.pid;
const dir = path.join(root,'.local/checks/WATER-W1-AC-ADDENDUM-02');
const args = ['F:/new/smaltalk/water/scripts/ac-mirror-check.mjs','--task','WATER-W1-AC','--mode',mode,...(final ? ['--final'] : [])];
const launch = {startedAt:new Date().toISOString(),pid:process.pid,binary:process.execPath,args,mode,final};
fs.writeFileSync(path.join(dir,stamp+'.json'),JSON.stringify(launch,null,2)+'\n',{flag:'wx'});
const r = cp.spawnSync(process.execPath,args,{cwd:root,encoding:'utf8',windowsHide:true,maxBuffer:64*1024*1024});
fs.writeFileSync(path.join(dir,stamp+'-stdout.log'),r.stdout??'',{flag:'wx'});
fs.writeFileSync(path.join(dir,stamp+'-stderr.log'),r.stderr??'',{flag:'wx'});
fs.writeFileSync(path.join(dir,stamp+'-exit.json'),JSON.stringify({...launch,endedAt:new Date().toISOString(),exitCode:r.status,signal:r.signal,error:r.error?.message??null},null,2)+'\n',{flag:'wx'});
process.exitCode = r.status ?? 1;
