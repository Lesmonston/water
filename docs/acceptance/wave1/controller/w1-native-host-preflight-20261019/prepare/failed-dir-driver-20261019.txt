// Controller: library tests use markers; CLI uses one explicit bounded UI action.
import assert from 'node:assert/strict';
import {spawn} from 'node:child_process';
import {resolve} from 'node:path';
const argv=process.argv.slice(2),mode=argv.shift();assert(['tests','cli'].includes(mode));const action=mode==='cli'?argv.shift():null;if(action)assert(['cancel','select','select-dir'].includes(action));const binary=argv.shift();assert(binary&&/\.exe$/i.test(binary));const child=spawn(binary,argv,{stdio:['ignore','pipe','pipe'],windowsHide:false});let line='',fault,chain=Promise.resolve(),actions=[];
const timer=setTimeout(()=>{fault=new Error('Native session driver timeout');child.kill();},170000);
function enqueue(value){chain=chain.then(()=>new Promise((ok,bad)=>{const p=spawn('powershell.exe',['-NoProfile','-ExecutionPolicy','Bypass','-File',resolve('scripts/native-session-ui-action.ps1'),'-TestProcessId',String(child.pid),'-Action',value],{windowsHide:true,stdio:['ignore','pipe','pipe']});const t=setTimeout(()=>p.kill(),20000);let output='';p.stdout.on('data',b=>{output+=b;process.stderr.write(b);});p.stderr.on('data',b=>process.stderr.write(b));p.on('error',bad);p.on('close',code=>{clearTimeout(t);if(code!==0||!output.includes('WATER_UI_ACTION'))bad(new Error('Native session UI action failed'));else{actions.push(value);ok();}});})).catch(e=>{fault=e;child.kill();});}
child.stdout.on('data',b=>process.stdout.write(b));child.stderr.on('data',b=>{process.stderr.write(b);line+=b;let n;while((n=line.indexOf('\n'))>=0){const text=line.slice(0,n).trim();line=line.slice(n+1);const m=/^WATER_PICKER_CASE (cancel|select|select-dir)$/.exec(text);if(m){assert.equal(mode,'tests');enqueue(m[1]);}}});child.on('error',e=>{fault=e;});if(action)enqueue(action);const code=await new Promise(ok=>child.on('close',ok));await chain;clearTimeout(timer);assert(!fault,fault?.message);if(mode==='cli')assert.deepEqual(actions,[action]);else{assert(actions.includes('cancel')&&actions.includes('select')&&actions.includes('select-dir'),'Need actual cancellation/current/dir interactions');}process.exitCode=code??1;
