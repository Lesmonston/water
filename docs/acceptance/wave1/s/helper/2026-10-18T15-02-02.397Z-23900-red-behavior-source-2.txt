//! Native picker v1; no path-based constructor for a confirmed selection.
#![forbid(unsafe_code)]

use std::fmt;
use water_contracts::v1::OpaqueId;
use water_native_root::v1::{self as native, RootError};
use water_scope_registry::v1::{RegisteredScope, Registry, RegistryError};

pub struct PickerOptions {
    initial_directory: String,
}

impl PickerOptions {
    pub fn new(initial_directory: &str) -> Result<Self, PickerError> {
        Ok(Self {
            initial_directory: initial_directory.to_owned(),
        })
    }
}

pub struct SelectedScope {
    scope: RegisteredScope,
    root_path: String,
}

impl SelectedScope {
    pub fn scope(&self) -> &RegisteredScope {
        &self.scope
    }

    pub fn root_path(&self) -> &str {
        &self.root_path
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PickerError {
    Root(RootError),
    Source(RegistryError),
    UnsupportedPlatform,
    ApartmentUnavailable,
    DialogFailed,
    InvalidSelection,
}

impl fmt::Display for PickerError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Root(error) => error.fmt(f),
            Self::Source(error) => error.fmt(f),
            error => write!(f, "{error:?} at /picker"),
        }
    }
}

impl std::error::Error for PickerError {}

pub fn select_scope(
    registry: &mut Registry,
    scope_id: &OpaqueId,
    options: &PickerOptions,
) -> Result<Option<SelectedScope>, PickerError> {
    #[cfg(all(target_os = "windows", target_arch = "x86_64"))]
    {
        let _initial = native::open_directory(scope_id, &options.initial_directory)
            .map_err(PickerError::Root)?;
        let _chosen = crate::native_windows::choose(&options.initial_directory)?;
        let _ = registry;
        Err(PickerError::InvalidSelection)
    }
    #[cfg(not(all(target_os = "windows", target_arch = "x86_64")))]
    {
        let _ = (registry, scope_id, &options.initial_directory);
        Err(PickerError::UnsupportedPlatform)
    }
}
