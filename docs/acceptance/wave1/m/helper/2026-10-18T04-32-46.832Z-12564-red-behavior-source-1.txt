//! Logical lengths describe observations, never allocated or reclaimable space.

use serde::Serialize;
use serde_json::Value;
use std::fmt;
use water_contracts::v1::{OpaqueId, Platform, ScanStatus, ScanTotals};
use water_scan_report::v1::{AssemblyError, ReportContext, assemble_json};

const MAX_OUTPUT_BYTES: usize = 33_554_432;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum AnalysisError {
    Source(AssemblyError),
    OutputTooLarge,
    SerializationFailed,
}

impl fmt::Display for AnalysisError {
    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::Source(error) => error.fmt(formatter),
            Self::OutputTooLarge => formatter.write_str("OutputTooLarge at /analysis"),
            Self::SerializationFailed => formatter.write_str("SerializationFailed at /analysis"),
        }
    }
}

impl std::error::Error for AnalysisError {
    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
        match self {
            Self::Source(error) => Some(error),
            _ => None,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Serialize)]
struct Counts {
    scopes: u32,
    entries: u32,
    file_entries: u32,
    non_file_entries: u32,
    known_identity_file_entries: u32,
    unknown_identity_file_entries: u32,
    known_identity_groups: u32,
    conflicting_identity_groups: u32,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
struct EntryRef {
    finding_id: OpaqueId,
    scope_id: OpaqueId,
}

#[derive(Debug, Clone, PartialEq, Serialize)]
struct IdentityGroup {
    device: String,
    inode: String,
    state: &'static str,
    logical_bytes: Option<String>,
    logical_byte_values: Vec<String>,
    link_count_values: Vec<String>,
    entry_refs: Vec<EntryRef>,
}

/// An immutable sidecar. Serialize for inspection; full facts are not redacted.
#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct SizeLedger {
    ledger_version: &'static str,
    read_only: bool,
    scan_id: OpaqueId,
    source_run_id: String,
    platform: Platform,
    status: ScanStatus,
    source_facts: Value,
    source_totals: ScanTotals,
    counts: Counts,
    scope_coverage_complete: bool,
    observed_file_logical_bytes: String,
    known_identity_logical_bytes: Option<String>,
    unknown_identity_observed_logical_bytes: String,
    identity_deduplicated_logical_bytes: Option<String>,
    observed_set_identity_complete: bool,
    volume_unique_allocated_bytes: Option<String>,
    reclaimable_bytes: Option<String>,
    identity_groups: Vec<IdentityGroup>,
    unknown_identity_entries: Vec<EntryRef>,
}

pub fn analyze_json(input: &str, context: &ReportContext) -> Result<SizeLedger, AnalysisError> {
    let bundle = assemble_json(input, context).map_err(AnalysisError::Source)?;
    let report = bundle.report();
    let facts = bundle.source_facts();
    let ledger = SizeLedger {
        ledger_version: "water.size-ledger.v1",
        read_only: true,
        scan_id: report.scan_id.clone(),
        source_run_id: facts["run_id"]
            .as_str()
            .expect("G validates run ID")
            .to_owned(),
        platform: context.platform.clone(),
        status: report.status.clone(),
        source_facts: facts.clone(),
        source_totals: report.totals.clone(),
        counts: Counts {
            scopes: 0,
            entries: 0,
            file_entries: 0,
            non_file_entries: 0,
            known_identity_file_entries: 0,
            unknown_identity_file_entries: 0,
            known_identity_groups: 0,
            conflicting_identity_groups: 0,
        },
        scope_coverage_complete: report.status == ScanStatus::Completed,
        observed_file_logical_bytes: "0".to_owned(),
        known_identity_logical_bytes: Some("0".to_owned()),
        unknown_identity_observed_logical_bytes: "0".to_owned(),
        identity_deduplicated_logical_bytes: Some("0".to_owned()),
        observed_set_identity_complete: true,
        volume_unique_allocated_bytes: None,
        reclaimable_bytes: None,
        identity_groups: Vec::new(),
        unknown_identity_entries: Vec::new(),
    };
    to_json(&ledger)?;
    Ok(ledger)
}

pub fn to_json(output: &SizeLedger) -> Result<String, AnalysisError> {
    let json = serde_json::to_string(output).map_err(|_| AnalysisError::SerializationFailed)?;
    if json.len() > MAX_OUTPUT_BYTES {
        return Err(AnalysisError::OutputTooLarge);
    }
    Ok(json)
}
