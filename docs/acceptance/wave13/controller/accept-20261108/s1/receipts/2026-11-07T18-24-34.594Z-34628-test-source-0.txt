pub(crate) fn idle() -> Value {
    json!({"clean_id":null,"state":"idle","done_items":0,"total_items":0,"freed_bytes":"0","group_index":0,"group_count":0,"current_group_title_zh":null,"phase":"delete","quarantined_bytes":"0"})
}

pub(crate) fn prepare(
    core: &Arc<Core>,
    ids: &[Value],
    excluded: &[Value],
) -> Result<Value, IpcError> {
    prepare_mode(core, ids, excluded, None)
}

pub(crate) fn prepare_mode(
    core: &Arc<Core>,
    ids: &[Value],
    excluded: &[Value],
    mode: Option<(&str, &str)>,
) -> Result<Value, IpcError> {
    let mut s = lock(core)?;
    if s.busy != "idle" {
        return Err(IpcError::new("busy"));
    }
    let r = s.last_record().ok_or_else(|| IpcError::new("stale"))?;
    if let Some((_, version)) = mode {
        conclusion::ensure(core, &mut s)?;
        conclusion::current(&s, s.m35.conclusion.as_ref().unwrap(), version)?;
    }
    if s.consumed.contains(&r.result.scan_id) {
        return Err(IpcError::new("stale"));
    }
    // Trust boundary: only a scan this process completed can be a cleaning source.
    if !s.local_scans.contains(&r.result.scan_id) {
        return Err(IpcError::new("stale"));
    }
    // A partial scan is a source only when no budget ran out and no protected group is unclear.
    if r.result.status == "partial"
        && r.result.issues.iter().any(|i| {
            i.code == "budget_exhausted"
                || r.result
                    .groups
                    .iter()
                    .find(|g| g.group_id == i.group_id)
                    .is_none_or(|g| g.level == water_junk_scan::Level::Protected)
        })
    {
        return Err(IpcError {
            code: "stale".into(),
            message_zh: "这次体检没有完整完成，请重新体检后再清理。".into(),
        });
    }
    let mut items = Vec::new();
    let mut groups = Vec::new();
    let mut by_drive = BTreeMap::<String, u64>::new();
    let mut unavailable = BTreeMap::<String, usize>::new();
    let mut available = BTreeMap::<String, bool>::new();
    let mut skipped_session = false;
    // Disk settings can only narrow a locally authorised scan, never add paths.
    let disk_skips = core
        .store
        .read("", "settings.json")
        .and_then(|v| serde_json::from_value::<crate::service::Settings>(v).ok())
        .map(|v| v.skipped)
        .unwrap_or_default();
    let projection = r.projection();
    let excluded: HashSet<&str> = excluded.iter().filter_map(Value::as_str).collect();
    for eid in &excluded {
        let item = projection
            .item_index(eid)
            .map(|index| &r.result.items[index])
            .ok_or_else(|| IpcError::new("invalid"))?;
        if !ids
            .iter()
            .any(|id| id.as_str() == Some(item.group_id.as_str()))
            || (!item.eligible && item.ineligible_reason.as_deref() != Some("user_skipped"))
        {
            return Err(IpcError::new("invalid"));
        }
    }
    let mut excluded_count = 0;
    for id in ids {
        let id = id.as_str().expect("validated group ID");
        let group = r
            .result
            .groups
            .iter()
            .find(|g| g.group_id == id)
            .ok_or_else(|| IpcError::new("invalid"))?;
        if !group.level.cleanable()
            || group.eligible_count == 0
            || !views::enabled(&s.settings, &views::category(group))
        {
            return Err(IpcError::new("invalid"));
        }
        let before = items.len();
        for &index in projection.group_items(id) {
            let item = &r.result.items[index];
            if !item.eligible || item.identity.is_none() {
                continue;
            }
            if projection.is_skipped(index) {
                continue;
            }
            if excluded.contains(item.item_id.as_str())
                || s.settings
                    .skipped
                    .iter()
                    .chain(&disk_skips)
                    .any(|v| v.matches(item))
            {
                excluded_count += 1;
                continue;
            }
            let target = r
                .plan
                .targets
                .iter()
                .filter(|t| t.group_id == id)
                .filter_map(|t| {
                    let root = match &t.match_spec {
                        water_junk_rules::MatchSpec::Pattern { base, .. } => base,
                        water_junk_rules::MatchSpec::Marker { root, .. }
                        | water_junk_rules::MatchSpec::Discovery { root } => root,
                    };
                    (util::contains(root, &item.path) && !root.eq_ignore_ascii_case(&item.path))
                        .then_some((root, t))
                })
                .max_by_key(|(root, _)| root.len());
            let (root, age, unlocks) = if let Some(cluster) = &group.cluster {
                if water_junk_scan::name_template(m35::name(&item.path), item.kind == "file")
                    .as_deref()
                    != Some(cluster.template.as_str())
                    || std::fs::symlink_metadata(&item.path).is_err()
                {
                    continue;
                }
                (cluster.parent.clone(), 24, Vec::new())
            } else {
                let (root, t) = target.ok_or_else(|| IpcError::new("invalid"))?;
                let unlocks = if t.unlocks_target_ids.is_empty() {
                    Vec::new()
                } else {
                    r.result
                        .items
                        .iter()
                        .filter(|p| p.path != item.path && util::contains(&p.path, &item.path))
                        .filter(|p| {
                            r.plan.targets.iter().any(|target| {
                                target.group_id == p.group_id
                                    && t.unlocks_target_ids.contains(&target.target_id)
                            })
                        })
                        .map(|p| p.path.clone())
                        .collect::<Vec<_>>()
                };
                (
                    root.clone(),
                    if r.saved["unfinished"].get(&item.item_id).is_some() {
                        0
                    } else {
                        t.min_age_hours
                    },
                    unlocks,
                )
            };
            items.push(json!({"item_id":item.item_id,"group_id":id,"rule_id":group.rule_id,"level":group.level,"path":item.path,"root":root,"kind":item.kind,"identity":item.identity,"total_bytes":item.total_bytes,"min_age_hours":age,"unlocked_paths":unlocks}));
        }
        if items.len() > before {
            groups.push(json!({"group_id":id,"title_zh":projection.details[id]["group"]["title_zh"],"level":group.level,"item_count":0,"total_bytes":"0","in_use_note_zh":if projection.details[id]["group"]["in_use"].is_null(){None}else{Some("运行中，占用的项会跳过")}}));
        }
    }
    retain_outermost(&mut items);
    items.retain(|p| {
        let item = &r.result.items[projection
            .item_index(p["item_id"].as_str().unwrap())
            .unwrap()];
        if mode.is_some_and(|m| m.0 == "quarantine")
            && conclusion::is_chosen(core, &r, item)
            && !*available.entry(util::drive(&item.path)).or_insert_with(|| {
                quarantine::registered_base(core, &util::drive(&item.path)).is_some()
            })
        {
            *unavailable.entry(util::drive(&item.path)).or_default() += 1;
            skipped_session |= core
                .rules
                .rule(p["rule_id"].as_str().unwrap())
                .is_some_and(|rule| rule.session_tool.is_some());
            false
        } else {
            true
        }
    });
    let mut totals = BTreeMap::<String, (usize, u64)>::new();
    for p in &items {
        let item = &r.result.items[projection
            .item_index(p["item_id"].as_str().unwrap())
            .unwrap()];
        let n = util::number(&item.unique_bytes);
        let total = totals.entry(item.group_id.clone()).or_default();
        total.0 += 1;
        total.1 += n;
        *by_drive.entry(util::drive(&item.path)).or_default() += n;
    }
    groups.retain_mut(|g| {
        let Some((count, bytes)) = totals.get(g["group_id"].as_str().unwrap()) else {
            return false;
        };
        g["item_count"] = json!(count);
        g["total_bytes"] = json!(bytes.to_string());
        true
    });
    if items.is_empty() {
        return Err(IpcError::new("invalid"));
    }
    let protected_groups: HashSet<&str> = r
        .result
        .groups
        .iter()
        .filter(|g| g.level == water_junk_scan::Level::Protected)
        .map(|g| g.group_id.as_str())
        .collect();
    let protected_paths: Vec<_> = r
        .result
        .items
        .iter()
        .filter(|i| protected_groups.contains(i.group_id.as_str()))
        .map(|i| i.path.clone())
        .collect();
    let now = core.platform.now();
    let mut id = util::new_id("plan", now);
    while s.plans.contains_key(&id) {
        id = util::new_id("plan", now);
    }
    let plan = json!({"plan_version":"water.clean-plan.v1","plan_id":id,"created_at":util::timestamp(now),"items":items,"protected_paths":protected_paths,"hard_guards":r.plan.hard_guards});
    clean::CleanPlan::from_json(&plan.to_string()).map_err(|_| IpcError::new("invalid"))?;
    let token = util::random_hex();
    let chosen_ids = if mode.is_some() {
        r.result
            .items
            .iter()
            .filter(|i| {
                items.iter().any(|p| p["item_id"] == i.item_id)
                    && conclusion::is_chosen(core, &r, i)
            })
            .map(|i| i.item_id.clone())
            .collect::<Vec<_>>()
    } else {
        vec![]
    };
    let mut stored = json!({"plan":plan,"scan_id":r.result.scan_id,"token_digest":util::digest(&token),"used":false});
    if let Some((chosen_mode, _)) = mode {
        stored["chosen_ids"] = json!(chosen_ids);
        stored["chosen_mode"] = json!(chosen_mode);
    }
    core.store.write("plans", &format!("{id}.json"), &stored)?;
    s.plans.insert(id.clone(), stored);
    let total: u64 = by_drive.values().sum();
    drop(projection);
    let mut extra = preview_parts(
        core,
        &r,
        &items,
        &chosen_ids,
        mode.map(|x| x.0).unwrap_or("delete"),
    );
    if !unavailable.is_empty() {
        if extra["chosen"].is_null() {
            extra["chosen"] = json!({"bytes":"0","items":[],"mode":"quarantine","note_zh":null});
        }
        let mut notes = Vec::new();
        if let Some(note) = extra["chosen"]["note_zh"].as_str() {
            notes.push(note.to_owned());
        } else if skipped_session {
            notes.push("清掉后这些对话不能再续接或回看".into());
        }
        notes.extend(
            unavailable
                .into_iter()
                .map(|(drive, count)| quarantine::unavailable_note(&drive, count)),
        );
        extra["chosen"]["note_zh"] = json!(notes.join("；"));
    }
    Ok(
        json!({"plan_id":id,"confirm_token":token,"item_count":items.len(),"total_bytes":total.to_string(),"groups":groups,"by_drive":by_drive.into_iter().map(|(root,n)|json!({"root":root,"bytes":n.to_string()})).collect::<Vec<_>>(),"irreversible":chosen_ids.is_empty() || mode.is_some_and(|x|x.0=="delete"),"excluded_count":excluded_count,"delete":extra["delete"],"chosen":extra["chosen"],"expected_skip":extra["expected_skip"]}),
    )
}

pub(crate) fn retain_outermost(items: &mut Vec<Value>) {
    // Same component-prefix ordering and ASCII folding as the executor's
    // validate_disjoint_items. Keep input order (and the outer item's mode).
    let mut paths = items
        .iter()
        .enumerate()
        .map(|(index, item)| {
            (
                format!(
                    "{}\\",
                    item["path"]
                        .as_str()
                        .unwrap()
                        .trim_end_matches('\\')
                        .to_ascii_lowercase()
                ),
                index,
            )
        })
        .collect::<Vec<_>>();
    paths.sort_unstable();
    let mut outer: Option<&str> = None;
    let mut retained = HashSet::new();
    for (path, index) in &paths {
        if outer.is_none_or(|parent| !path.starts_with(parent)) {
            retained.insert(*index);
            outer = Some(path);
        }
    }
    let mut index = 0;
    items.retain(|_| {
        let keep = retained.contains(&index);
        index += 1;
        keep
    });
}

pub(crate) fn preview_parts(
    core: &Core,
    r: &Record,
    items: &[Value],
    chosen: &[String],
    mode: &str,
) -> Value {
    let mut cats = BTreeMap::<String, u64>::new();
    let mut titles = BTreeMap::<String, u64>::new();
    let mut skips = BTreeMap::<String, u64>::new();
    let mut chosen_bytes = 0u64;
    let projection = r.projection();
    core.platform.begin_in_use_batch();
    for p in items {
        let Some(i) = p["item_id"]
            .as_str()
            .and_then(|id| projection.item_index(id).map(|n| &r.result.items[n]))
        else {
            continue;
        };
        let Some(g) = r.result.groups.iter().find(|g| g.group_id == i.group_id) else {
            continue;
        };
        let n = util::number(&i.unique_bytes);
        if chosen.contains(&i.item_id) {
            chosen_bytes += n;
            let title = projection.details[&g.group_id]["group"]["title_zh"]
                .as_str()
                .unwrap_or("重复文件")
                .to_owned();
            *titles.entry(title).or_default() += n;
        } else {
            *cats.entry(views::category(g)).or_default() += n;
        }
        if let Some(name) = core.platform.in_use(
            &views::in_use_spec(g),
            &m35::in_use_path(&i.path, g.in_use.root_levels),
        ) {
            *skips.entry(views::product_name(&name)).or_default() += n;
        }
    }
    let skip_bytes = skips.values().sum::<u64>();
    let session = items.iter().any(|p| {
        r.result.groups.iter().any(|g| {
            p["group_id"] == g.group_id
                && core
                    .rules
                    .rule(&g.rule_id)
                    .is_some_and(|r| r.session_tool.is_some())
        })
    });
    json!({"delete":{"bytes":cats.values().sum::<u64>().to_string(),"categories":cats.into_iter().map(|(c,n)|json!({"title_zh":views::CATEGORIES.iter().find(|x|x.0==c).map(|x|x.1).unwrap_or("其它"),"bytes":n.to_string()})).collect::<Vec<_>>()},"chosen":if chosen.is_empty(){Value::Null}else{json!({"bytes":chosen_bytes.to_string(),"items":titles.into_iter().map(|(title,n)|json!({"title_zh":title,"bytes":n.to_string()})).collect::<Vec<_>>(),"mode":mode,"note_zh":if session{Some("清掉后这些对话不能再续接或回看")}else{None}})},"expected_skip":if skip_bytes==0{Value::Null}else{json!({"bytes":skip_bytes.to_string(),"notes_zh":skips.into_iter().map(|(name,n)|format!("{name} 开着，会跳过约 {}",m35::amount(n))).collect::<Vec<_>>()})}})
}

pub(crate) struct Execution<'a> {
    pub guard: &'a dyn clean::ItemGuard,
    pub journal: &'a mut dyn std::io::Write,
    pub stop: &'a clean::StopFlag,
    pub observer: &'a dyn clean::CleanObserver,
    pub now: std::time::SystemTime,
}

pub(crate) trait Executor: Send + Sync {
    fn execute(
        &self,
        plan: &clean::CleanPlan,
        execution: Execution<'_>,
    ) -> Result<clean::CleanReport, clean::CleanError>;
    fn quarantine(
        &self,
        plan: &clean::CleanPlan,
        id: &str,
        target: &clean::QuarantineTarget,
        execution: Execution<'_>,
    ) -> Result<clean::QuarantineReport, clean::CleanError>;
}

pub(crate) struct NativeExecutor;
impl Executor for NativeExecutor {
    fn execute(
        &self,
        plan: &clean::CleanPlan,
        e: Execution<'_>,
    ) -> Result<clean::CleanReport, clean::CleanError> {
        clean::execute(plan, e.guard, e.journal, e.stop, e.observer, e.now)
    }
    fn quarantine(
        &self,
        plan: &clean::CleanPlan,
        id: &str,
        target: &clean::QuarantineTarget,
        e: Execution<'_>,
    ) -> Result<clean::QuarantineReport, clean::CleanError> {
        clean::quarantine(
            plan, id, target, e.guard, e.journal, e.stop, e.observer, e.now,
        )
    }
}

pub(crate) fn start(core: &Arc<Core>, id: &str, token: &str) -> Result<Value, IpcError> {
    start_with_executor(core, id, token, Arc::new(NativeExecutor))
}

pub(crate) fn start_with_executor(
    core: &Arc<Core>,
    id: &str,
    token: &str,
    executor: Arc<dyn Executor>,
) -> Result<Value, IpcError> {
    let mut s = lock(core)?;
    let mut stored = s
        .plans
        .get(id)
        .cloned()
        .ok_or_else(|| IpcError::new("not_found"))?;
    if stored["used"] == true || stored["token_digest"] != util::digest(token) {
        return Err(IpcError::new("invalid"));
    }
    let dedupe = stored["dedupe_id"].as_str().map(str::to_owned);
    if s.stale_plans.contains(id)
        || (dedupe.is_none()
            && (s.last.as_deref() != stored["scan_id"].as_str()
                || stored["scan_id"]
                    .as_str()
                    .is_none_or(|id| s.consumed.contains(id))))
        || dedupe
            .as_ref()
            .is_some_and(|id| !s.m35.dedupes.get(id).is_some_and(|r| r.local))
    {
        return Err(IpcError::new("stale"));
    }
    if s.busy != "idle" {
        return Err(IpcError::new("busy"));
    }
    let r = if dedupe.is_some() {
        s.m35.plan_sources.get(id).cloned()
    } else {
        s.last_record()
    }
    .ok_or_else(|| IpcError::new("stale"))?;
    let plan = clean::CleanPlan::from_json(&stored["plan"].to_string())
        .map_err(|_| IpcError::new("invalid"))?;
    // The token is spent as soon as the checks pass. A later failure does not restore it.
    consume_confirmation(&mut stored, &mut s, &r.result.scan_id);
    s.plans.insert(id.into(), stored.clone());
    // Anything that can fail happens before busy=cleaning, so a panic cannot leave the service stuck.
    let prepared = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
        let now = core.platform.now();
        let mut clean_id = util::new_id("clean", now);
        while s.cleans.contains_key(&clean_id) {
            clean_id = util::new_id("clean", now);
        }
        let group_ids = ordered_groups(&plan);
        let title = group_ids.first().and_then(|gid| {
            r.projection()
                .details
                .get(gid)
                .and_then(|d| d.get("group"))
                .and_then(|g| g.get("title_zh"))
                .cloned()
        });
        core.store.write("plans", &format!("{id}.json"), &stored)?;
        let mut saved = r.saved.clone();
        saved["consumed"] = json!(true);
        Ok::<_, IpcError>((clean_id, group_ids, title, saved, now))
    }));
    let (clean_id, group_ids, title, saved, now) = match prepared {
        Ok(Ok(v)) => v,
        Ok(Err(e)) => return Err(e),
        Err(_) => return Err(IpcError::new("unavailable")),
    };
    let stop = clean::StopFlag::default();
    s.stop = Some(stop.clone());
    s.busy = "cleaning";
    s.clean_progress = json!({"clean_id":clean_id,"state":"running","done_items":0,"total_items":plan.items.len(),"freed_bytes":"0","group_index":usize::from(!group_ids.is_empty()),"group_count":group_ids.len(),"current_group_title_zh":title,"phase":"delete","quarantined_bytes":"0"});
    drop(s);
    let worker_core = core.clone();
    let worker_id = clean_id.clone();
    let spawned = std::thread::Builder::new()
        .name("water-clean".into())
        .spawn(move || {
            // The whole task is guarded; a panic leaves the service idle and keeps the journal.
            let guarded = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
                let run = std::panic::catch_unwind(std::panic::AssertUnwindSafe(
                    || -> Result<_, IpcError> {
                        if dedupe.is_none() {
                            worker_core.store.write(
                                "scans",
                                &format!("{}.json", r.result.scan_id),
                                &saved,
                            )?;
                        }
                        worker_core.platform.begin_in_use_batch();
                        let guard = Guard {
                            core: worker_core.clone(),
                            record: r.clone(),
                        };
                        let mut journal = worker_core.store.journal(&worker_id)?;
                        let selected = stored["chosen_ids"]
                            .as_array()
                            .into_iter()
                            .flatten()
                            .filter_map(Value::as_str)
                            .collect::<BTreeSet<_>>();
                        let quarantine_mode = stored["chosen_mode"] == "quarantine";
                        let mut deletion = plan.clone();
                        deletion
                            .items
                            .retain(|i| !quarantine_mode || !selected.contains(i.item_id.as_str()));
                        let mut report = if deletion.items.is_empty() {
                            clean::CleanReport {
                                report_version: "water.clean-report.v1".into(),
                                plan_id: plan.plan_id.clone(),
                                started_at: util::timestamp(now),
                                ended_at: util::timestamp(now),
                                status: clean::RunStatus::Completed,
                                freed_bytes: "0".into(),
                                estimate_bytes: plan
                                    .items
                                    .iter()
                                    .map(|i| util::number(&i.total_bytes))
                                    .sum::<u64>()
                                    .to_string(),
                                volumes: vec![],
                                items: vec![],
                                journal_error: None,
                            }
                        } else {
                            let observer = Observer {
                                core: worker_core.clone(),
                                record: r.clone(),
                                plan: &plan,
                                groups: &group_ids,
                                offset: 0,
                                freed_before: 0,
                                quarantined_before: 0,
                                quarantining: false,
                            };
                            executor
                                .execute(
                                    &deletion,
                                    Execution {
                                        guard: &guard,
                                        journal: &mut journal,
                                        stop: &stop,
                                        observer: &observer,
                                        now,
                                    },
                                )
                                .map_err(quarantine::error)?
                        };
                        let mut reports = Vec::<clean::QuarantineReport>::new();
                        let mut batch_errors = Vec::new();
                        let batches = if quarantine_mode {
                            quarantine_batches(&plan, &selected)
                        } else {
                            BTreeMap::new()
                        };
                        let mut done = deletion.items.len();
                        let mut moved = 0u64;
                        for (root, items) in batches {
                            let id = {
                                let mut s = lock(&worker_core)?;
                                let id = quarantine::unique_id(&worker_core, &s, &root);
                                s.m35.quarantine_ids.insert(id.clone());
                                id
                            };
                            let mut batch = plan.clone();
                            batch.items = items;
                            if report.journal_error.is_some()
                                || reports.iter().any(|r| r.journal_error.is_some())
                            {
                                let skipped = journal_failed_batch(&batch, &id, now);
                                report.status = clean::RunStatus::Stopped;
                                done += batch.items.len();
                                reports.push(skipped);
                                continue;
                            }
                            {
                                let mut s = lock(&worker_core)?;
                                s.clean_progress["phase"] = json!("quarantine");
                            }
                            let observer = Observer {
                                core: worker_core.clone(),
                                record: r.clone(),
                                plan: &plan,
                                groups: &group_ids,
                                offset: done,
                                freed_before: util::number(&report.freed_bytes),
                                quarantined_before: moved,
                                quarantining: true,
                            };
                            // Recheck each drive immediately before calling the executor.
                            // An unavailable batch produces failures without moving files.
                            let moved_report = if let Some((_, target)) =
                                quarantine::registered_target(&worker_core, &root)
                            {
                                let report = match executor.quarantine(
                                    &batch,
                                    &id,
                                    &target,
                                    Execution {
                                        guard: &guard,
                                        journal: &mut journal,
                                        stop: &stop,
                                        observer: &observer,
                                        now,
                                    },
                                ) {
                                    Ok(report) => report,
                                    Err(e) => {
                                        if e == clean::CleanError::Journal {
                                            let acted = report.items.iter().any(|i| {
                                                matches!(
                                                    i.status,
                                                    clean::ItemStatus::Deleted
                                                        | clean::ItemStatus::Partial
                                                )
                                            }) || reports
                                                .iter()
                                                .flat_map(|r| &r.items)
                                                .any(|i| i.status == clean::MoveStatus::Moved);
                                            if !acted {
                                                return Err(quarantine::error(e));
                                            }
                                            let mut skipped =
                                                journal_failed_batch(&batch, &id, now);
                                            skipped.journal_error = Some(e.to_string());
                                            skipped
                                        } else {
                                            for item in &batch.items {
                                                batch_errors.push((
                                                    item.group_id.clone(),
                                                    quarantine_failure_reason(e).to_owned(),
                                                ));
                                            }
                                            failed_batch(&batch, &id, now)
                                        }
                                    }
                                };
                                quarantine::save_moves(
                                    &worker_core,
                                    &r,
                                    &batch,
                                    &report,
                                    &worker_id,
                                )
                                .map_err(|_| IpcError {
                                    code: "unavailable".into(),
                                    message_zh: quarantine::JOURNAL_MESSAGE.into(),
                                })?;
                                report
                            } else {
                                if let Some(last) = batch.items.last() {
                                    clean::CleanObserver::on_item(
                                        &observer,
                                        batch.items.len(),
                                        batch.items.len(),
                                        0,
                                        &last.item_id,
                                    );
                                }
                                clean::QuarantineReport {
                                    report_version: "water.quarantine-report.v1".into(),
                                    quarantine_id: id,
                                    started_at: util::timestamp(now),
                                    ended_at: util::timestamp(worker_core.platform.now()),
                                    status: clean::RunStatus::Completed,
                                    moved_bytes: "0".into(),
                                    items: batch
                                        .items
                                        .iter()
                                        .map(|i| clean::MoveReport {
                                            item_id: i.item_id.clone(),
                                            status: clean::MoveStatus::Failed,
                                            skip_reason: None,
                                            move_error: Some(
                                                clean::MoveError::QuarantineUnavailable,
                                            ),
                                            error_code: None,
                                            destination: None,
                                            identity: None,
                                            bytes: "0".into(),
                                        })
                                        .collect(),
                                    journal_error: None,
                                }
                            };
                            if moved_report.status == clean::RunStatus::Stopped {
                                report.status = clean::RunStatus::Stopped;
                            }
                            moved += util::number(&moved_report.moved_bytes);
                            done += batch.items.len();
                            reports.push(moved_report);
                        }
                        report.ended_at = util::timestamp(worker_core.platform.now());
                        Ok((report, reports, batch_errors))
                    },
                ));
                let panicked = run.is_err();
                let finished = match run {
                    Ok(Ok((report, quarantines, batch_errors))) => save_result(
                        &worker_core,
                        &r,
                        &plan,
                        &report,
                        &quarantines,
                        &batch_errors,
                        &worker_id,
                    )
                    .map(|stored| (stored, report.status)),
                    Ok(Err(e)) => Err(e),
                    Err(_) => Err(IpcError::new("unavailable")),
                };
                let _ = quarantine::refresh(&worker_core);
                match finished {
                    Ok((stored, status)) => {
                        let ids = if let Ok(mut s) = lock(&worker_core) {
                            s.clean_progress["group_index"] = json!(group_ids.len());
                            s.clean_progress["current_group_title_zh"] = Value::Null;
                            s.clean_order.push(worker_id.clone());
                            s.cleans.insert(worker_id, stored);
                            Some(trim_state(&mut s))
                        } else {
                            None
                        };
                        if let Some((scans, cleans)) = ids {
                            trim_files(&worker_core.store, &scans, &cleans);
                        }
                        if let Ok(mut s) = lock(&worker_core) {
                            s.clean_progress["state"] =
                                json!(if status == clean::RunStatus::Stopped {
                                    "stopped"
                                } else {
                                    "completed"
                                });
                            s.busy = "idle";
                            s.stop = None;
                        }
                    }
                    Err(_) if panicked => {
                        if let Ok(mut s) = lock(&worker_core) {
                            s.clean_progress = idle();
                            s.busy = "idle";
                            s.stop = None;
                        }
                    }
                    Err(e) => {
                        if let Ok(mut s) = lock(&worker_core) {
                            s.clean_progress["state"] = json!("failed");
                            s.m35.failures.insert(worker_id.clone(), e);
                            s.busy = "idle";
                            s.stop = None;
                        }
                    }
                }
            }));
            if guarded.is_err() {
                let mut s = worker_core
                    .state
                    .lock()
                    .unwrap_or_else(std::sync::PoisonError::into_inner);
                s.clean_progress = idle();
                s.busy = "idle";
                s.stop = None;
            }
        });
    if spawned.is_err() {
        let mut s = lock(core)?;
        s.busy = "idle";
        s.stop = None;
        s.clean_progress = idle();
        return Err(IpcError::new("unavailable"));
    }
    Ok(json!({"clean_id":clean_id}))
}

fn consume_confirmation(stored: &mut Value, state: &mut State, scan_id: &str) {
    // Both authorities are consumed under the same task mutex before execution can start.
    stored["used"] = json!(true);
    state.consumed.insert(scan_id.into());
    if state.last.as_deref() == Some(scan_id) {
        state
            .m35
            .stale
            .insert(scan_id.into(), "这次体检已被清理用掉".into());
    }
}

pub(crate) fn quarantine_batches(
    plan: &clean::CleanPlan,
    selected: &BTreeSet<&str>,
) -> BTreeMap<String, Vec<clean::PlanItem>> {
    let mut batches = BTreeMap::<String, Vec<clean::PlanItem>>::new();
    for i in plan
        .items
        .iter()
        .filter(|i| selected.contains(i.item_id.as_str()))
    {
        batches
            .entry(util::drive(&i.path))
            .or_default()
            .push(i.clone());
    }
    batches
}

fn ordered_groups(plan: &clean::CleanPlan) -> Vec<String> {
    let mut groups = Vec::new();
    for item in &plan.items {
        if !groups.contains(&item.group_id) {
            groups.push(item.group_id.clone());
        }
    }
    groups
}

struct Guard {
    core: Arc<Core>,
    record: Arc<Record>,
}
impl clean::ItemGuard for Guard {
    fn in_use(&self, item: &clean::PlanItem) -> Option<String> {
        let Some(g) = self
            .record
            .result
            .groups
            .iter()
            .find(|g| g.group_id == item.group_id)
        else {
            return Some("无法确认规则".into());
        };
        self.core.platform.in_use(
            &views::in_use_spec(g),
            &m35::in_use_path(&item.path, g.in_use.root_levels),
        )
    }
}

struct Observer<'a> {
    core: Arc<Core>,
    record: Arc<Record>,
    plan: &'a clean::CleanPlan,
    groups: &'a [String],
    offset: usize,
    freed_before: u64,
    quarantined_before: u64,
    quarantining: bool,
}
impl clean::CleanObserver for Observer<'_> {
    fn on_item(&self, done: usize, total: usize, freed_bytes: u64, item_id: &str) {
        if let Ok(mut s) = lock(&self.core) {
            let group = self
                .plan
                .items
                .iter()
                .find(|i| i.item_id == item_id)
                .map(|i| &i.group_id);
            let index = group
                .and_then(|g| self.groups.iter().position(|v| v == g))
                .map(|index| index + 1)
                .unwrap_or(0);
            s.clean_progress["done_items"] = json!(self.offset + done);
            s.clean_progress["total_items"] = json!(self.plan.items.len());
            if self.quarantining {
                s.clean_progress["quarantined_bytes"] = json!(
                    self.quarantined_before
                        .saturating_add(freed_bytes)
                        .to_string()
                );
            } else {
                s.clean_progress["freed_bytes"] =
                    json!(self.freed_before.saturating_add(freed_bytes).to_string());
            }
            let _ = total;
            s.clean_progress["group_index"] = json!(index);
            s.clean_progress["current_group_title_zh"] = group
                .map(|g| self.record.projection().details[g]["group"]["title_zh"].clone())
                .unwrap_or(Value::Null);
        }
    }
}

fn result_view(
    core: &Core,
    r: &Record,
    plan: &clean::CleanPlan,
    report: &clean::CleanReport,
    id: &str,
) -> Value {
    let groups:Vec<_>=ordered_groups(plan).into_iter().map(|group|{
        let reports:Vec<_>=report.items.iter().filter(|item|plan.items.iter().any(|p|p.item_id==item.item_id && p.group_id==group)).collect();
        let deleted=reports.iter().filter(|r|r.status==clean::ItemStatus::Deleted).count();
        let any_deleted=reports.iter().any(|r|r.status==clean::ItemStatus::Deleted || r.status==clean::ItemStatus::Partial);
        let failed=reports.iter().any(|r|r.status==clean::ItemStatus::Failed || r.failed_entries>0);
        let status=if !reports.is_empty() && deleted==reports.len(){"done"}else if any_deleted{"partial"}else if failed{"failed"}else{"skipped"};
        let mut counts=[0u64;7];
        for item in &reports {
            use clean::SkipReason::*;
            match item.skip_reason {Some(InUse)=>counts[0]+=1,Some(Changed|Gone)=>counts[1]+=1,Some(TooRecent)=>counts[2]+=1,Some(Reparse|OutsideRoot|Guarded|ProtectedOverlap)=>counts[3]+=1,Some(Cancelled)=>counts[4]+=1,Some(JournalFailed)=>counts[6]+=1,None=>{}}
            counts[5]+=item.failed_entries;
        }
        let tails=[" 项正在使用，已跳过"," 项已经变化，已跳过"," 项刚有改动，已跳过"," 项未通过安全复核，已跳过"," 项因停止未处理"," 个文件无法删除"," 项因日志写不进去未处理"];
        let notes:Vec<_>=counts.into_iter().zip(tails).filter(|(n,_)|*n>0).map(|(n,t)|format!("{n}{t}")).collect();
        json!({"group_id":group,"title_zh":r.projection().details[&group]["group"]["title_zh"],"status":status,"freed_bytes":reports.iter().map(|i|util::number(&i.bytes_freed)).sum::<u64>().to_string(),"notes_zh":notes})
    }).collect();
    let estimate: u64 = plan
        .items
        .iter()
        .filter_map(|p| r.result.items.iter().find(|i| i.item_id == p.item_id))
        .map(|i| util::number(&i.unique_bytes))
        .sum();
    json!({"clean_id":id,"finished_at":report.ended_at,"estimate_bytes":estimate.to_string(),"freed_bytes":report.freed_bytes,"drives":report.volumes,"groups":groups,"journal_display_path":core.store.root.join("journal").join(format!("{id}.jsonl")).to_string_lossy(),"quarantined":null,"journal_note_zh":null})
}

pub(crate) fn quarantine_failure_reason(error: clean::CleanError) -> &'static str {
    match error {
        clean::CleanError::InvalidPlan => "清理计划无效",
        clean::CleanError::Native => "filesystem operation failed",
        clean::CleanError::Unsupported => "这台电脑不支持",
        clean::CleanError::Journal => "日志写不进去",
        clean::CleanError::Overflow => "数量超出范围",
    }
}

fn journal_failed_batch(
    batch: &clean::CleanPlan,
    id: &str,
    now: std::time::SystemTime,
) -> clean::QuarantineReport {
    let mut skipped = failed_batch(batch, id, now);
    skipped.status = clean::RunStatus::Stopped;
    for i in &mut skipped.items {
        i.status = clean::MoveStatus::Skipped;
        i.skip_reason = Some(clean::SkipReason::JournalFailed);
        i.move_error = None;
    }
    skipped
}

fn failed_batch(
    batch: &clean::CleanPlan,
    id: &str,
    now: std::time::SystemTime,
) -> clean::QuarantineReport {
    clean::QuarantineReport {
        report_version: "water.quarantine-report.v1".into(),
        quarantine_id: id.into(),
        started_at: util::timestamp(now),
        ended_at: util::timestamp(now),
        status: clean::RunStatus::Completed,
        moved_bytes: "0".into(),
        journal_error: None,
        items: batch
            .items
            .iter()
            .map(|i| clean::MoveReport {
                item_id: i.item_id.clone(),
                status: clean::MoveStatus::Failed,
                skip_reason: None,
                move_error: Some(clean::MoveError::Other),
                error_code: None,
                destination: None,
                identity: None,
                bytes: "0".into(),
            })
            .collect(),
    }
}

pub(crate) fn save_result(
    core: &Core,
    r: &Record,
    plan: &clean::CleanPlan,
    report: &clean::CleanReport,
    quarantines: &[clean::QuarantineReport],
    errors: &[(String, String)],
    id: &str,
) -> Result<Value, IpcError> {
    let mut view = result_view(core, r, plan, report, id);
    apply_quarantines(r, plan, &mut view, quarantines);
    let journal_error =
        report.journal_error.is_some() || quarantines.iter().any(|q| q.journal_error.is_some());
    let n = report
        .items
        .iter()
        .filter(|i| i.skip_reason == Some(clean::SkipReason::JournalFailed))
        .count()
        + quarantines
            .iter()
            .flat_map(|q| &q.items)
            .filter(|i| i.skip_reason == Some(clean::SkipReason::JournalFailed))
            .count();
    if journal_error {
        view["journal_note_zh"] = if n == 0 {
            json!("清理日志没写全，这次要清的都已处理")
        } else {
            json!(format!("清理日志没写全，后面的 {n} 项没有清"))
        };
    }
    for (id, reason) in errors {
        for g in view["groups"]
            .as_array_mut()
            .into_iter()
            .flatten()
            .filter(|g| g["group_id"] == *id)
        {
            g["status"] = json!("failed");
            let note = json!(format!("没能放进隔离区：{reason}"));
            let notes = g["notes_zh"].as_array_mut().unwrap();
            if !notes.contains(&note) {
                notes.push(note);
            }
        }
    }
    let stored = json!({"clean_id":id,"plan_id":plan.plan_id,"scan_id":r.result.scan_id,"report":report,"quarantine_reports":quarantines,"plan_items":plan.items,"view":view});
    core.store.write("cleans", &format!("{id}.json"), &stored)?;
    Ok(stored)
}

fn apply_quarantines(
    r: &Record,
    plan: &clean::CleanPlan,
    view: &mut Value,
    reports: &[clean::QuarantineReport],
) {
    let count = reports
        .iter()
        .flat_map(|r| &r.items)
        .filter(|i| i.status == clean::MoveStatus::Moved)
        .count();
    if reports.is_empty() {
        return;
    }
    let bytes = reports
        .iter()
        .map(|r| util::number(&r.moved_bytes))
        .sum::<u64>();
    let mut ids = reports
        .iter()
        .filter(|r| r.items.iter().any(|i| i.status == clean::MoveStatus::Moved))
        .map(|r| r.quarantine_id.clone())
        .collect::<Vec<_>>();
    ids.sort_by_key(|s| s.rsplit('-').next().unwrap_or("").to_owned());
    if count > 0 {
        view["quarantined"] = json!({"quarantine_ids":ids,"bytes":bytes.to_string(),"item_count":count,"notes_zh":["移入隔离区尚未腾出空间，确认没问题后再永久删除"]});
    }
    for group in view["groups"].as_array_mut().into_iter().flatten() {
        let moved = reports
            .iter()
            .flat_map(|r| &r.items)
            .filter(|i| {
                plan.items
                    .iter()
                    .any(|p| p.item_id == i.item_id && p.group_id == group["group_id"])
            })
            .collect::<Vec<_>>();
        if moved.is_empty() {
            continue;
        }
        let done = moved
            .iter()
            .filter(|i| i.status == clean::MoveStatus::Moved)
            .count();
        group["status"] = json!(if done == moved.len() {
            "quarantined"
        } else if done > 0 {
            "partial"
        } else if moved.iter().any(|i| i.status == clean::MoveStatus::Failed) {
            "failed"
        } else {
            "skipped"
        });
        let notes = group["notes_zh"].as_array_mut().unwrap();
        let journal_failed = moved
            .iter()
            .filter(|i| i.skip_reason == Some(clean::SkipReason::JournalFailed))
            .count();
        if journal_failed > 0 {
            notes.push(json!(format!("{journal_failed} 项因日志写不进去未处理")));
        }
        if done > 0 {
            notes.push(json!(format!("{done} 项进了隔离区")));
        }
        for i in moved.iter().filter(|i| {
            i.status != clean::MoveStatus::Moved
                && i.skip_reason != Some(clean::SkipReason::JournalFailed)
        }) {
            notes.push(json!(
                if i.move_error == Some(clean::MoveError::QuarantineUnavailable) {
                    let drive = plan
                        .items
                        .iter()
                        .find(|p| p.item_id == i.item_id)
                        .map(|p| util::drive(&p.path))
                        .unwrap_or_default();
                    quarantine::unavailable_note(&drive, 1)
                } else if i.skip_reason == Some(clean::SkipReason::InUse) {
                    "正在使用，已跳过".into()
                } else if i.skip_reason == Some(clean::SkipReason::Cancelled) {
                    "因停止未处理".into()
                } else {
                    "没有移入，未通过复核或移动失败".into()
                }
            ));
        }
    }
    let _ = r;
}
