    struct Handle(*mut c_void);
    impl Drop for Handle {
        fn drop(&mut self) {
            // SAFETY: this wrapper owns exactly one live handle.
            unsafe {
                CloseHandle(self.0);
            }
        }
    }

    // A traversal owns parents before children. Even an early return must close the
    // innermost handle first, just as unwinding the former recursive calls did.
    struct FrameStack<T>(Vec<T>);
    impl<T> FrameStack<T> {
        fn new() -> Self {
            Self(Vec::new())
        }
    }
    impl<T> std::ops::Deref for FrameStack<T> {
        type Target = Vec<T>;
        fn deref(&self) -> &Self::Target {
            &self.0
        }
    }
    impl<T> std::ops::DerefMut for FrameStack<T> {
        fn deref_mut(&mut self) -> &mut Self::Target {
            &mut self.0
        }
    }
    impl<T> Drop for FrameStack<T> {
        fn drop(&mut self) {
            while self.0.pop().is_some() {}
        }
    }

    fn wide(s: &str) -> Vec<u16> {
        s.encode_utf16().chain(Some(0)).collect()
    }

    // The drive root, identity_of and guard resolution still open by name; long names need
    // the \\?\ prefix so they are not capped at MAX_PATH.
    fn prefixed(s: &str) -> Vec<u16> {
        if s.len() >= 248 && !s.starts_with(r"\\") {
            wide(&format!(r"\\?\{s}"))
        } else {
            wide(s)
        }
    }

    // Convert a FILETIME tick count to SystemTime with checked arithmetic; any overflow yields
    // None so the caller treats the item as `changed` rather than wrapping to a bogus time.
    fn filetime(ticks: i64) -> Option<SystemTime> {
        let delta = ticks.checked_sub(FILETIME_UNIX)?;
        let nanos = delta.unsigned_abs().checked_mul(100)?;
        let dur = Duration::from_nanos(nanos);
        if delta >= 0 {
            SystemTime::UNIX_EPOCH.checked_add(dur)
        } else {
            SystemTime::UNIX_EPOCH.checked_sub(dur)
        }
    }

    fn open_root(drive: &str) -> Option<Handle> {
        let name = wide(drive);
        // SAFETY: name is NUL-terminated; only read/list access with backup semantics.
        let raw = unsafe {
            CreateFileW(
                name.as_ptr(),
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            None
        } else {
            Some(Handle(raw))
        }
    }

    // Open a single-segment child relative to the parent handle. Returns the Win32 error on
    // failure so callers can tell "missing" from "changed".
    fn nt_open(parent: &Handle, seg: &str, access: u32, share: u32) -> Result<Handle, u32> {
        nt_open_at(parent.0, seg, access, share)
    }

    fn nt_open_at(parent: *mut c_void, seg: &str, access: u32, share: u32) -> Result<Handle, u32> {
        let mut name: Vec<u16> = seg.encode_utf16().collect();
        // A real NTFS path component is 1..=255 UTF-16 units. Reject an empty name (length 0
        // would open the parent directory itself instead of a child) and anything longer than 255
        // (so the UNICODE_STRING byte length can never overflow u16 and wrap).
        let byte_len = match name.len().checked_mul(2) {
            Some(n) if (1..=255).contains(&name.len()) => n as u16,
            _ => return Err(ERR_INVALID_NAME),
        };
        let mut us = UnicodeString {
            length: byte_len,
            maximum_length: byte_len,
            buffer: name.as_mut_ptr(),
        };
        let oa = ObjectAttributes {
            length: size_of::<ObjectAttributes>() as u32,
            root_directory: parent,
            object_name: &mut us,
            attributes: OBJ_CASE_INSENSITIVE,
            security_descriptor: ptr::null_mut(),
            security_qos: ptr::null_mut(),
        };
        let mut io = IoStatusBlock {
            status_or_pointer: 0,
            information: 0,
        };
        let mut handle: *mut c_void = ptr::null_mut();
        // SAFETY: oa/io/handle are valid and the single-segment name lives across the call.
        let status = unsafe { NtOpenFile(&mut handle, access, &oa, &mut io, share, OPTS) };
        // Own any returned handle immediately so RAII closes it even on an unusual non-zero
        // status; only status 0 is a success.
        let owned = if handle.is_null() {
            None
        } else {
            Some(Handle(handle))
        };
        if status != 0 {
            // SAFETY: pure status-to-DOS-error translation, no pointers. `owned` drops here,
            // closing any handle returned alongside the failure.
            return Err(unsafe { RtlNtStatusToDosError(status) });
        }
        owned.ok_or(ERR_INVALID_NAME)
    }

    fn info<T>(handle: &Handle, class: i32, value: &mut T) -> bool {
        // SAFETY: value is an initialized, aligned buffer of exactly size_of::<T>() bytes.
        0 != unsafe {
            GetFileInformationByHandleEx(
                handle.0,
                class,
                (value as *mut T).cast(),
                size_of::<T>() as u32,
            )
        }
    }

    fn attr_tag(handle: &Handle) -> Option<(u32, u32)> {
        let mut info_buf = FileAttributeTagInfo {
            file_attributes: 0,
            reparse_tag: 0,
        };
        info(handle, 9, &mut info_buf).then_some((info_buf.file_attributes, info_buf.reparse_tag))
    }

    fn file_id(handle: &Handle) -> Option<[u8; 16]> {
        let mut id = FileIdInfo {
            volume_serial: 0,
            file_id: [0; 16],
        };
        info(handle, 18, &mut id).then_some(id.file_id)
    }

    fn write_time(handle: &Handle) -> Option<i64> {
        let mut basic = FileBasicInfo {
            creation: 0,
            last_access: 0,
            last_write: 0,
            change: 0,
            attributes: 0,
        };
        info(handle, 0, &mut basic).then_some(basic.last_write)
    }

    fn standard_info(handle: &Handle) -> Option<(u64, u32)> {
        let mut std_info = FileStandardInfo {
            allocation: 0,
            end_of_file: 0,
            links: 0,
            delete_pending: 0,
            directory: 0,
        };
        info(handle, 1, &mut std_info)
            .then_some((std_info.end_of_file.max(0) as u64, std_info.links))
    }

    fn end_of_file(handle: &Handle) -> Option<u64> {
        standard_info(handle).map(|(size, _)| size)
    }

    fn volume_serial(handle: &Handle) -> Option<u32> {
        let mut serial = 0u32;
        // SAFETY: handle is live; optional outputs are null with zero sizes.
        let ok = unsafe {
            GetVolumeInformationByHandleW(
                handle.0,
                ptr::null_mut(),
                0,
                &mut serial,
                ptr::null_mut(),
                ptr::null_mut(),
                ptr::null_mut(),
                0,
            )
        };
        (ok != 0).then_some(serial)
    }

    fn identity_from(handle: &Handle) -> Option<Identity> {
        let id = file_id(handle)?;
        let serial = volume_serial(handle)?;
        Some(Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: id.iter().map(|b| format!("{b:02x}")).collect(),
        })
    }

    #[derive(Clone)]
    struct Raw {
        name: String,
        name_ok: bool,
        attr: u32,
        tag: u32,
        file_id: [u8; 16],
        size: u64,
        write: i64,
    }

    // Enumerate one directory handle. Class 20 restarts, class 19 continues; the listing ends
    // only when GetFileInformationByHandleEx fails with ERROR_NO_MORE_FILES. Every other
    // failure is an error, never a silent "no entries".
    fn enumerate(handle: &Handle) -> Result<Vec<Raw>, ()> {
        let mut out = Vec::new();
        let mut buf = vec![0u64; 4096];
        let byte_len = buf.len() * 8;
        let mut class = 20i32;
        loop {
            // SAFETY: buf is 8-byte aligned (Vec<u64>) and its full byte length is passed.
            let ok = unsafe {
                GetFileInformationByHandleEx(
                    handle.0,
                    class,
                    buf.as_mut_ptr().cast(),
                    byte_len as u32,
                )
            };
            if ok == 0 {
                // last_os_error reads GetLastError immediately after the failing call.
                return if std::io::Error::last_os_error().raw_os_error() == Some(ERR_NO_MORE_FILES)
                {
                    Ok(out)
                } else {
                    Err(())
                };
            }
            class = 19;
            let base = buf.as_ptr() as *const u8;
            let mut off = 0usize;
            loop {
                // The fixed 88-byte header must lie fully inside the filled buffer.
                if off.checked_add(88).is_none_or(|end| end > byte_len) {
                    return Err(());
                }
                // SAFETY: [off, off+88) is in-bounds (checked just above); each fixed field is
                // read unaligned at its SPEC offset from the 8-byte-aligned buffer.
                let (next, write, size, attr, name_len, tag, file_id) = unsafe {
                    let p = base.add(off);
                    let next = ptr::read_unaligned(p as *const u32) as usize;
                    let write = ptr::read_unaligned(p.add(24) as *const i64);
                    let size = ptr::read_unaligned(p.add(40) as *const i64).max(0) as u64;
                    let attr = ptr::read_unaligned(p.add(56) as *const u32);
                    let name_len = ptr::read_unaligned(p.add(60) as *const u32) as usize;
                    let tag = ptr::read_unaligned(p.add(68) as *const u32);
                    let mut file_id = [0u8; 16];
                    ptr::copy_nonoverlapping(p.add(72), file_id.as_mut_ptr(), 16);
                    (next, write, size, attr, name_len, tag, file_id)
                };
                // The variable-length name must also lie fully inside the buffer.
                if off
                    .checked_add(88)
                    .and_then(|h| h.checked_add(name_len))
                    .is_none_or(|end| end > byte_len)
                {
                    return Err(());
                }
                // SAFETY: [off+88, off+88+name_len) is in-bounds (checked just above); the
                // pointer is u16-aligned because `off` is always a multiple of 8 (it starts at 0
                // and every NextEntryOffset advance is checked to be a multiple of 8) and 88 is
                // even, so base+off+88 is 2-byte aligned; NTFS names are UTF-16 so name_len bytes
                // decode as name_len/2 u16 units.
                let (name, name_ok) = unsafe {
                    let units =
                        std::slice::from_raw_parts(base.add(off + 88) as *const u16, name_len / 2);
                    match String::from_utf16(units) {
                        Ok(s) => (s, true),
                        // NUL cannot occur in an actual component. This private snapshot
                        // key is injective even for names with identical lossy spellings,
                        // and is never passed to an open call (name_ok is false).
                        Err(_) => (
                            format!(
                                "\0{}",
                                String::from_utf16_lossy(units)
                            ),
                            false,
                        ),
                    }
                };
                if name != "." && name != ".." {
                    out.push(Raw {
                        name,
                        name_ok,
                        attr,
                        tag,
                        file_id,
                        size,
                        write,
                    });
                }
                if next == 0 {
                    break;
                }
                // NextEntryOffset must stay 8-byte aligned and strictly advance forward.
                if next % 8 != 0 {
                    return Err(());
                }
                off = off.checked_add(next).ok_or(())?;
            }
        }
    }

    fn set_delete(handle: &Handle) -> bool {
        let ex = FileDispositionInfoEx {
            flags: 0x1 | 0x2 | 0x10,
        };
        // SAFETY: class 21 reads a 4-byte flags field.
        let ok = unsafe {
            SetFileInformationByHandle(
                handle.0,
                21,
                (&ex as *const FileDispositionInfoEx).cast(),
                size_of::<FileDispositionInfoEx>() as u32,
            )
        };
        if ok != 0 {
            return true;
        }
        let plain = FileDispositionInfo { delete_file: 1 };
        // SAFETY: class 4 reads a 1-byte flag.
        0 != unsafe {
            SetFileInformationByHandle(
                handle.0,
                4,
                (&plain as *const FileDispositionInfo).cast(),
                size_of::<FileDispositionInfo>() as u32,
            )
        }
    }

    pub(crate) fn identity(name: &str) -> Result<Identity, CleanError> {
        let wide_name = prefixed(name);
        // SAFETY: NUL-terminated name; attribute-only read of the reparse point itself.
        let raw = unsafe {
            CreateFileW(
                wide_name.as_ptr(),
                FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS | FLAG_OPEN_REPARSE_POINT,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            return Err(CleanError::Native);
        }
        let handle = Handle(raw);
        identity_from(&handle).ok_or(CleanError::Native)
    }

    // Follow links and read attributes only: used to resolve guard and protected identities.
    pub(crate) fn resolve_follow(name: &str) -> Option<Identity> {
        let wide_name = prefixed(name);
        // SAFETY: NUL-terminated name; attribute-only, following any link to its target.
        let raw = unsafe {
            CreateFileW(
                wide_name.as_ptr(),
                FILE_READ_ATTRIBUTES,
                SHARE_RWD,
                ptr::null(),
                3,
                FLAG_BACKUP_SEMANTICS,
                ptr::null_mut(),
            )
        };
        if raw as isize == -1 || raw.is_null() {
            return None;
        }
        identity_from(&Handle(raw))
    }

    pub(crate) fn volume_free(name: &str) -> Result<u64, CleanError> {
        let wide_name = wide(name);
        let mut available = 0u64;
        // SAFETY: NUL-terminated name; only the available-bytes output is written.
        let ok = unsafe {
            GetDiskFreeSpaceExW(
                wide_name.as_ptr(),
                &mut available,
                ptr::null_mut(),
                ptr::null_mut(),
            )
        };
        if ok == 0 {
            Err(CleanError::Native)
        } else {
            Ok(available)
        }
    }

    fn last_segment(name: &str) -> &str {
        name.rsplit('\\').next().unwrap_or(name)
    }

    struct Walked {
        handle: Handle,
        attr: u32,
        tag: u32,
        file_id: [u8; 16],
        serial: u32,
        write: i64,
        size: u64,
        is_dir: bool,
        ancestors: Vec<Identity>,
    }

    // Walk from the drive root to the item, opening each level relative to the previous one
    // and never following reparse points. The item's identity is verified from its own handle.
    fn walk_item(item: &PlanItem) -> Result<Walked, SkipReason> {
        if !path::strictly_contains(&item.root, &item.path) {
            return Err(SkipReason::OutsideRoot);
        }
        let root = open_root(&item.path[..3]).ok_or(SkipReason::Changed)?;
        let mut ancestors = Vec::new();
        // The drive root is an ancestor too, so a protected path aliased onto the volume root
        // is still caught by identity.
        ancestors.push(identity_from(&root).ok_or(SkipReason::Changed)?);
        let mut parent = root;
        let segs: Vec<&str> = item.path[3..].split('\\').collect();
        for seg in &segs[..segs.len() - 1] {
            let handle = nt_open(
                &parent,
                seg,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            )
            .map_err(gone_or_changed)?;
            let (attr, _) = attr_tag(&handle).ok_or(SkipReason::Changed)?;
            if attr & REPARSE != 0 {
                return Err(SkipReason::OutsideRoot);
            }
            if attr & DIRECTORY == 0 {
                return Err(SkipReason::Changed);
            }
            // An ancestor whose identity cannot be read is treated as `changed`, never silently
            // dropped, so a protected-path alias to it is never missed.
            ancestors.push(identity_from(&handle).ok_or(SkipReason::Changed)?);
            parent = handle;
        }
        let name = segs[segs.len() - 1];
        let probe = nt_open(&parent, name, FILE_READ_ATTRIBUTES | SYNCHRONIZE, SHARE_RWD)
            .map_err(gone_or_changed)?;
        let (attr, tag) = attr_tag(&probe).ok_or(SkipReason::Changed)?;
        if attr & REPARSE != 0 {
            return Err(SkipReason::Reparse);
        }
        let is_dir = attr & DIRECTORY != 0;
        match item.kind {
            Kind::Dir if !is_dir => return Err(SkipReason::Changed),
            Kind::File if is_dir => return Err(SkipReason::Changed),
            _ => {}
        }
        let item_fid = file_id(&probe).ok_or(SkipReason::Changed)?;
        let serial = volume_serial(&probe).ok_or(SkipReason::Changed)?;
        // Step 3: identity is read from this walked handle (not a by-name open) and compared to
        // the plan; a mismatch means the object was replaced.
        let actual = Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: item_fid.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if actual != item.identity {
            return Err(SkipReason::Changed);
        }
        let write = write_time(&probe).ok_or(SkipReason::Changed)?;
        let (handle, size) = if is_dir {
            drop(probe);
            let listing = nt_open(
                &parent,
                name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            )
            .map_err(gone_or_changed)?;
            match (attr_tag(&listing), file_id(&listing)) {
                (Some((a, _)), Some(id))
                    if a & REPARSE == 0 && a & DIRECTORY != 0 && id == item_fid => {}
                _ => return Err(SkipReason::Changed),
            }
            (listing, 0)
        } else {
            let len = end_of_file(&probe).ok_or(SkipReason::Changed)?;
            (probe, len)
        };
        Ok(Walked {
            handle,
            attr,
            tag,
            file_id: item_fid,
            serial,
            write,
            size,
            is_dir,
            ancestors,
        })
    }

    fn gone_or_changed(werr: u32) -> SkipReason {
        if werr == ERR_FILE_NOT_FOUND || werr == ERR_PATH_NOT_FOUND {
            SkipReason::Gone
        } else {
            SkipReason::Changed
        }
    }

    pub(crate) fn check_chain(item: &PlanItem) -> Result<(), SkipReason> {
        walk_item(item).map(|_| ())
    }

    pub(crate) fn snapshot(
        item: &PlanItem,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
    ) -> Result<Snapshot, SkipReason> {
        let walked = walk_item(item)?;
        // An opened ancestor (including the drive root) sharing a protected identity means the
        // item sits inside protected data.
        for ancestor in &walked.ancestors {
            if protected_ids.contains(ancestor) {
                return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
            }
        }
        let item_id = Identity {
            volume_serial: format!("{:08X}", walked.serial),
            file_id: walked.file_id.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if guard_ids.contains(&item_id) {
            return Err(SkipReason::Guarded); // CLEAN_GUARD_ID
        }
        if protected_ids.contains(&item_id) {
            return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
        }
        let item_time = filetime(walked.write).ok_or(SkipReason::Changed)?;
        let mut snap = Snapshot {
            bytes: 0,
            newest: item_time,
            entries: vec![Entry {
                rel: String::new(),
                file_id: walked.file_id,
                size: walked.size,
                mtime: item_time,
                tag: walked.tag,
                attr: walked.attr,
                name_ok: true,
            }],
        };
        if walked.is_dir {
            walk_subtree(
                &walked.handle,
                "",
                walked.serial,
                guard_ids,
                protected_ids,
                &mut snap,
            )?;
        } else {
            snap.bytes = walked.size;
        }
        snap.entries.sort_by(|a, b| a.rel.cmp(&b.rel));
        Ok(snap)
    }

    struct SnapFrame {
        handle: Handle,
        base: String,
        kids: Vec<Raw>,
        i: usize,
    }

    // Record one enumerated object on the snapshot. Returns true when the caller should open
    // that directory and walk its children. Invalid UTF-16 names are stored but never opened.
    fn consider_entry(
        raw: &Raw,
        rel: String,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<bool, SkipReason> {
        let id = Identity {
            volume_serial: format!("{serial:08X}"),
            file_id: raw.file_id.iter().map(|b| format!("{b:02x}")).collect(),
        };
        if guard_ids.contains(&id) {
            return Err(SkipReason::Guarded); // CLEAN_GUARD_ID
        }
        if protected_ids.contains(&id) {
            return Err(SkipReason::ProtectedOverlap); // CLEAN_GUARD_ID
        }
        let mtime = filetime(raw.write).ok_or(SkipReason::Changed)?;
        snap.newest = snap.newest.max(mtime);
        let reparse = raw.attr & REPARSE != 0;
        if reparse {
            // CLEAN_REPARSE_KIND: only name-surrogate links are treated as links; any other
            // reparse point in the subtree poisons the whole item.
            if raw.tag & NAME_SURROGATE == 0 {
                return Err(SkipReason::Reparse);
            }
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: 0,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(false)
        } else if raw.attr & DIRECTORY != 0 {
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: 0,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(raw.name_ok)
        } else {
            snap.bytes = snap
                .bytes
                .checked_add(raw.size)
                .ok_or(SkipReason::Changed)?;
            snap.entries.push(Entry {
                rel,
                file_id: raw.file_id,
                size: raw.size,
                mtime,
                tag: raw.tag,
                attr: raw.attr,
                name_ok: raw.name_ok,
            });
            Ok(false)
        }
    }

    fn push_snap_dir(
        parent: *mut c_void,
        name: &str,
        rel: String,
        stack: &mut FrameStack<SnapFrame>,
    ) -> Result<(), SkipReason> {
        let child = nt_open_at(
            parent,
            name,
            FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
            SHARE_RWD,
        )
        .map_err(|_| SkipReason::Changed)?;
        let kids = enumerate(&child).map_err(|()| SkipReason::Changed)?;
        stack.push(SnapFrame {
            handle: child,
            base: rel,
            kids,
            i: 0,
        });
        Ok(())
    }

    fn walk_subtree(
        dir: &Handle,
        base: &str,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<(), SkipReason> {
        let mut stack = FrameStack::new();
        for raw in enumerate(dir).map_err(|()| SkipReason::Changed)? {
            let rel = if base.is_empty() {
                raw.name.clone()
            } else {
                format!("{base}\\{}", raw.name)
            };
            if consider_entry(&raw, rel.clone(), serial, guard_ids, protected_ids, snap)? {
                push_snap_dir(dir.0, &raw.name, rel, &mut stack)?;
            }
            drain_snap(&mut stack, serial, guard_ids, protected_ids, snap)?;
        }
        Ok(())
    }

    fn drain_snap(
        stack: &mut FrameStack<SnapFrame>,
        serial: u32,
        guard_ids: &[Identity],
        protected_ids: &[Identity],
        snap: &mut Snapshot,
    ) -> Result<(), SkipReason> {
        while !stack.is_empty() {
            let top = stack.len() - 1;
            if stack[top].i >= stack[top].kids.len() {
                stack.pop();
                continue;
            }
            let raw = stack[top].kids[stack[top].i].clone();
            stack[top].i += 1;
            let rel = if stack[top].base.is_empty() {
                raw.name.clone()
            } else {
                format!("{}\\{}", stack[top].base, raw.name)
            };
            let parent = stack[top].handle.0;
            if consider_entry(&raw, rel.clone(), serial, guard_ids, protected_ids, snap)? {
                push_snap_dir(parent, &raw.name, rel, stack)?;
            }
        }
        Ok(())
    }

    enum ParentErr {
        Gone,
        Other,
    }

    // Walk only to the item's parent, rejecting any reparse ancestor. Error 2 or 3 at any
    // level means the item is no longer under that name (CLEAN-M2 section 3).
    fn walk_parent(item: &PlanItem) -> Result<Handle, ParentErr> {
        let mut parent = open_root(&item.path[..3]).ok_or(ParentErr::Other)?;
        let segs: Vec<&str> = item.path[3..].split('\\').collect();
        for seg in &segs[..segs.len() - 1] {
            let handle = match nt_open(
                &parent,
                seg,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(h) => h,
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    return Err(ParentErr::Gone);
                }
                Err(_) => return Err(ParentErr::Other),
            };
            match attr_tag(&handle) {
                Some((a, _)) if a & REPARSE == 0 && a & DIRECTORY != 0 => {}
                _ => return Err(ParentErr::Other),
            }
            parent = handle;
        }
        Ok(parent)
    }

    // Use one Unicode simple-uppercase character per input character, without full-uppercase
    // expansions such as sharp-s -> SS. Greek iota-subscript letters have a simple mapping even
    // though char::to_uppercase expands them (UnicodeData.txt, Unicode 17.0.0).
    fn uppercase_key(rel: &str) -> String {
        rel.chars()
            .map(|c| match c {
                '\u{1f80}'..='\u{1f87}' | '\u{1f90}'..='\u{1f97}' | '\u{1fa0}'..='\u{1fa7}' => {
                    char::from_u32(c as u32 + 8).unwrap_or(c)
                }
                '\u{1fb3}' => '\u{1fbc}',
                '\u{1fc3}' => '\u{1fcc}',
                '\u{1ff3}' => '\u{1ffc}',
                _ => {
                    let mut upper = c.to_uppercase();
                    let first = upper.next().unwrap_or(c);
                    if upper.next().is_none() { first } else { c }
                }
            })
            .collect()
    }

    struct DelFrame {
        handle: Handle,
        rel: String,
        index: usize,
        kids: Vec<Raw>,
        kid_i: usize,
    }

    struct DelCtx<'a> {
        before: &'a Snapshot,
        stop: &'a StopFlag,
        folded: Vec<(String, usize)>,
        accounted: Vec<bool>,
        deleted: Vec<bool>,
        removed: u64,
        failed: u64,
        bytes_freed: u64,
        changed_stop: bool,
        initial_skip: Option<SkipReason>,
        // Set once the item itself (rel == "") has had its delete flag set this run. The residual
        // pass needs it: if a different object later bears the item's name, the item may only be
        // reported deleted when this run actually deleted it (addendum 02 change 4).
        self_deleted: bool,
    }

    impl DelCtx<'_> {
        // Prefer the exact spelling; only a miss uses the simple-uppercase index. Both searches
        // are logarithmic (M2's cache directories hold on the order of 1e5 entries).
        fn expected_index(&self, rel: &str) -> Option<usize> {
            self.before
                .entries
                .binary_search_by(|e| e.rel.as_str().cmp(rel))
                .ok()
                .or_else(|| {
                    let key = uppercase_key(rel);
                    self.folded
                        .binary_search_by(|(k, _)| k.cmp(&key))
                        .ok()
                        .map(|i| self.folded[i].1)
                })
        }

        fn fail(&mut self, index: usize) {
            if !self.accounted[index] {
                self.failed += 1;
                self.accounted[index] = true;
            }
        }

        // After a directory's own delete fails, report whether the failure is expected rather
        // than counted. At least one entry must remain, and each must be new or an already
        // accounted-for snapshot entry. Empty directories and enumeration errors still count.
        fn only_post_snapshot(&self, handle: &Handle, rel: &str) -> bool {
            match enumerate(handle) {
                Ok(rest) => {
                    !rest.is_empty()
                        && rest.iter().all(|k| {
                            let crel = if rel.is_empty() {
                                k.name.clone()
                            } else {
                                format!("{rel}\\{}", k.name)
                            };
                            self.expected_index(&crel)
                                .is_none_or(|i| self.accounted[i] || self.deleted[i])
                        })
                }
                Err(()) => false,
            }
        }

        fn note_removed(&mut self, index: usize, rel: &str) {
            self.removed += 1;
            self.deleted[index] = true;
            if rel.is_empty() {
                self.self_deleted = true;
            }
        }

        fn finish_dir(&mut self, handle: &Handle, rel: &str, index: usize) {
            if set_delete(handle) {
                self.note_removed(index, rel);
            } else if self.only_post_snapshot(handle, rel) {
                self.accounted[index] = true;
            } else {
                self.fail(index);
            }
        }

        // Open one snapshot node relative to its parent. Directories return a frame so the
        // caller can walk children on an explicit stack; files and links are finished here.
        fn enter(&mut self, parent: *mut c_void, name: &str, rel: &str) -> Option<DelFrame> {
            if self.changed_stop {
                return None;
            }
            let Some(index) = self.expected_index(rel) else {
                return None; // appeared after the snapshot: never touched, never counted
            };
            let exp = &self.before.entries[index];
            if !exp.name_ok {
                self.fail(index);
                return None;
            }
            let (exp_attr, exp_tag, exp_id) = (exp.attr, exp.tag, exp.file_id);
            if exp_attr & REPARSE != 0 {
                if exp_tag & NAME_SURROGATE != 0 {
                    // CLEAN_NO_FOLLOW / H1: re-verify from the link's own handle that it is still
                    // a name-surrogate reparse point with the snapshot FileId before deleting, so
                    // a concurrently swapped-in non-link reparse point (e.g. a cloud placeholder)
                    // is never removed.
                    match nt_open_at(
                        parent,
                        name,
                        DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                        SHARE_RWD,
                    ) {
                        Ok(handle) => match (attr_tag(&handle), file_id(&handle)) {
                            (Some((a, t)), Some(id))
                                if a & REPARSE != 0 && t & NAME_SURROGATE != 0 && id == exp_id =>
                            {
                                if set_delete(&handle) {
                                    self.note_removed(index, rel);
                                } else {
                                    self.fail(index);
                                }
                            }
                            _ => self.fail(index),
                        },
                        Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {}
                        Err(_) => self.fail(index),
                    }
                }
                return None;
            }
            if exp_attr & DIRECTORY != 0 {
                let handle = match nt_open_at(
                    parent,
                    name,
                    FILE_LIST_DIRECTORY | DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                    SHARE_RW,
                ) {
                    Ok(h) => h,
                    Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                        if rel.is_empty() {
                            self.initial_skip = Some(SkipReason::Gone);
                        }
                        return None;
                    }
                    Err(_) => {
                        self.fail(index);
                        return None;
                    }
                };
                let id = file_id(&handle);
                if rel.is_empty() && id.is_some_and(|id| id != exp_id) {
                    self.initial_skip = Some(SkipReason::Changed);
                    return None;
                }
                match (attr_tag(&handle), id) {
                    (Some((a, _)), Some(id))
                        if a & REPARSE == 0 && a & DIRECTORY != 0 && id == exp_id => {}
                    _ => {
                        // The directory was replaced under us: stop deleting this item.
                        self.fail(index);
                        self.changed_stop = true;
                        return None;
                    }
                }
                let kids = match enumerate(&handle) {
                    Ok(k) => k,
                    Err(()) => {
                        // Enumeration failed (not "no more files"): stop this item, count one.
                        self.fail(index);
                        self.changed_stop = true;
                        return None;
                    }
                };
                return Some(DelFrame {
                    handle,
                    rel: rel.to_owned(),
                    index,
                    kids,
                    kid_i: 0,
                });
            }
            match nt_open_at(
                parent,
                name,
                DELETE | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(handle) => {
                    let id = file_id(&handle);
                    if rel.is_empty() && id.is_some_and(|id| id != exp_id) {
                        self.initial_skip = Some(SkipReason::Changed);
                        return None;
                    }
                    match (attr_tag(&handle), id) {
                        (Some((a, _)), Some(id)) if a & REPARSE == 0 && id == exp_id => {
                            let Some((size, links)) = standard_info(&handle) else {
                                self.fail(index);
                                return None;
                            };
                            if set_delete(&handle) {
                                self.note_removed(index, rel);
                                if links <= 1 {
                                    self.bytes_freed = self.bytes_freed.saturating_add(size);
                                }
                            } else {
                                self.fail(index);
                            }
                        }
                        // became a link, FileId changed, or attributes unreadable: skip + count
                        _ => self.fail(index),
                    }
                }
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    if rel.is_empty() {
                        self.initial_skip = Some(SkipReason::Gone);
                    }
                }
                Err(_) => self.fail(index),
            }
            None
        }

        // Delete one node by handle, relative to its parent. Returns true if a stop cut it short.
        fn delete_from(&mut self, parent: &Handle, name: &str, rel: &str) -> bool {
            let mut stack = FrameStack::new();
            if let Some(frame) = self.enter(parent.0, name, rel) {
                stack.push(frame);
            }
            while !stack.is_empty() {
                if self.changed_stop {
                    return false;
                }
                let top = stack.len() - 1;
                if stack[top].kid_i < stack[top].kids.len() {
                    if self.stop.is_stopped() {
                        return true;
                    }
                    let kid = stack[top].kids[stack[top].kid_i].clone();
                    stack[top].kid_i += 1;
                    let child_rel = if stack[top].rel.is_empty() {
                        kid.name.clone()
                    } else {
                        format!("{}\\{}", stack[top].rel, kid.name)
                    };
                    let parent_ptr = stack[top].handle.0;
                    // Copy the snapshot FileId and kind out before any &mut self call so the
                    // immutable borrow of `before.entries` does not span enter().
                    let expected = self.expected_index(&child_rel).map(|i| {
                        let e = &self.before.entries[i];
                        (i, e.file_id, e.attr)
                    });
                    match expected {
                        None => {} // appeared after the snapshot: ignore, not a failure
                        Some((_, fid, _)) if fid == kid.file_id => {
                            if let Some(frame) = self.enter(parent_ptr, &kid.name, &child_rel) {
                                stack.push(frame);
                                // The recursive call used to finish this subtree before
                                // the parent's post-child stop checkpoint.
                                continue;
                            }
                            if self.changed_stop {
                                return false;
                            }
                        }
                        Some((i, _, snap_attr)) => {
                            // Name is in the snapshot but the object was replaced (FileId
                            // differs). Decide by the SNAPSHOT entry's kind (SPEC section 4 step
                            // 7), never by the live object that replaced it.
                            self.fail(i);
                            if snap_attr & DIRECTORY != 0 && snap_attr & REPARSE == 0 {
                                // The snapshot entry was a non-reparse directory: stop this item
                                // whatever replaced it (dir, file, link, or other reparse point).
                                self.changed_stop = true;
                                return false;
                            }
                            // The snapshot entry was a file or link: do not delete it, count one
                            // failure, and continue with the next sibling.
                        }
                    }
                    if self.stop.is_stopped() {
                        return true;
                    }
                } else {
                    let frame = stack.pop().unwrap();
                    if self.changed_stop {
                        return false;
                    }
                    self.finish_dir(&frame.handle, &frame.rel, frame.index);
                    drop(frame);
                    // After a child directory finishes (including its own delete), the parent
                    // checks the stop flag before the next sibling or its own delete — same
                    // checkpoints as the previous recursive node().
                    if !stack.is_empty() && self.stop.is_stopped() {
                        return true;
                    }
                }
            }
            false
        }
    }

    struct MeasFrame {
        handle: Handle,
        kids: Vec<Raw>,
        i: usize,
    }

    fn measure_kid(
        parent: *mut c_void,
        kid: &Raw,
        total: &mut u64,
        stack: &mut FrameStack<MeasFrame>,
    ) -> Option<()> {
        if kid.attr & REPARSE != 0 {
            return Some(());
        }
        if kid.attr & DIRECTORY != 0 {
            if !kid.name_ok {
                // Its subtree cannot be measured without opening the raw name.
                return None;
            }
            match nt_open_at(
                parent,
                &kid.name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                Ok(child) => {
                    let kids = enumerate(&child).ok()?;
                    stack.push(MeasFrame {
                        handle: child,
                        kids,
                        i: 0,
                    });
                    Some(())
                }
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => Some(()),
                Err(_) => None,
            }
        } else {
            *total = total.saturating_add(kid.size);
            Some(())
        }
    }

    // Residual bytes still under the item, measured by handle. None means the item is gone; the
    // second field is 1 when the residual could not be determined (never report deleted then).
    fn measure_tree(dir: &Handle) -> Option<u64> {
        let mut total = 0u64;
        let mut stack = FrameStack::new();
        for kid in enumerate(dir).ok()? {
            measure_kid(dir.0, &kid, &mut total, &mut stack)?;
            drain_measure(&mut total, &mut stack)?;
        }
        Some(total)
    }

    fn drain_measure(total: &mut u64, stack: &mut FrameStack<MeasFrame>) -> Option<()> {
        while !stack.is_empty() {
            let top = stack.len() - 1;
            if stack[top].i >= stack[top].kids.len() {
                stack.pop();
                continue;
            }
            let kid = stack[top].kids[stack[top].i].clone();
            stack[top].i += 1;
            let parent = stack[top].handle.0;
            measure_kid(parent, &kid, total, stack)?;
        }
        Some(())
    }

    fn measure_residual(
        item: &PlanItem,
        before: &Snapshot,
        self_deleted: bool,
    ) -> (Option<u64>, u64) {
        let before_bytes = before.bytes;
        // The original name may be missing or belong to a different FileId. Only this run's
        // successful delete flag on the item proves it was deleted; otherwise its residual is
        // unknown because someone else moved/replaced it. Keep bytes_before without an extra
        // failure (CLEAN-REPORT section 2, extending addendum 02 change 4).
        let replaced = |sd: bool| -> (Option<u64>, u64) {
            if sd {
                (None, 0)
            } else {
                (Some(before_bytes), 0)
            }
        };
        let Some(item_entry) = before.entries.iter().find(|e| e.rel.is_empty()) else {
            return (Some(before_bytes), 1);
        };
        let parent = match walk_parent(item) {
            Ok(p) => p,
            Err(ParentErr::Gone) => return replaced(self_deleted),
            Err(ParentErr::Other) => return (Some(before_bytes), 1),
        };
        let name = last_segment(&item.path);
        // Open the item itself relative to the parent handle (never following a link) and
        // identify it by FileId, instead of listing the parent and matching names.
        let handle = match nt_open(&parent, name, FILE_READ_ATTRIBUTES | SYNCHRONIZE, SHARE_RWD) {
            Ok(h) => h,
            Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                return replaced(self_deleted);
            }
            Err(_) => return (Some(before_bytes), 1),
        };
        let Some((attr, _)) = attr_tag(&handle) else {
            return (Some(before_bytes), 1);
        };
        let Some(id) = file_id(&handle) else {
            return (Some(before_bytes), 1);
        };
        if id != item_entry.file_id {
            return replaced(self_deleted);
        }
        if attr & DIRECTORY != 0 {
            match nt_open(
                &parent,
                name,
                FILE_LIST_DIRECTORY | FILE_READ_ATTRIBUTES | SYNCHRONIZE,
                SHARE_RWD,
            ) {
                // addendum 02 change 2: the handle used to sum the residual must itself carry the
                // snapshot FileId, so a swap between the two opens cannot make a different
                // object's tree read as the item's residual.
                Ok(dir) => match file_id(&dir) {
                    Some(did) if did == item_entry.file_id => match measure_tree(&dir) {
                        Some(bytes) => (Some(bytes), 0),
                        None => (Some(before_bytes), 1),
                    },
                    Some(_) => replaced(self_deleted),
                    None => (Some(before_bytes), 1),
                },
                Err(w) if w == ERR_FILE_NOT_FOUND || w == ERR_PATH_NOT_FOUND => {
                    replaced(self_deleted)
                }
                Err(_) => (Some(before_bytes), 1),
            }
        } else {
            match end_of_file(&handle) {
                Some(len) => (Some(len), 0),
                None => (Some(before_bytes), 1),
            }
        }
    }

    pub(crate) fn delete(
        item: &PlanItem,
        before: &Snapshot,
        stop: &StopFlag,
    ) -> (ItemReport, bool) {
        let mut folded: Vec<(String, usize)> = before
            .entries
            .iter()
            .enumerate()
            .map(|(i, e)| (uppercase_key(&e.rel), i))
            .collect();
        folded.sort_unstable();
        let mut ctx = DelCtx {
            before,
            stop,
            folded,
            accounted: vec![false; before.entries.len()],
            deleted: vec![false; before.entries.len()],
            removed: 0,
            failed: 0,
            bytes_freed: 0,
            changed_stop: false,
            initial_skip: None,
            self_deleted: false,
        };
        let mut interrupted = false;
        match walk_parent(item) {
            Ok(parent) => {
                interrupted = ctx.delete_from(&parent, last_segment(&item.path), "");
            }
            Err(ParentErr::Gone) => {
                let mut report = ItemReport::skipped(item, SkipReason::Gone);
                report.bytes_before = before.bytes.to_string();
                return (report, false);
            }
            Err(ParentErr::Other) => {}
        }
        if let Some(reason) = ctx.initial_skip {
            let mut report = ItemReport::skipped(item, reason);
            report.bytes_before = before.bytes.to_string();
            return (report, false);
        }
        let (residual, extra_failed) = measure_residual(item, before, ctx.self_deleted);
        ctx.failed += extra_failed;
        let status = if residual.is_none() {
            ItemStatus::Deleted
        } else if ctx.removed > 0 {
            ItemStatus::Partial
        } else if interrupted {
            ItemStatus::Skipped
        } else {
            ItemStatus::Failed
        };
        let cancelled = status == ItemStatus::Skipped;
        let report = ItemReport {
            item_id: item.item_id.clone(),
            status,
            skip_reason: cancelled.then_some(SkipReason::Cancelled),
            bytes_before: before.bytes.to_string(),
            bytes_freed: ctx.bytes_freed.to_string(),
            failed_entries: ctx.failed,
        };
        (report, interrupted)
    }
}

#[cfg(windows)]
pub(crate) use windows::{check_chain, delete, identity, resolve_follow, snapshot, volume_free};

#[cfg(not(windows))]
pub(crate) fn identity(_: &str) -> Result<Identity, CleanError> {
    Err(CleanError::Unsupported)
}
#[cfg(not(windows))]
pub(crate) fn resolve_follow(_: &str) -> Option<Identity> {
    None
}
#[cfg(not(windows))]
pub(crate) fn volume_free(_: &str) -> Result<u64, CleanError> {
    Err(CleanError::Unsupported)
}
#[cfg(not(windows))]
pub(crate) fn check_chain(_: &PlanItem) -> Result<(), SkipReason> {
    Err(SkipReason::Changed)
}
#[cfg(not(windows))]
pub(crate) fn snapshot(
    _: &PlanItem,
    _: &[Identity],
    _: &[Identity],
) -> Result<Snapshot, SkipReason> {
    Err(SkipReason::Changed)
}
#[cfg(not(windows))]
pub(crate) fn delete(item: &PlanItem, _: &Snapshot, _: &StopFlag) -> (ItemReport, bool) {
    (ItemReport::skipped(item, SkipReason::Changed), false)
}
