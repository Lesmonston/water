use serde_json::{Value, json};

const CANARY: &str = "CANARY_ECHO_VALUE_9f3c2a";
const CREATED_AT: &str = "2026-10-26T04:00:00Z";

fn m35_reject(rule: Value, label: &str) {
    let err =
        RuleSet::from_json(&wrap(vec![rule])).expect_err(&format!("RULES_M35_FIELDS: {label}"));
    assert!(
        !err.to_string().contains(CANARY),
        "RULES_M35_FIELDS: input not echoed"
    );
}

fn m35_marker(kind: &str) -> Value {
    let mut rule = sample_rule();
    rule["category"] = json!("custom");
    rule["item_type"] = json!("any");
    rule["locate"] =
        json!([{"kind":"marker","marker":{"kind":kind,"patterns":["*.cache"]},"max_depth":4}]);
    rule
}

            .iter()

        let error =
            RuleSet::from_json(&data.to_string()).expect_err("RULES_M35_SESSIONS: session rules");
        assert!(
            !error.to_string().contains(CANARY),
            "RULES_M35_SESSIONS: no echo"
        );
    }
}

fn m35_resolve() {
    let folders = KnownFolders::from_json(&folders_json()).unwrap();
    let rules = RuleSet::embedded();
    let base = resolve_plan(&rules, &folders, &[], CREATED_AT)
        .expect("RULES_M35_SESSIONS: default resolve");
    assert!(
        base.targets
            .iter()
            .all(|t| rules.rule(&t.rule_id).unwrap().session_tool.is_none()),
        "RULES_M35_SESSIONS: default off"
    );
    let default = resolve_plan_with(&rules, &folders, &[], CREATED_AT, &PlanOptions::default())
        .expect("RULES_M35_SESSIONS: default options");
    assert_eq!(
        plan_to_json(&base),
        plan_to_json(&default),
        "RULES_M35_SESSIONS: default bytes"
    );
    let options = PlanOptions {
        sessions: vec![SessionSwitch {
            tool: "claude-code".into(),
            min_age_days: 30,
        }],
    };
    let enabled = resolve_plan_with(&rules, &folders, &[], CREATED_AT, &options)
        .expect("RULES_M35_SESSIONS: enabled");
    assert_eq!(
        &enabled.targets[..base.targets.len()],
        &base.targets,
        "RULES_M35_SESSIONS: normal unchanged"
    );
    let sessions = &enabled.targets[base.targets.len()..];
    assert_eq!(sessions.len(), 2, "RULES_M35_SESSIONS: two Claude targets");
    let protected: Vec<_> = base
        .targets
        .iter()
        .filter(|t| t.rule_id == "claude-code-sessions")
        .map(|t| t.target_id.clone())
        .collect();
    for target in sessions {
        assert_eq!(
            target.rule_id, "claude-code-old-sessions",
            "RULES_M35_SESSIONS: only selected tool"
        );
        assert_eq!(target.min_age_hours, 720, "RULES_M35_SESSIONS: hours");
        assert_eq!(
            target.unlocks_target_ids, protected,
            "RULES_M35_SESSIONS: all protected IDs"
        );
    }
    for (tools, index) in [
        (vec![(CANARY, 30)], 0),
        (vec![("claude-code", 30), ("claude-code", 31)], 1),
        (vec![("codex", 0)], 0),
        (vec![("grok", 366)], 0),
    ] {
        let options = PlanOptions {
            sessions: tools
                .iter()
                .map(|(tool, days)| SessionSwitch {
                    tool: tool.to_string(),
                    min_age_days: *days,
                })
                .collect(),
        };
        let error = resolve_plan_with(&rules, &folders, &[], CREATED_AT, &options)
            .expect_err("RULES_M35_SESSIONS: invalid switch");
        assert_eq!(
            error.code, "RULES_INVALID_SESSIONS",
            "RULES_M35_SESSIONS: code"
        );
        assert_eq!(
            error.path,
            format!("/sessions/{index}"),
            "RULES_M35_SESSIONS: index"
        );
        assert!(
            !error.to_string().contains(CANARY),
            "RULES_M35_SESSIONS: no echo"
        );
    }
    let reverse = PlanOptions {
        sessions: [
            ("kimi-code", 365),
            ("grok", 1),
            ("cursor", 30),
            ("codex", 30),
            ("claude-code", 30),
        ]
        .into_iter()
        .map(|(tool, min_age_days)| SessionSwitch {
            tool: tool.into(),
            min_age_days,
        })
        .collect(),
    };
    let all = resolve_plan_with(&rules, &folders, &[], CREATED_AT, &reverse)
        .expect("RULES_M35_SESSIONS: all tools");
    assert_eq!(
        all.targets[base.targets.len()..]
            .iter()
            .map(|t| t.rule_id.as_str())
            .collect::<Vec<_>>(),
        [
            "claude-code-old-sessions",
            "claude-code-old-sessions",
            "codex-old-sessions",
            "cursor-old-chats",
            "grok-old-sessions",
            "kimi-code-old-sessions"
        ],
        "RULES_M35_SESSIONS: rule order not switch order"
    );
    assert_eq!(
        all.targets.last().unwrap().min_age_hours,
        8760,
        "RULES_M35_SESSIONS: 365 days"
    );
    let roots = [
        CustomRoot {
            root_id: "tmp-root".into(),
            path: "F:\\tmp".into(),
        },
        CustomRoot {
            root_id: "temp-root".into(),
            path: "G:\\Temp".into(),
        },
        CustomRoot {
            root_id: "other-root".into(),
            path: "F:\\Templates".into(),
        },
    ];
    let plan =
        resolve_plan(&rules, &folders, &roots, CREATED_AT).expect("RULES_M35_FIELDS: custom roots");
    let discoveries: Vec<_> = plan
        .targets
        .iter()
        .filter(|t| t.rule_id == "custom-temp-old")
        .collect();
    assert_eq!(
        discoveries.len(),
        2,
        "RULES_M35_FIELDS: discovery name filter"
    );
    for (target, root) in discoveries.iter().zip(&roots[..2]) {
        assert_eq!(
            target.target_id,
            format!("custom:{}:custom-temp-old#1", root.root_id),
            "RULES_M35_FIELDS: discovery ID"
        );
        assert_eq!(
            target.match_spec,
            MatchSpec::Discovery {
                root: root.path.clone()
            },
            "RULES_M35_FIELDS: discovery shape"
        );
    }
    assert_eq!(
        folders.get(water_junk_rules::KnownFolder::SystemDrive),
        "C:\\",
        "RULES_M35_FIELDS: derived drive"
    );
    let page = plan
        .targets
        .iter()
        .find(|t| t.rule_id == "pagefile")
        .unwrap();
    assert!(
        matches!(&page.match_spec,MatchSpec::Pattern{base,..} if base=="C:\\"),
        "RULES_M35_FIELDS: drive pattern"
    );
    let encoded = plan_to_json(&plan);
    for key in [
        "advice_zh",
        "in_use_zh",
        "retention_zh",
        "installed_names",
        "session_tool",
    ] {
        assert!(
            !encoded.contains(key),
            "RULES_M35_FIELDS: text/options stay out of plan"
        );
    }
    assert_eq!(
        ScanPlan::from_json(&encoded).expect("RULES_M35_FIELDS: read all new keys"),
        plan,
        "RULES_M35_FIELDS: full roundtrip"
    );
    assert_eq!(
        plan_to_json(&ScanPlan::from_json(&encoded).unwrap()),
        encoded,
        "RULES_M35_FIELDS: ordered bytes"
    );
    for (id, kind) in [
        ("audition-peak-files", "file_name"),
        ("app-old-versions", "dir_name_pattern"),
    ] {
        let wire: Value = serde_json::from_str(&encoded).unwrap();
        let target = wire["targets"]
            .as_array()
            .unwrap()
            .iter()
            .find(|t| t["rule_id"] == id)
            .unwrap();
        assert_eq!(
            target["match"]["marker"]["kind"], kind,
            "RULES_M35_FIELDS: new marker plan"
        );
    }

fn m3d_pair() -> Vec<Value> {
    let mut main = sample_rule();
    main["rule_id"] = json!("node-modules");
    main["category"] = json!("custom");
    main["locate"] = json!([{"kind":"marker","marker":{"kind":"dir_name","names":["node_modules"]},"max_depth":6,"conditions":m3d_conditions_json("node-modules")}]);

fn m3d_reject(rules: Vec<Value>, label: &str, marker: &str) {
    let err = RuleSet::from_json(&wrap(rules)).expect_err(&format!("{marker}: {label}"));
    assert!(
        !err.to_string().contains(CANARY),
        "{marker}: no input echo for {label}"
    );
}

fn m3d_plan_targets(plan: &ScanPlan, roots: &[CustomRoot]) {
    for root in roots {
        for (main_id, fallback_id) in [
            ("node-modules", "bundled-node-modules"),
            ("test-browser-profiles", "browser-data-dirs"),
        ] {
            let main_target = format!("custom:{}:{main_id}#1", root.root_id);
            let fallback_target = format!("custom:{}:{fallback_id}#1", root.root_id);
            let main = plan
                .targets
                .iter()
                .find(|t| t.target_id == main_target)
                .expect("RULES_M3D_CONDITIONS: main target");
            let fallback = plan
                .targets
                .iter()
                .find(|t| t.target_id == fallback_target)
                .expect("RULES_M3D_FALLBACK: fallback target");
            let MatchSpec::Marker {
                root: main_root,
                marker,
                max_depth,
                conditions,
                fallback_for,
            } = &main.match_spec
            else {
                panic!("RULES_M3D_CONDITIONS: main is marker");
            };
            assert_eq!(main_root, &root.path, "RULES_M3D_CONDITIONS: root");
            assert_eq!(
                conditions,
                &m3d_conditions(main_id),
                "RULES_M3D_CONDITIONS: target conditions"
            );
            assert!(
                fallback_for.is_none(),
                "RULES_M3D_FALLBACK: main is not fallback"
            );
            assert_eq!(
                fallback.match_spec,
                MatchSpec::Marker {
                    root: root.path.clone(),
                    marker: marker.clone(),
                    max_depth: *max_depth,
                    conditions: vec![],
                    fallback_for: Some(main_target),
                },
                "RULES_M3D_FALLBACK: same root marker depth and main target ID"
            );
            assert_eq!(
                fallback.level,
                Level::Advisory,
                "RULES_M3D_FALLBACK: advisory"
            );
        }
    }
}

            }

                "fallback has no conditions"

            5 => {
                rules[0]["category"] = json!("dev_cache");

                "main single locator"
            }
            7 => {

                rules[0]["locate"] = sample_rule()["locate"].clone();

            }
            8 => {
                rules[0]["level"] = json!("advisory");
                rules[0]["locate"][0]
                    .as_object_mut()
                    .unwrap()
                    .remove("conditions");
                rules[0]["locate"][0]["fallback_for"] = json!("bundled-node-modules");
                "main not a fallback"
            }
            9 => {
                rules[1]["locate"][0]["marker"]["names"] = json!(["NODE_MODULES"]);
                "identical marker"
            }
            10 => {
                rules[1]["locate"][0]["max_depth"] = json!(7);
                "identical depth"

            }
            12 => {

                "cannot point to self"
            }
            13 => {
                rules[1]["locate"][0]["fallback_for"] = json!(42);
                "fallback string"
            }
            _ => {
                rules[1]["locate"][0]["fallback_for"] = json!("bad id");
                "fallback rule ID syntax"
            }
        };
        m3d_reject(rules, label, "RULES_M3D_FALLBACK");
    }
    for case in 0..4 {
        let mut rule = pair[0].clone();
        match case {
            0 => rule["category"] = json!("dev_cache"),
            1 => {
                rule["category"] = json!("dev_cache");
                rule["locate"][0] = json!({"kind":"pattern","base":"local_app_data","segments":["SampleCache"],"conditions":m3d_conditions_json("node-modules")});

fn precedence_rule(id: &str, level: &str, base: &str, segments: &[&str]) -> Value {
    let mut rule = sample_rule();
    rule["rule_id"] = json!(id);
    rule["title_zh"] = json!("优先判定样例");
    rule["category"] = json!("browser");
    rule["level"] = json!(level);
    rule["locate"] = json!([{"kind":"pattern","base":base,"segments":segments}]);
    if level == "protected" {
        rule["what_zh"] = json!("这是登录相关的网站数据。");
        rule["why_zh"] = json!("清掉之后需要重新登录许多网站。");
        rule["after_zh"] = json!("这里只统计占用，不会被清理。");
    }
    rule
}

fn precedence_pair(
    level: &str,
    base: &str,
    cleanable: &[&str],
    protected_base: &str,
    protected: &[&str],
) -> String {
    wrap(vec![
        precedence_rule("cleanable-sample", level, base, cleanable),
        precedence_rule("protected-sample", "protected", protected_base, protected),
    ])
}

fn expect_precedence(input: &str, case: &str) {
    let err = match RuleSet::from_json(input) {
        Ok(set) => panic!("{case}: expected RULES_PRECEDENCE, got {set:?}"),
        Err(err) => err,
    };
    assert_eq!(err.code, "RULES_PRECEDENCE", "{case}: {err}");
    assert!(
        err.to_string().contains("RULES_PRECEDENCE"),
        "{case}: {err}"
    );
}

fn expect_no_precedence(input: &str, case: &str) {
    if let Err(err) = RuleSet::from_json(input) {
        panic!("{case} must pass, got {err}");
    }
}

#[test]
fn rules_protection_precedence() {
    let la = "local_app_data";
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "*"],
            la,
            &["User Data", "*", "Cookies"],
        ),
        "wildcard profile over wildcard cookies",
    );
    expect_no_precedence(
        &precedence_pair("recommended", la, &["*", "Cache"], la, &["*", "Network"]),
        "sibling cache and network",
    );
    let profiles = [
        ["User Data", "Default", "Cookies"],
        ["User Data", "Profile *", "Cookies"],
        ["User Data", "Guest Profile", "Cookies"],
        ["User Data", "System Profile", "Cookies"],
    ];
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default"],
            la,
            &profiles[0],
        ),
        "literal Default over Default cookies",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default", "Network"],
            la,
            &["User Data", "Default", "Network"],
        ),
        "literal Default Network equals protected",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 3"],
            la,
            &profiles[1],
        ),
        "literal Profile 3 matched by Profile *",
    );
    expect_precedence(
        &precedence_pair("confirm", la, &["user data", "profile 3"], la, &profiles[1]),
        "case-insensitive literal against pattern",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "ShaderCache"],
            la,
            &profiles[1],
        ),
        "root shader cache against Profile *",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default"],
            "user_profile",
            &profiles[0],
        ),
        "different base",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            "user_profile",
            &[".codex", "sessions"],
            "user_profile",
            &[".codex*", "sessions"],
        ),
        "literal .codex sessions matched by .codex*",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 12"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash matches digits",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile x"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash needs digits",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile "],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash needs at least one digit",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 7"],
            la,
            &["User Data", "Profile ?", "Cookies"],
        ),
        "question mark matches one character",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 12"],
            la,
            &["User Data", "Profile ?", "Cookies"],
        ),
        "question mark matches exactly one character",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Prof*", "Cookies"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "both segments wildcard",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 3", "Cook?es"],
            la,
            &["User Data", "Profile *", "Cookies"],
        ),
        "cleanable wildcard against protected literal",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default", "Network", "Cache"],
            la,
            &["User Data", "Default", "Network"],
        ),
        "deeper cleanable is not checked by this rule",
    );

    let raw: Value =
        serde_json::from_str(include_str!("../rules/windows.json")).expect("seed json");
    let subset: Vec<Value> = raw["rules"]
        .as_array()
        .expect("rules")
        .iter()
        .filter(|rule| {
            let id = rule["rule_id"].as_str().unwrap_or("");
            id.ends_with("-code-gpu-cache") || id.ends_with("-login-and-site-data")
        })
        .cloned()
        .collect();
    assert_eq!(subset.len(), 6);
    let shader_locators = subset
        .iter()
        .flat_map(|rule| rule["locate"].as_array().expect("locate").iter())
        .filter(|locator| {
            let segments = locator["segments"].as_array().expect("segments");
            segments.len() == 4
                && ["ShaderCache", "GrShaderCache", "GraphiteDawnCache"]
                    .contains(&segments[3].as_str().unwrap_or(""))
        })
        .count();
    assert_eq!(shader_locators, 9);
    let subset_set = RuleSet::from_json(
        &json!({
            "rules_version": "water.junk-rules.v1",
            "data_version": "2026-10-26.2",
            "rules": subset
        })
        .to_string(),
    )
    .expect("root shader caches must not collide with protected profiles");
    assert_eq!(subset_set.rules().len(), 6);
    let embedded = RuleSet::embedded();
    assert_eq!(embedded.rules().len(), 110, "seed set must load");

    let agent_rules: Value = serde_json::from_str(include_str!("../rules/windows.json"))
        .expect("RULES_AGENT_TOOLS: embedded JSON");
    let agent_rule = |id: &str| {
        agent_rules["rules"]
            .as_array()
            .expect("RULES_AGENT_TOOLS: rule array")
            .iter()
            .find(|rule| rule["rule_id"] == id)
            .cloned()
            .expect("RULES_AGENT_TOOLS: rule exists")
    };
    let siblings = RuleSet::from_json(&wrap(vec![
        agent_rule("agent-tool-logs"),
        agent_rule("kimi-code-sessions"),
    ]))
    .expect("RULES_AGENT_TOOLS: Kimi logs and protected sessions are siblings");
    assert_eq!(
        siblings.rules()[0].level,
        Level::Recommended,
        "RULES_AGENT_TOOLS: logs remain cleanable"
    );
    assert_eq!(
        siblings.rules()[1].level,
        Level::Protected,
        "RULES_AGENT_TOOLS: sessions remain protected"
    );
    let mut numbered_home = sample_rule();
    numbered_home["locate"] =
        json!([{"kind":"pattern","base":"user_profile","segments":[".grok7"]}]);
    let overlap = wrap(vec![numbered_home.clone(), agent_rule("grok-other-homes")]);
    let err = RuleSet::from_json(&overlap)
        .expect_err("RULES_AGENT_TOOLS: .grok7 is covered by protected .grok#");
    assert_eq!(
        err.code, "RULES_PRECEDENCE",
        "RULES_AGENT_TOOLS: reject cleaning a protected numbered home"
    );
    numbered_home["locate"][0]["segments"][0] = json!(".grokx");
    RuleSet::from_json(&wrap(vec![numbered_home, agent_rule("grok-other-homes")]))
        .expect("RULES_AGENT_TOOLS: # requires digits, not an arbitrary suffix");
}
#[test]
fn rules_resolve_plan() {
    m35_resolve();
    let rules = RuleSet::embedded();
    let folders = KnownFolders::from_json(&folders_json()).expect("folders");
    let custom = [
        CustomRoot {
            root_id: "alpha".to_string(),
            path: "D:\\work\\alpha".to_string(),
        },
        CustomRoot {
            root_id: "beta".to_string(),
            path: "E:\\work\\beta".to_string(),
        },
    ];
    let plan = resolve_plan(&rules, &folders, &custom, CREATED_AT).expect("plan");
    m3d_plan_targets(&plan, &custom);
    let encoded = plan_to_json(&plan);
    let decoded = ScanPlan::from_json(&encoded).expect("RULES_M3D_CONDITIONS: parse new JSON");
    assert_eq!(
        decoded, plan,
        "RULES_M3D_CONDITIONS: complete plan roundtrip"
    );
    assert_eq!(
        plan_to_json(&decoded),
        encoded,
        "RULES_M3D_CONDITIONS: byte roundtrip"
    );
    let wire: Value = serde_json::from_str(&encoded).expect("RULES_M3D_CONDITIONS: JSON");
    let main_index = plan
        .targets
        .iter()
        .position(|t| t.target_id == "custom:alpha:node-modules#1")
        .unwrap();
    let fallback_index = plan
        .targets
        .iter()
        .position(|t| t.target_id == "custom:alpha:bundled-node-modules#1")
        .unwrap();
    for conditions in [
        json!([]),
        json!(null),
        json!([{"kind":"unknown","name":"a"}]),
        json!([{"kind":"parent_has_file","name":"bad name"}]),
    ] {
        let mut invalid = wire.clone();
        invalid["targets"][main_index]["match"]["conditions"] = conditions;
        ScanPlan::from_json(&invalid.to_string())
            .expect_err("RULES_M3D_CONDITIONS: strict plan condition validation");
    }
    for case in 0..10 {
        let mut invalid = wire.clone();
        let targets = invalid["targets"].as_array_mut().unwrap();
        match case {
            0 => targets[fallback_index]["match"]["fallback_for"] = json!("absent"),
            1 => {
                targets[fallback_index]["match"]["fallback_for"] =
                    targets[fallback_index]["target_id"].clone()
            }
            2 => {
                targets[main_index]["match"]
                    .as_object_mut()
                    .unwrap()
                    .remove("conditions");
            }
            3 => {
                targets[fallback_index]["match"]["conditions"] = m3d_conditions_json("node-modules")
            }
            4 => {
                targets[main_index]["match"]["fallback_for"] =
                    targets[fallback_index]["target_id"].clone()
            }
            5 => targets[fallback_index]["match"]["root"] = json!("F:\\other"),
            6 => targets[fallback_index]["match"]["marker"]["names"] = json!(["other"]),
            7 => targets[fallback_index]["match"]["max_depth"] = json!(7),
            8 => {
                let mut duplicate = targets[fallback_index].clone();
                duplicate["target_id"] = json!("another-fallback");
                targets.push(duplicate);
            }
            _ => {
                targets[main_index]["match"] =
                    json!({"kind":"pattern","base":"F:\\alpha","segments":["node_modules"]})
            }
        }
        ScanPlan::from_json(&invalid.to_string())
            .expect_err("RULES_M3D_FALLBACK: strict plan fallback validation");
    }
    let mut case_insensitive_root = wire.clone();
    case_insensitive_root["targets"][fallback_index]["match"]["root"] =
        json!(custom[0].path.to_ascii_lowercase());
    ScanPlan::from_json(&case_insensitive_root.to_string())
        .expect("RULES_M3D_FALLBACK: ASCII case insensitive root");
    for (target, value) in plan
        .targets
        .iter()
        .zip(wire["targets"].as_array().unwrap())
        .filter(|(target, _)| {
            rules.rules()[..78]
                .iter()
                .any(|rule| rule.rule_id == target.rule_id)
        })
    {
        let match_json = &value["match"];
        match &target.match_spec {
            MatchSpec::Marker {
                root,
                marker,
                max_depth,
                conditions,
                fallback_for,
            } => {
                if conditions.is_empty() {
                    assert!(
                        match_json.get("conditions").is_none(),
                        "RULES_M3D_CONDITIONS: empty omitted"
                    );
                } else if ["node-modules", "test-browser-profiles"]
                    .contains(&target.rule_id.as_str())
                {
                    assert_eq!(
                        match_json["conditions"],
                        m3d_conditions_json(&target.rule_id),
                        "RULES_M3D_CONDITIONS: wire conditions"
                    );
                }
                if let Some(id) = fallback_for {
                    assert_eq!(
                        match_json["fallback_for"], *id,
                        "RULES_M3D_FALLBACK: wire target ID"
                    );
                } else {
                    assert!(
                        match_json.get("fallback_for").is_none(),
                        "RULES_M3D_CONDITIONS: absent fallback omitted"
                    );
                }
                // Reproduce the old serializer's field order explicitly for every unchanged marker.
                let marker_json = match marker {
                    Marker::File {
                        name,
                        min_bytes,
                        max_bytes,
                    } => format!(
                        "{{\"kind\":\"file\",\"name\":{},\"min_bytes\":{min_bytes},\"max_bytes\":{max_bytes}}}",
                        json!(name)
                    ),
                    Marker::ChildrenAll { names } => {
                        format!("{{\"kind\":\"children_all\",\"names\":{}}}", json!(names))
                    }
                    Marker::DirName { names } => {
                        format!("{{\"kind\":\"dir_name\",\"names\":{}}}", json!(names))
                    }
                    Marker::DirNamePattern { patterns } => format!(
                        "{{\"kind\":\"dir_name_pattern\",\"patterns\":{}}}",
                        json!(patterns)
                    ),
                    Marker::FileName { patterns } => format!(
                        "{{\"kind\":\"file_name\",\"patterns\":{}}}",
                        json!(patterns)
                    ),
                };
                let legacy = format!(
                    "{{\"kind\":\"marker\",\"root\":{},\"marker\":{marker_json},\"max_depth\":{max_depth}}}",
                    json!(root)
                );
                if conditions.is_empty() && fallback_for.is_none() {
                    assert!(
                        encoded.contains(&format!("\"match\":{legacy},\"item_type\"")),
                        "RULES_M3D_CONDITIONS: unchanged marker bytes"
                    );
                } else {
                    let mut expected = legacy.trim_end_matches('}').to_string();
                    if !conditions.is_empty()
                        && ["node-modules", "test-browser-profiles"]
                            .contains(&target.rule_id.as_str())
                    {
                        // serde_json::Value sorts keys; build condition objects in the frozen wire order.
                        let condition_json = m3d_conditions_json(&target.rule_id)
                            .as_array()
                            .unwrap()
                            .iter()
                            .map(|c| {
                                let kind = c["kind"].as_str().unwrap();
                                let key = match kind {
                                    "parent_has_file" => "name",
                                    "parent_has_other_entries" => "except",
                                    "no_ancestor_suffix" => "suffixes",
                                    _ => "names",
                                };
                                format!("{{\"kind\":{},\"{key}\":{}}}", json!(kind), c[key])
                            })
                            .collect::<Vec<_>>()
                            .join(",");
                        expected.push_str(&format!(",\"conditions\":[{condition_json}]"));
                    }
                    if let Some(id) = fallback_for {
                        expected.push_str(&format!(",\"fallback_for\":{}", json!(id)));
                    }
                    expected.push('}');
                    assert!(
                        encoded.contains(&format!("\"match\":{expected},\"item_type\"")),
                        "RULES_M3D_CONDITIONS: ordered new keys"
                    );
                }
            }
            _ => assert!(
                match_json.get("conditions").is_none() && match_json.get("fallback_for").is_none(),
                "RULES_M3D_CONDITIONS: unchanged non-marker shape"
            ),
        }
    }
    let mut legacy_rules_json: Value =
        serde_json::from_str(include_str!("../rules/windows.json")).unwrap();
    let old_rules = legacy_rules_json["rules"].as_array_mut().unwrap();
    old_rules.truncate(78);
    old_rules.retain(|r| {
        !["bundled-node-modules", "browser-data-dirs"].contains(&r["rule_id"].as_str().unwrap())
    });
    for rule in old_rules {
        for locator in rule["locate"].as_array_mut().unwrap() {
            locator.as_object_mut().unwrap().remove("conditions");
        }
    }
    let legacy_rules = RuleSet::from_json(&legacy_rules_json.to_string())
        .expect("RULES_M3D_CONDITIONS: old rules accepted");

                        text.matches("Flick").count()
                    } else {

                    }

            Value::Array(values) => values.iter().map(check_rename).sum(),
            _ => 0,
        }

fn expected_target_count(rules: &RuleSet, custom_roots: usize) -> usize {
    rules
        .rules()
        .iter()
        .filter(|rule| rule.session_tool.is_none())
        .map(|rule| {
            rule.locate
                .iter()
                .map(|locator| match locator {
                    Locator::Marker { .. } => custom_roots,
                    Locator::CustomDiscovery { .. } => 0,
                    _ => 1,
                })
                .sum::<usize>()
        })
        .sum()
}

fn locator_summary(rule: &water_junk_rules::Rule) -> Vec<String> {
    rule.locate
        .iter()
        .map(|locator| match locator {
            Locator::Pattern { base, segments } => {
                format!("pattern:{}:{}", base.as_str(), segments.join("/"))
            }
            Locator::Discovery { base } => format!("discovery:{}", base.as_str()),
            Locator::CustomDiscovery { root_names } => {
                format!("custom_discovery:{}", root_names.join(","))
            }
            Locator::Marker {
                marker, max_depth, ..
            } => match marker {
                Marker::File {
                    name,
                    min_bytes,
                    max_bytes,
                } => format!("marker:file:{name}:{min_bytes}:{max_bytes}:{max_depth}"),
                Marker::ChildrenAll { names } => {
                    format!("marker:children_all:{}:{max_depth}", names.join(","))
                }
                Marker::DirName { names } => {
                    format!("marker:dir_name:{}:{max_depth}", names.join(","))
                }
                Marker::DirNamePattern { patterns } => {
                    format!("marker:dir_name_pattern:{}:{max_depth}", patterns.join(","))
                }
                Marker::FileName { patterns } => {
                    format!("marker:file_name:{}:{max_depth}", patterns.join(","))
                }
            },
        })
        .collect()
}

struct SeedSpec {
    id: &'static str,
    category: Category,
    level: Level,
    item_type: ItemType,
    min_age_hours: u32,
    keep_newest: u32,
    in_use: InUseMode,
    processes: Vec<&'static str>,
    growth: Growth,
    locators: Vec<String>,
}

fn processes_for(id: &str) -> Vec<&'static str> {
    let family = |prefix: &str| -> Option<&'static str> {
        match prefix {
            "chrome" => Some("chrome.exe"),
            "edge" => Some("msedge.exe"),
            "brave" => Some("brave.exe"),
            _ => None,
        }
    };
    match id {
        "claude-desktop-old-versions" => vec!["claude.exe"],
        "claude-code-old-versions" => vec!["claude.exe"],
        "headless-browser-profiles" => {
            vec!["chrome.exe", "msedge.exe", "chrome-headless-shell.exe"]
        }
        "firefox-cache" => vec!["firefox.exe"],
        "jest-cache" => vec!["node.exe"],
        "test-browser-profiles" => vec!["chrome.exe", "msedge.exe"],
        _ => {
            let (prefix, rest) = id.split_once('-').unwrap_or((id, ""));
            match family(prefix) {
                Some(process)
                    if ["cache", "code-gpu-cache", "service-worker-cache"].contains(&rest) =>
                {
                    vec![process]
                }
                _ => Vec::new(),
            }
        }
    }
}

#[allow(clippy::too_many_arguments)]
fn seed(
    id: &'static str,
    category: Category,
    level: Level,
    item_type: ItemType,
    min_age_hours: u32,
    keep_newest: u32,
    in_use: InUseMode,
    growth: Growth,
    locators: Vec<String>,
) -> SeedSpec {
    SeedSpec {
        id,
        category,
        level,
        item_type,
        min_age_hours,
        keep_newest,
        in_use,
        processes: processes_for(id),
        growth,
        locators,
    }
}

fn p(base: &str, segments: &str) -> String {
    format!("pattern:{base}:{segments}")
}
