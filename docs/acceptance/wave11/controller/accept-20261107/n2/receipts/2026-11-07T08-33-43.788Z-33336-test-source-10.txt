//! The only process-launching adapter: Explorer receives one checked folder argument.
use std::{fs, path::Path, process::Command};

#[allow(clippy::result_unit_err)]
pub trait Opener: Send + Sync {
    fn open_folder(&self, path: &str) -> Result<(), ()>;
}

pub struct ExplorerOpener;

// Reject alternate namespaces, shell delimiters and ambiguous Win32 spellings.
fn local_disk_path(path: &str) -> bool {
    let bytes = path.as_bytes();
    bytes.len() > 3
        && bytes[0].is_ascii_alphabetic()
        && bytes[1..3] == *b":\\"
        && !path
            .chars()
            .any(|c| c.is_control() || "/\",%;<>|?*".contains(c))
        && path[3..].split('\\').all(|part| {
            !part.is_empty()
                && part != "."
                && part != ".."
                && !part.contains(':')
                && !part.ends_with(['.', ' '])
        })
}

fn plain_metadata(path: &Path) -> Option<fs::Metadata> {
    let metadata = fs::symlink_metadata(path).ok()?;
    if metadata.file_type().is_symlink() {
        return None;
    }
    #[cfg(windows)]
    {
        use std::os::windows::fs::MetadataExt;
        if metadata.file_attributes() & 0x400 != 0 {
            return None;
        }
    }
    Some(metadata)
}

pub(crate) fn valid_quarantine_path(path: &str) -> bool {
    if !local_disk_path(path) {
        return false;
    }
    let mut parts = path.rsplit('\\');
    let last = parts.next().unwrap_or_default();
    let is_quarantine = |name: &str| {
        name == ".flick-quarantine"
            || name.eq_ignore_ascii_case(".water-quarantine")
    };
    let quarantine = is_quarantine(last)
        || (parts.next().is_some_and(is_quarantine)
            && (1..=48).contains(&last.len())
            && last.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-'));
    quarantine && plain_metadata(Path::new(path)).is_some_and(|metadata| metadata.is_dir())
}

impl Opener for ExplorerOpener {
    fn open_folder(&self, path: &str) -> Result<(), ()> {
        // Recheck immediately before launching, including for callers outside IPC.
        if !valid_quarantine_path(path) {
            return Err(());
        }
        let system_root = std::env::var("SystemRoot").map_err(|_| ())?;
        if !local_disk_path(&system_root) {
            return Err(());
        }
        let explorer = Path::new(&system_root).join("explorer.exe");
        if !plain_metadata(&explorer).is_some_and(|metadata| metadata.is_file()) {
            return Err(());
        }
        Command::new(explorer)
            .arg(path)
            .spawn()
            .map(|_| ())
            .map_err(|_| ())
    }
}
