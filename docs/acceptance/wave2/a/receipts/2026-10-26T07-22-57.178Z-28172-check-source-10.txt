fn wrap(rules: Vec<Value>) -> String {
    json!({
        "rules_version": "water.junk-rules.v1",
        "data_version": "2026-10-26.1",
        "rules": rules
    })
    .to_string()
}

fn reject(input: &str, code: &str) {
    let err = RuleSet::from_json(input).expect_err("schema must reject");
    let shown = err.to_string();
    assert!(!shown.contains(CANARY), "error echoed input: {shown}");
    assert_eq!(err.code, code, "unexpected error code: {shown}");
    assert!(
        shown.contains(code),
        "error text must carry its code: {shown}"
    );
}

fn mutate_rule(mutator: impl FnOnce(&mut Value)) -> String {
    let mut rule = sample_rule();
    mutator(&mut rule);
    wrap(vec![rule])
}

fn precedence_rule(id: &str, level: &str, base: &str, segments: &[&str]) -> Value {
    let mut rule = sample_rule();
    rule["rule_id"] = json!(id);
    rule["title_zh"] = json!("优先判定样例");
    rule["category"] = json!("browser");
    rule["level"] = json!(level);
    rule["locate"] = json!([{"kind":"pattern","base":base,"segments":segments}]);
    if level == "protected" {
        rule["what_zh"] = json!("这是登录相关的网站数据。");
        rule["why_zh"] = json!("清掉之后需要重新登录许多网站。");
        rule["after_zh"] = json!("这里只统计占用，不会被清理。");
    }
    rule
}

fn precedence_pair(
    level: &str,
    base: &str,
    cleanable: &[&str],
    protected_base: &str,
    protected: &[&str],
) -> String {
    wrap(vec![
        precedence_rule("cleanable-sample", level, base, cleanable),
        precedence_rule("protected-sample", "protected", protected_base, protected),
    ])
}

fn expect_precedence(input: &str, case: &str) {
    let err = RuleSet::from_json(input)
        .expect_err(&format!("{case}: expected RULES_PRECEDENCE"));
    assert_eq!(err.code, "RULES_PRECEDENCE", "{case}: {err}");
    assert!(
        err.to_string().contains("RULES_PRECEDENCE"),
        "{case}: {err}"
    );
}

fn expect_no_precedence(input: &str, case: &str) {
    if let Err(err) = RuleSet::from_json(input) {
        panic!("{case} must pass, got {err}");
    }
}

#[test]
fn rules_protection_precedence() {
    let la = "local_app_data";
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "*"],
            la,
            &["User Data", "*", "Cookies"],
        ),
        "wildcard profile over wildcard cookies",
    );
    expect_no_precedence(
        &precedence_pair("recommended", la, &["*", "Cache"], la, &["*", "Network"]),
        "sibling cache and network",
    );
    let profiles = [
        ["User Data", "Default", "Cookies"],
        ["User Data", "Profile *", "Cookies"],
        ["User Data", "Guest Profile", "Cookies"],
        ["User Data", "System Profile", "Cookies"],
    ];
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default"],
            la,
            &profiles[0],
        ),
        "literal Default over Default cookies",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default", "Network"],
            la,
            &["User Data", "Default", "Network"],
        ),
        "literal Default Network equals protected",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 3"],
            la,
            &profiles[1],
        ),
        "literal Profile 3 matched by Profile *",
    );
    expect_precedence(
        &precedence_pair("confirm", la, &["user data", "profile 3"], la, &profiles[1]),
        "case-insensitive literal against pattern",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "ShaderCache"],
            la,
            &profiles[1],
        ),
        "root shader cache against Profile *",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default"],
            "user_profile",
            &profiles[0],
        ),
        "different base",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            "user_profile",
            &[".codex", "sessions"],
            "user_profile",
            &[".codex*", "sessions"],
        ),
        "literal .codex sessions matched by .codex*",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 12"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash matches digits",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile x"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash needs digits",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile "],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "hash needs at least one digit",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 7"],
            la,
            &["User Data", "Profile ?", "Cookies"],
        ),
        "question mark matches one character",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 12"],
            la,
            &["User Data", "Profile ?", "Cookies"],
        ),
        "question mark matches exactly one character",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Prof*", "Cookies"],
            la,
            &["User Data", "Profile #", "Cookies"],
        ),
        "both segments wildcard",
    );
    expect_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Profile 3", "Cook?es"],
            la,
            &["User Data", "Profile *", "Cookies"],
        ),
        "cleanable wildcard against protected literal",
    );
    expect_no_precedence(
        &precedence_pair(
            "recommended",
            la,
            &["User Data", "Default", "Network", "Cache"],
            la,
            &["User Data", "Default", "Network"],
        ),
        "deeper cleanable is not checked by this rule",
    );

    let raw: Value =
        serde_json::from_str(include_str!("../rules/windows.json")).expect("seed json");
    let subset: Vec<Value> = raw["rules"]
        .as_array()
        .expect("rules")
        .iter()
        .filter(|rule| {
            let id = rule["rule_id"].as_str().unwrap_or("");
            id.ends_with("-code-gpu-cache") || id.ends_with("-login-and-site-data")
        })
        .cloned()
        .collect();
    assert_eq!(subset.len(), 6);
    let shader_locators = subset
        .iter()
        .flat_map(|rule| rule["locate"].as_array().expect("locate").iter())
        .filter(|locator| {
            let segments = locator["segments"].as_array().expect("segments");
            segments.len() == 4
                && ["ShaderCache", "GrShaderCache", "GraphiteDawnCache"]
                    .contains(&segments[3].as_str().unwrap_or(""))
        })
        .count();
    assert_eq!(shader_locators, 9);
    let subset_set = RuleSet::from_json(
        &json!({
            "rules_version": "water.junk-rules.v1",
            "data_version": "2026-10-26.2",
            "rules": subset
        })
        .to_string(),
    )
    .expect("root shader caches must not collide with protected profiles");
    assert_eq!(subset_set.rules().len(), 6);
    let embedded = RuleSet::embedded();
    assert_eq!(embedded.rules().len(), 60, "seed set must load");
}
#[test]
fn rules_resolve_plan() {
    let rules = RuleSet::embedded();
    let folders = KnownFolders::from_json(&folders_json()).expect("folders");
    let custom = [
        CustomRoot {
            root_id: "alpha".to_string(),
            path: "D:\\work\\alpha".to_string(),
        },
        CustomRoot {
            root_id: "beta".to_string(),
            path: "E:\\work\\beta".to_string(),
        },
    ];
    let plan = resolve_plan(&rules, &folders, &custom, CREATED_AT).expect("plan");
    let expected = expected_target_count(&rules, custom.len());
    assert_eq!(plan.targets.len(), expected);
    assert_eq!(expected, 214);
    assert_eq!(plan.plan_version, "water.scan-plan.v1");
    assert_eq!(plan.rules_data_version, "2026-10-26.2");
    assert_eq!(plan.created_at, CREATED_AT);

    let first = &plan.targets[0];
    assert_eq!(first.target_id, "codex-marketplace-staging#1");
    assert_eq!(first.group_id, "codex-marketplace-staging");
    assert_eq!(first.rule_id, "codex-marketplace-staging");
    match &first.match_spec {
        MatchSpec::Pattern { base, segments } => {
            assert_eq!(base, "C:\\Users\\u");
            assert_eq!(
                segments,
                &[
                    ".codex*".to_string(),
                    ".tmp".to_string(),
                    "marketplaces".to_string(),
                    ".staging".to_string(),
                    "marketplace-upgrade-*".to_string()
                ]
            );
        }
        other => panic!("first target {other:?}"),
    }

    let chrome = plan
        .targets
        .iter()
        .find(|target| target.target_id == "chrome-cache#1")
        .expect("chrome-cache");
    assert_eq!(chrome.group_id, "chrome-cache");
    assert_eq!(chrome.level, Level::Recommended);
    assert_eq!(chrome.category, Category::Browser);
    assert_eq!(chrome.in_use.mode, InUseMode::ProcessRunning);
    assert_eq!(chrome.in_use.processes, ["chrome.exe"]);

    for (family, ud) in [
        ("chrome", ["Google", "Chrome", "User Data"]),
        ("edge", ["Microsoft", "Edge", "User Data"]),
        ("brave", ["BraveSoftware", "Brave-Browser", "User Data"]),
    ] {
        let rule_id = format!("{family}-login-and-site-data");
        let positions: Vec<usize> = plan
            .targets
            .iter()
            .enumerate()
            .filter(|(_, target)| target.rule_id == rule_id)
            .map(|(index, _)| index)
            .collect();
        assert_eq!(positions.len(), 32, "{rule_id}");
        assert_eq!(positions[31] - positions[0], 31, "{rule_id} contiguous");
        let mut ordinal = 0;
        for profile in ["Default", "Profile *", "Guest Profile", "System Profile"] {
            for name in LOGIN_NAMES {
                let target = &plan.targets[positions[ordinal]];
                ordinal += 1;
                assert_eq!(target.target_id, format!("{rule_id}#{ordinal}"));
                assert_eq!(target.group_id, rule_id);
                assert_eq!(target.level, Level::Protected);
                match &target.match_spec {
                    MatchSpec::Pattern { base, segments } => {
                        assert_eq!(base, "C:\\Users\\u\\AppData\\Local");
                        let expected_segments: Vec<String> = ud
                            .iter()
                            .copied()
                            .chain([profile, name])
                            .map(str::to_string)
                            .collect();
                        assert_eq!(segments, &expected_segments, "{rule_id}#{ordinal}");
                    }
                    other => panic!("{rule_id}#{ordinal} {other:?}"),
                }
            }
        }
    }

    let marker_targets: Vec<_> = plan
        .targets
        .iter()
        .filter(|target| target.rule_id == "cachedir-tagged")
        .collect();
    assert_eq!(marker_targets.len(), 2);
    assert_eq!(marker_targets[0].group_id, "custom:alpha:cachedir-tagged");
    assert_eq!(
        marker_targets[0].target_id,
        "custom:alpha:cachedir-tagged#1"
    );
    assert_eq!(marker_targets[1].group_id, "custom:beta:cachedir-tagged");
    let text = rules
        .text("custom:alpha:cachedir-tagged")
        .expect("custom text");
    assert_eq!(text.title_zh, "带缓存标记的目录");

    assert_eq!(
        plan.hard_guards,
        vec![
            "C:\\Users\\u".to_string(),
            "C:\\Users\\u\\AppData\\Local".to_string(),
            "C:\\Users\\u\\AppData\\Roaming".to_string(),
            "C:\\Users\\u\\AppData\\Local\\Temp".to_string(),
            "C:\\Windows\\Temp".to_string(),
            "C:\\Windows".to_string(),
            "C:\\ProgramData".to_string(),
            "C:\\Users\\u\\Downloads".to_string(),
            "C:\\".to_string(),
            "D:\\work\\alpha".to_string(),
            "E:\\work\\beta".to_string(),
        ]
    );

    let encoded = plan_to_json(&plan);
    let parsed = serde_json::from_str::<Value>(&encoded).expect("json");
    assert_eq!(parsed["plan_version"], "water.scan-plan.v1");
    assert_eq!(parsed["targets"].as_array().map(Vec::len), Some(expected));
    assert!(parsed["targets"][0].get("match").is_some());
    let roundtrip = ScanPlan::from_json(&encoded).expect("strict parse back");
    assert_eq!(roundtrip, plan);
}

#[test]
fn rules_seed_coverage() {
    let rules = RuleSet::embedded();
    let expected = seed_table();
    assert_eq!(rules.rules().len(), expected.len());
    for (index, spec) in expected.iter().enumerate() {
        let rule = &rules.rules()[index];
        assert_eq!(rule.rule_id, spec.id, "order {index}");
        assert_eq!(rule.category, spec.category, "{}", spec.id);
        assert_eq!(rule.level, spec.level, "{}", spec.id);
        assert_eq!(rule.item_type, spec.item_type, "{}", spec.id);
        assert_eq!(rule.min_age_hours, spec.min_age_hours, "{}", spec.id);
        assert_eq!(rule.keep_newest, spec.keep_newest, "{}", spec.id);
        assert_eq!(rule.in_use.mode, spec.in_use, "{}", spec.id);
        assert_eq!(rule.in_use.processes, spec.processes, "{}", spec.id);
        assert_eq!(rule.growth, spec.growth, "{}", spec.id);
        assert_eq!(locator_summary(rule), spec.locators, "{}", spec.id);
        assert!(!rule.title_zh.is_empty());
        assert!(!rule.what_zh.contains("垃圾"));
        assert!(!rule.why_zh.contains("垃圾"));
        assert!(!rule.after_zh.contains("垃圾"));
        assert!(!rule.what_zh.contains(&rule.rule_id));
        assert!(!rule.why_zh.contains(&rule.rule_id));
        assert!(!rule.after_zh.contains(&rule.rule_id));
        assert!(!rule.sources.is_empty());
        let text = rules.text(&rule.rule_id).expect("text");
        assert_eq!(text.title_zh, rule.title_zh);
    }
}

fn expected_target_count(rules: &RuleSet, custom_roots: usize) -> usize {
    rules
        .rules()
        .iter()
        .map(|rule| {
            rule.locate
                .iter()
                .map(|locator| match locator {
                    Locator::Marker { .. } => custom_roots,
                    _ => 1,
                })
                .sum::<usize>()
        })
        .sum()
}

fn locator_summary(rule: &water_junk_rules::Rule) -> Vec<String> {
    rule.locate
        .iter()
        .map(|locator| match locator {
            Locator::Pattern { base, segments } => {
                format!("pattern:{}:{}", base.as_str(), segments.join("/"))
            }
            Locator::Discovery { base } => format!("discovery:{}", base.as_str()),
            Locator::Marker { marker, max_depth } => match marker {
                Marker::File {
                    name,
                    min_bytes,
                    max_bytes,
                } => format!("marker:file:{name}:{min_bytes}:{max_bytes}:{max_depth}"),
                Marker::ChildrenAll { names } => {
                    format!("marker:children_all:{}:{max_depth}", names.join(","))
                }
                Marker::DirName { names } => {
                    format!("marker:dir_name:{}:{max_depth}", names.join(","))
                }
            },
        })
        .collect()
}

struct SeedSpec {
    id: &'static str,
    category: Category,
    level: Level,
    item_type: ItemType,
    min_age_hours: u32,
    keep_newest: u32,
    in_use: InUseMode,
    processes: Vec<&'static str>,
    growth: Growth,
    locators: Vec<String>,
}

fn processes_for(id: &str) -> Vec<&'static str> {
    let family = |prefix: &str| -> Option<&'static str> {
        match prefix {
            "chrome" => Some("chrome.exe"),
            "edge" => Some("msedge.exe"),
            "brave" => Some("brave.exe"),
            _ => None,
        }
    };
    match id {
        "claude-desktop-old-versions" => vec!["claude.exe"],
        "headless-browser-profiles" => {
            vec!["chrome.exe", "msedge.exe", "chrome-headless-shell.exe"]
        }
        "firefox-cache" => vec!["firefox.exe"],
        "test-browser-profiles" => vec!["chrome.exe", "msedge.exe"],
        _ => {
            let (prefix, rest) = id.split_once('-').unwrap_or((id, ""));
            match family(prefix) {
                Some(process)
                    if ["cache", "code-gpu-cache", "service-worker-cache"].contains(&rest) =>
                {
                    vec![process]
                }
                _ => Vec::new(),
            }
        }
    }
}

#[allow(clippy::too_many_arguments)]
fn seed(
    id: &'static str,
    category: Category,
    level: Level,
    item_type: ItemType,
    min_age_hours: u32,
    keep_newest: u32,
    in_use: InUseMode,
    growth: Growth,
    locators: Vec<String>,
) -> SeedSpec {
    SeedSpec {
        id,
        category,
        level,
        item_type,
        min_age_hours,
        keep_newest,
        in_use,
        processes: processes_for(id),
        growth,
        locators,
    }
}

fn p(base: &str, segments: &str) -> String {
    format!("pattern:{base}:{segments}")
}
