import {t, serviceText, listText, notesText} from './strings.mjs';
import {add, el, heading, card, button, row, toggle, tag} from './common.mjs';
import {formatBytes, dateOnly} from '../format.mjs';
import {about} from './about.mjs';
export function locations(s, a) {
  const current = s.route.split('/')[1], titles = {content:t('scanContent'), skip:t('skipList'), decl:t('declaration'), language:t('language'), about:t('aboutPrivacy')}, page = add(el('section', 'screen'), heading(titles[current], current === 'skip' ? t('settingsSkipNote') : current === 'language' ? t('languageNote') : current === 'about' ? '' : t('settingsNextNote')));
  const grid = el('div', 'settings-grid'), nav = card('', 'settings-nav');
  for (const [key, title] of Object.entries(titles)) {
    const b = button(title, () => a.navigate(`settings/${key}`), key === current ? 'settings-link on' : 'settings-link', `settings-${key}`); if (key === current) b.setAttribute('aria-current', 'page'); add(nav, b);
  }
  add(grid, nav); const body = el('div', 'settings-content'), value = s.locations;
  if (current === 'language') {
    const language = card(t('language'));
    const select = el('select', 'language-select', undefined, {'aria-label':t('language'), 'data-key':'language'});
    for (const [choice, label] of [['system', 'followSystem'], ['zh', 'chinese'], ['en', 'english']]) {
      const option = el('option', '', t(label), {value:choice}); option.selected = s.language === choice; add(select, option);
    }
    select.addEventListener('change', () => a.language(select.value)); add(body, add(language, select));
  } else if (!value) add(body, el('p', 'sub', t('loadingSettings')));
  else if (current === 'content') {
    const categories = card(t('commonContents'));
    for (const c of value.categories) add(categories, row(serviceText(c, 'title'), serviceText(c, 'description'), toggle(serviceText(c, 'title'), c.enabled, () => a.category(c.category, !c.enabled), `setting-${c.category}`, c.locked || a.busy())));
    const sessions = card(t('sessions'));
    add(sessions, el('p', 'sub', t('sessionWarning')));
    for (const session of value.sessions) {
      const days = el('select', 'days-select', undefined, {'aria-label':t('retentionDays', {name:serviceText(session, 'title')}), 'data-key':`days-${session.tool}`});
      for (const day of [7, 30, 90]) { const option = el('option', '', t('days', {n:day}), {value:day}); option.selected = session.days === day; add(days, option); }
      days.disabled = !session.enabled || a.busy(); days.addEventListener('change', () => a.session(session.tool, session.enabled, Number(days.value)));
      add(sessions, row(serviceText(session, 'title'), [serviceText(session, 'retention'), session.largest_bytes !== null ? t('largestSession', {size:formatBytes(session.largest_bytes)}) : ''].filter(Boolean).join(' · '), session.bytes !== null && el('span', '', formatBytes(session.bytes)), days, toggle(t('allowSessions', {name:serviceText(session, 'title'), n:session.days}), session.enabled, () => a.session(session.tool, !session.enabled, session.days), `session-${session.tool}`, a.busy())));
    }
    add(body, categories, sessions);
  } else if (current === 'skip') {
    const list = card(t('alwaysSkip'));
    for (const item of value.skipped) add(list, row(serviceText(item, 'display') ?? item.display_path, t('addedAt', {date:dateOnly(item.added_at)}), button(t('remove'), () => a.removeSkip(item.skip_id), 'btn soft', `skip-remove-${item.skip_id}`, a.busy())));
    if (!value.skipped.length) add(list, el('p', 'sub', t('skipEmpty'))); add(body, list);
  } else if (current === 'decl') {
    for (const m of value.manifests) {
      const node = card(serviceText(m, 'project') ?? t('declaration'), '', [tag({trusted:t('trusted'), untrusted:t('untrusted'), changed:t('changed'), invalid:t('hasErrors')}[m.state], m.state === 'trusted' ? 'lime' : 'hot')]);
      add(node, el('code', 'path', m.display_path), el('p', 'sub', t('declarationCounts', {protected:m.protected_count, trees:m.worktree_count})));
      for (const d of m.disposable) add(node, row(serviceText(d, 'title'), t('pathCount', {kind:d.kind, n:d.paths_count, ask:d.ask ? t('askSuffix') : ''})));
      if (serviceText(m, 'error')) add(node, el('p', 'warning', serviceText(m, 'error')));
      add(node, button(m.state === 'trusted' ? t('distrust') : t('trust'), () => a.trust(m.manifest_id, m.state !== 'trusted'), 'btn soft', `trust-${m.manifest_id}`, m.state === 'invalid' || a.busy())); add(body, node);
    }
    if (!value.manifests.length) add(body, card(t('noDeclarations')));
  } else add(body, about(s, a));
  const privacy = add(card(t('privacy'), 'dark main-card'), row(t('localCheckup'), t('metadataAttributes')), row(t('offline'), t('noUpload')), row(t('cleanupConsent'), t('quarantineSafety')), row(t('declarationPriority'), t('protectedDeclarations')), el('p', 'sub bottom', t('localRecords', {version:s.app?.app_version ?? ''})));
  return add(page, add(grid, body, privacy));
}
