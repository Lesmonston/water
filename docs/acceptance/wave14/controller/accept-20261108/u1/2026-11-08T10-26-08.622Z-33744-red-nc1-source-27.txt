export const strings = Object.freeze(Object.fromEntries(['zh', 'en'].map((lang, i) => [lang, Object.freeze(Object.fromEntries(Object.entries(pairs).map(([key, values]) => [key, values[i]])))])));
export const languageKey = 'flick.language.v1';
let language = 'zh';
export const currentLanguage = () => language;
export function resolveLanguage(choice, systemLanguage, browserLanguage = globalThis.navigator?.language ?? '') {
  if (choice === 'zh' || choice === 'en') return choice;
  return /^zh/i.test(systemLanguage?.trim() || browserLanguage) ? 'zh' : 'en';
}
export function setLanguage(choice, systemLanguage, browserLanguage) { language = resolveLanguage(choice, systemLanguage, browserLanguage); return language; }
export function readLanguage() { try { const value = localStorage.getItem(languageKey); return ['zh', 'en'].includes(value) ? value : 'system'; } catch { return 'system'; } }
export function saveLanguage(value) { try { localStorage.setItem(languageKey, value); } catch {} }
export function t(key, params = {}) {
  const value = strings[language][key];
  if (value === undefined) throw new Error(`Unknown text key: ${key}`);
  return value.replace(/\{(\w+)\}/g, (_, name) => {
    const part = params[name];
    return typeof part === 'number' && !['year', 'month', 'day'].includes(name) ? part.toLocaleString('en-US') : String(part ?? '');
  });
}
export const errorText = value => value?.key ? t(value.key) : serviceText(value, 'message') || t('serviceUnavailable');
export function serviceText(value, name) {
  const base = name === 'copy_text' ? name : `${name}_zh`;
  const english = name === 'copy_text' ? 'copy_text_en' : `${name}_en`;
  return value?.[base];
}
export function listText(values) { return language === 'zh' ? values.join('、') : new Intl.ListFormat('en-US', {style:'long', type:'conjunction'}).format(values); }
export function notesText(values) { return values.join(language === 'zh' ? '；' : '; '); }
