use std::collections::{BTreeSet, HashMap};
use std::os::windows::process::CommandExt;
use std::path::{Path, PathBuf};
use std::process::{Child, Command, Stdio};
use std::thread;
use std::time::{Duration, SystemTime, UNIX_EPOCH};

use serde_json::Value;
use water_process_scan::{
    DEFAULT_ORPHAN_PATTERNS, InUseMode, InUseSpec, OrphanChain, ProcessSnapshot, ProcessSummary,
    ProcessSummaryEntry, in_use, orphans, snapshot, summary_to_json,
};

const CREATE_NO_WINDOW: u32 = 0x0800_0000;
/// Only these image names are ever ended by cleanup, and only when their
/// command line still carries this session's token.
const KILL_NAMES: &[&str] = &["cmd.exe", "waitfor.exe", "PING.EXE"];
/// Self-limiting wait: even if cleanup were skipped, waiters exit on their own.
const WAIT_SECS: u32 = 60;

/// One test's processes. Every process it starts carries `token` in its
/// command line: each `cmd.exe` as a `rem` word, each `waitfor.exe` as its
/// unique signal name `<token>s<n>`.
struct Session {
    token: String,
    dir: PathBuf,
    started_at: String,
    children: Vec<Child>,
    signals: Vec<String>,
    expected: usize,
    recorded: HashMap<u32, Option<String>>,
}

impl Session {
    fn new() -> Self {
        // waitfor signal names allow only ASCII letters and digits.
        let token = format!(
            "w2c{}t{:x}",
            std::process::id(),
            SystemTime::now()
                .duration_since(UNIX_EPOCH)
                .expect("clock")
                .as_nanos()
        );
        let fixtures = worktree_root().join(".local").join("w8-k1-fixtures");
        std::fs::create_dir_all(&fixtures).expect("create fixture parent");
        let dir = fixtures.join(format!("w8k1-{token}"));
        std::fs::create_dir(&dir).expect("create fixture dir");
        let started_at = snapshot().expect("snapshot").summary().taken_at;
        Self {
            token,
            dir,
            started_at,
            children: Vec::new(),
            signals: Vec::new(),
            expected: 0,
            recorded: HashMap::new(),
        }
    }

    fn path_str(&self) -> String {
        self.dir.to_string_lossy().into_owned()
    }

    fn next_signal(&mut self) -> String {
        let signal = format!("{}s{}", self.token, self.signals.len() + 1);
        self.signals.push(signal.clone());
        signal
    }

    /// `cmd.exe` (token in its command line) waiting on a token-named
    /// `waitfor.exe`. `extra` ends the command line verbatim.
    fn spawn_ping(&mut self, extra: &str) -> u32 {
        let signal = self.next_signal();
        let script = format!(
            "waitfor /t {WAIT_SECS} {signal} >NUL 2>&1 & rem {} --user-data-dir=\"{}\" {extra}",
            self.token,
            self.path_str()
        );
        self.expected += 2;
        let child = cmd(&script);
        let pid = child.id();
        self.children.push(child);
        pid
    }

    /// Outer `cmd.exe` starts an inner `cmd.exe` with `start /b` and exits,
    /// leaving the inner one (and its token-named `waitfor.exe`) orphaned.
    /// Returns the waitfor signal name.
    fn spawn_orphan(&mut self) -> String {
        let signal = self.next_signal();
        let inner = format!(
            "waitfor /t {WAIT_SECS} {signal} >NUL 2>&1 & rem {} jest-worker --user-data-dir={}",
            self.token,
            self.path_str()
        );
        self.expected += 3;
        let mut outer = cmd(&format!("start \"\" /b cmd /d /c \"{inner}\""));
        let _ = outer.wait();
        signal
    }

    /// Launch the copied PING using a relative executable spelling. Its
    /// command line carries the unique fixture token but no absolute item
    /// path, so only the new image-path branch can match that absolute item.
    fn spawn_image(&mut self) {
        let bin = self.dir.join("bin");
        std::fs::create_dir(&bin).expect("PROC_M3C_IMAGE: create bin");
        std::fs::copy(r"C:\Windows\System32\PING.EXE", bin.join("PING.EXE"))
            .expect("PROC_M3C_IMAGE: copy system PING into fixture");
        let relative_image = format!("w8k1-{}\\bin\\PING.EXE", self.token);
        self.signals.push(relative_image.clone());
        let script = format!(
            "{relative_image} -n {WAIT_SECS} -w 1000 127.0.0.1 >NUL 2>&1 & rem {}",
            self.token
        );
        self.expected += 2;
        self.children.push(cmd_in(&script, self.dir.parent()));
    }

    fn wait_for_image(&mut self) -> (ProcessSnapshot, u32) {
        for _ in 0..100 {
            let snap = snapshot().expect("snapshot");
            self.observe(&snap);
            let pids = self.token_pids(&snap, &["PING.EXE"]);
            if pids.len() == 1 {
                return (snap, *pids.first().expect("one image"));
            }
            thread::sleep(Duration::from_millis(50));
        }
        panic!("PROC_M3C_IMAGE: copied PING did not appear");
    }

    /// Independent test-only working-set sample; stdout is a byte count.
    /// The helper carries our token, runs hidden, exits synchronously, and
    /// never prints a process path or command line.
    fn root_working_set(&self, pid: u32) -> u64 {
        let script = format!(
            "(Get-Process -Id {pid} -ErrorAction Stop).WorkingSet64 # {}",
            self.token
        );
        let out = Command::new("powershell.exe")
            .args([
                "-NoLogo",
                "-NoProfile",
                "-NonInteractive",
                "-Command",
                &script,
            ])
            .creation_flags(CREATE_NO_WINDOW)
            .stdin(Stdio::null())
            .stderr(Stdio::null())
            .output()
            .expect("PROC_M3C_MEMORY: independent working-set sample");
        assert!(out.status.success(), "PROC_M3C_MEMORY: sample succeeded");
        String::from_utf8(out.stdout)
            .expect("PROC_M3C_MEMORY: numeric sample")
            .trim()
            .parse()
            .expect("PROC_M3C_MEMORY: byte count")
    }

    /// Every pid whose command line carries this session's token (as the
    /// bare token or as one of its waitfor signal names).
    fn token_pids(&self, snap: &ProcessSnapshot, names: &[&str]) -> BTreeSet<u32> {
        std::iter::once(&self.token)
            .chain(self.signals.iter())
            .flat_map(|word| snap.references_path(word, names))
            .collect()
    }

    /// Remember the creation time of each token process the first time it is
    /// seen (or again if the pid now belongs to a newer token process).
    fn observe(&mut self, snap: &ProcessSnapshot) {
        let created: HashMap<u32, Option<String>> = snap
            .summary()
            .processes
            .into_iter()
            .map(|p| (p.pid, p.created_at))
            .collect();
        for pid in self.token_pids(snap, &[]) {
            let now = created.get(&pid).cloned().flatten();
            if self.recorded.get(&pid) != Some(&now) {
                self.recorded.insert(pid, now);
            }
        }
    }

    fn wait_for_pids(&mut self, pids: &[u32]) -> ProcessSnapshot {
        let path = self.path_str();
        for _ in 0..100 {
            let snap = snapshot().expect("snapshot");
            self.observe(&snap);
            let have: BTreeSet<u32> = snap.references_path(&path, &[]).into_iter().collect();
            if pids.iter().all(|p| have.contains(p)) {
                return snap;
            }
            thread::sleep(Duration::from_millis(50));
        }
        panic!("expected child pids did not appear in the snapshot");
    }

    /// Waits until the homemade orphan, its token-named waitfor child and the
    /// given direct children are all visible. Returns the waitfor pid.
    fn wait_for_orphan(&mut self, signal: &str, pids: &[u32]) -> (ProcessSnapshot, u32) {
        let path = self.path_str();
        for _ in 0..100 {
            let snap = snapshot().expect("snapshot");
            self.observe(&snap);
            let chains = orphans(&snap, &[self.token.as_str()]);
            let waiter = snap.references_path(signal, &["waitfor.exe"]);
            let have: BTreeSet<u32> = snap.references_path(&path, &[]).into_iter().collect();
            if !chains.is_empty() && waiter.len() == 1 && pids.iter().all(|p| have.contains(p)) {
                return (snap, waiter[0]);
            }
            thread::sleep(Duration::from_millis(50));
        }
        panic!("homemade orphan did not appear");
    }

    /// Ends `pid` only if a fresh snapshot still shows the same process: same
    /// creation time as recorded, created after this session began, an
    /// allowed image name, and a command line that still carries the token.
    fn kill_verified(&self, pid: u32) {
        let Ok(fresh) = snapshot() else { return };
        if !self.token_pids(&fresh, KILL_NAMES).contains(&pid) {
            return;
        }
        let summary = fresh.summary();
        let Some(entry) = summary.processes.iter().find(|p| p.pid == pid) else {
            return;
        };
        let Some(Some(recorded)) = self.recorded.get(&pid) else {
            return;
        };
        let Some(now) = entry.created_at.as_ref() else {
            return;
        };
        if now != recorded || now.as_str() < self.started_at.as_str() {
            return;
        }
        taskkill(pid);
    }

    /// Returns true once no process carries the token. Never ends a process
    /// by descendant list or process tree.
    fn cleanup(&mut self) -> bool {
        for child in &mut self.children {
            let _ = child.kill();
            let _ = child.wait();
        }
        self.children.clear();
        for _ in 0..60 {
            let Ok(snap) = snapshot() else {
                thread::sleep(Duration::from_millis(50));
                continue;
            };
            self.observe(&snap);
            if self.token_pids(&snap, &[]).is_empty() {
                return true;
            }
            let candidates = self.token_pids(&snap, KILL_NAMES);
            if candidates.len() > self.expected {
                return false;
            }
            for pid in candidates {
                self.kill_verified(pid);
            }
            thread::sleep(Duration::from_millis(50));
        }
        false
    }

    fn stop(&mut self) {
        let clean = self.cleanup();
        let snap = snapshot().expect("snapshot after stop");
        assert!(clean, "cleanup must end every token process it started");
        assert!(
            self.token_pids(&snap, &[]).is_empty(),
            "no process with this session token may remain"
        );
        assert!(
            orphans(&snap, &[self.token.as_str()]).is_empty(),
            "token orphans must be gone"
        );
    }
}

impl Drop for Session {
    fn drop(&mut self) {
        let _ = self.cleanup();
        let name_ok = self
            .dir
            .file_name()
            .and_then(|n| n.to_str())
            .is_some_and(|n| n == format!("w8k1-{}", self.token));
        let parent_ok = self
            .dir
            .parent()
            .is_some_and(|p| p == worktree_root().join(".local").join("w8-k1-fixtures"));
        if name_ok && parent_ok {
            let _ = std::fs::remove_dir_all(&self.dir);
        }
    }
}

fn worktree_root() -> PathBuf {
    PathBuf::from(env!("CARGO_MANIFEST_DIR"))
        .parent()
        .and_then(Path::parent)
        .expect("packages/process-scan")
        .to_path_buf()
}

fn cmd(script: &str) -> Child {
    cmd_in(script, None)
}

fn cmd_in(script: &str, directory: Option<&Path>) -> Child {
    let mut command = Command::new("cmd.exe");
    if let Some(directory) = directory {
        command.current_dir(directory);
    }
    command
        .args(["/d", "/c"])
        .raw_arg(script)
        .creation_flags(CREATE_NO_WINDOW)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .spawn()
        .expect("spawn cmd")
}

/// Ends exactly one verified pid; no `/T`.
fn taskkill(pid: u32) {
    let _ = Command::new("taskkill")
        .args(["/F", "/PID", &pid.to_string()])
        .creation_flags(CREATE_NO_WINDOW)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .output();
}

fn object_keys(value: &Value) -> BTreeSet<&str> {
    value
        .as_object()
        .expect("object")
        .keys()
        .map(String::as_str)
        .collect()
}

fn entry(pid: u32, parent_pid: u32, created_at: Option<&str>) -> ProcessSummaryEntry {
    ProcessSummaryEntry {
        pid,
        parent_pid,
        name: "cmd.exe".to_string(),
        created_at: created_at.map(str::to_string),
    }
}

#[test]
fn process_snapshot_contract() {
    let snap = snapshot().expect("snapshot succeeds including system processes");
    let summary = snap.summary();
    let json = summary_to_json(&summary);
    assert!(
        !json.contains("1601-01-01"),
        "unknown creation times must be null, not a placeholder"
    );
    let parsed: Value = serde_json::from_str(&json).expect("summary json");
    assert_eq!(
        object_keys(&parsed),
        BTreeSet::from([
            "summary_version",
            "taken_at",
            "process_count",
            "processes",
            "issues"
        ])
    );
    assert_eq!(
        parsed["summary_version"].as_str(),
        Some("water.process-summary.v1")
    );
    let taken_at = parsed["taken_at"].as_str().expect("taken_at");
    assert!(
        regex_is_rfc3339(taken_at),
        "taken_at must be RFC 3339 UTC seconds"
    );
    let count = parsed["process_count"].as_u64().expect("process_count");
    let processes = parsed["processes"].as_array().expect("processes");
    assert_eq!(count, processes.len() as u64);
    assert!(count >= 1, "at least the test process");
    let self_pid = std::process::id();
    let me = processes
        .iter()
        .find(|p| p["pid"].as_u64() == Some(u64::from(self_pid)))
        .expect("snapshot must include the current process");
    assert!(
        me["created_at"].as_str().is_some_and(regex_is_rfc3339),
        "the current process has a readable creation time"
    );
    for process in processes {
        assert_eq!(
            object_keys(process),
            BTreeSet::from(["pid", "parent_pid", "name", "created_at"])
        );
        assert!(process["pid"].is_u64());
        assert!(process["parent_pid"].is_u64());
        assert!(process["name"].is_string());
        let created = &process["created_at"];
        assert!(
            created.is_null() || created.as_str().is_some_and(regex_is_rfc3339),
            "created_at must be RFC 3339 or null"
        );
        assert!(
            !process
                .as_object()
                .expect("obj")
                .contains_key("command_line")
        );
    }
    let issues = parsed["issues"].as_array().expect("issues");
    for issue in issues {
        assert_eq!(object_keys(issue), BTreeSet::from(["code", "count"]));
        let code = issue["code"].as_str().expect("code");
        assert!(code == "access_denied" || code == "exited" || code == "image_unavailable");
        assert!(issue["count"].as_u64().expect("count") >= 1);
    }
    let _ = serde_json::from_str::<ProcessSummary>(&json).expect("strict ProcessSummary");
    let order: Vec<&str> = summary
        .issues
        .iter()
        .map(|issue| issue.code.as_str())
        .collect();
    let expected: Vec<&str> = ["access_denied", "exited", "image_unavailable"]
        .into_iter()
        .filter(|code| order.contains(code))
        .collect();
    assert_eq!(
        order, expected,
        "PROC_M3C_IMAGE: stable nonzero issue order"
    );
    assert!(
        snap.image_path_count() >= 1,
        "PROC_M3C_IMAGE: own image readable"
    );
    assert!(
        snap.image_path_count() <= summary.process_count,
        "PROC_M3C_IMAGE: count bounded"
    );
}

#[test]
fn process_path_reference() {
    let zh_dir = "C:\\\u{7528}\u{6237}\\\u{6570}\u{636e}";
    let zh_rel = "\u{6570}\u{636e}\\\u{7f13}\u{5b58}";
    let mut session = Session::new();
    let at_end = session.spawn_ping(r"C:\a\b");
    let space_after = session.spawn_ping(r"C:\a\b tail");
    let backslash_after = session.spawn_ping(r"C:\a\b\cache");
    let slash_case = session.spawn_ping("c:/A/B");
    let double_quoted = session.spawn_ping(r#""C:\a\b""#);
    let single_quoted = session.spawn_ping(r"'C:\a\b'");
    let equals_before = session.spawn_ping(r"--flag=C:\a\b");
    let equals_after = session.spawn_ping(r"C:\a\b=1");
    let letter_before = session.spawn_ping(r"xC:\a\b");
    let digit_before = session.spawn_ping(r"9C:\a\b");
    let longer_segment = session.spawn_ping(r"C:\a\bc");
    let zh_inner = session.spawn_ping(&format!("\"{zh_dir}\\\u{7f13}\u{5b58}\""));
    let zh_longer = session.spawn_ping(&format!("{zh_dir}\u{7532}"));
    let zh_retry = session.spawn_ping(&format!("x{zh_rel} '{zh_rel}'"));
    let zh_miss = session.spawn_ping(&format!("x{zh_rel}"));
    let all = [
        at_end,
        space_after,
        backslash_after,
        slash_case,
        double_quoted,
        single_quoted,
        equals_before,
        equals_after,
        letter_before,
        digit_before,
        longer_segment,
        zh_inner,
        zh_longer,
        zh_retry,
        zh_miss,
    ];
    let snap = session.wait_for_pids(&all);

    let hits: BTreeSet<u32> = snap.references_path(r"C:\a\b", &[]).into_iter().collect();
    for (case, pid, expected) in [
        (
            "end of command line after the path must match",
            at_end,
            true,
        ),
        ("space after the path must match", space_after, true),
        (
            "\\ after the path (inner item) must match",
            backslash_after,
            true,
        ),
        ("slash and case must normalize", slash_case, true),
        (
            "double quotes before and after must match",
            double_quoted,
            true,
        ),
        (
            "single quotes before and after must match",
            single_quoted,
            true,
        ),
        ("= before the path must match", equals_before, true),
        ("= after the path must not match", equals_after, false),
        (
            "letter right before the path must not match",
            letter_before,
            false,
        ),
        (
            "digit right before the path must not match",
            digit_before,
            false,
        ),
        ("C:\\a\\bc must not match C:\\a\\b", longer_segment, false),
    ] {
        assert_eq!(hits.contains(&pid), expected, "PROC_PATH_BOUNDARY: {case}");
    }
    for query in ["c:/A/b/", r"C:\a\b\", "C:/a/b", r"c:\A\B\\"] {
        let alt: BTreeSet<u32> = snap.references_path(query, &[]).into_iter().collect();
        assert_eq!(
            alt, hits,
            "PROC_PATH_BOUNDARY: query {query:?} must normalize like C:\\a\\b"
        );
    }

    let zh_hits: BTreeSet<u32> = snap.references_path(zh_dir, &[]).into_iter().collect();
    assert!(
        zh_hits.contains(&zh_inner),
        "PROC_PATH_BOUNDARY: chinese path followed by \\ must match"
    );
    assert!(
        !zh_hits.contains(&zh_longer),
        "PROC_PATH_BOUNDARY: chinese path followed by another character must not match"
    );
    let zh_rel_hits: BTreeSet<u32> = snap.references_path(zh_rel, &[]).into_iter().collect();
    assert!(
        zh_rel_hits.contains(&zh_retry),
        "PROC_PATH_BOUNDARY: a later bounded occurrence after a rejected multibyte match must match"
    );
    assert!(
        !zh_rel_hits.contains(&zh_miss),
        "PROC_PATH_BOUNDARY: multibyte path after a letter must not match"
    );

    let cmd_hits = snap.references_path(r"C:\a\b", &["cmd.exe"]);
    assert!(
        cmd_hits.contains(&at_end),
        "PROC_PATH_BOUNDARY: name filter keeps cmd.exe"
    );
    assert!(
        !cmd_hits.contains(&longer_segment),
        "PROC_PATH_BOUNDARY: name filter keeps the boundary rule"
    );
    let chrome_hits = snap.references_path(r"C:\a\b", &["chrome.exe"]);
    assert!(
        all.iter().all(|pid| !chrome_hits.contains(pid)),
        "name filter must exclude cmd.exe"
    );
    let fixture_hits = snap.references_path(&session.path_str(), &[]);
    assert!(fixture_hits.contains(&at_end));
    assert!(fixture_hits.contains(&longer_segment));
    // The command-line half of the new mode preserves every old boundary.
    let combined: BTreeSet<u32> = snap
        .runs_from_or_references(r"C:\a\b", &[])
        .into_iter()
        .collect();
    for pid in all {
        assert_eq!(
            combined.contains(&pid),
            hits.contains(&pid),
            "PROC_M3C_IMAGE: command boundary unchanged"
        );
    }
    session.spawn_image();
    let (image_snap, image_pid) = session.wait_for_image();
    let item = session.path_str();
    assert!(
        !image_snap
            .references_path(&item, &["PING.EXE"])
            .contains(&image_pid),
        "PROC_M3C_IMAGE: image-only fixture"
    );
    for query in [
        item.clone(),
        item.to_uppercase().replace('\\', "/") + "/",
        format!("{item}\\bin\\PING.EXE"),
    ] {
        assert!(
            image_snap
                .runs_from_or_references(&query, &["ping.exe"])
                .contains(&image_pid),
            "PROC_M3C_IMAGE: item and exact executable match"
        );
    }
    for query in [format!("{item}2"), format!("{item}\\bin2"), String::new()] {
        assert!(
            !image_snap
                .runs_from_or_references(&query, &[])
                .contains(&image_pid),
            "PROC_M3C_IMAGE: sibling and empty paths excluded"
        );
    }
    session.stop();
}

#[test]
fn process_in_use_and_orphans() {
    for required in [
        "jest-worker",
        "vitest",
        "expo start",
        "cursor-agent",
        "codex",
        "grok",
        "playwright",
        "puppeteer",
        "--headless",
        "metro",
    ] {
        assert!(
            DEFAULT_ORPHAN_PATTERNS
                .iter()
                .any(|p| p.eq_ignore_ascii_case(required)),
            "missing default pattern {required}"
        );
    }

    // A record whose creation time is earlier than its recorded parent's is a
    // reused PID and never counts as a descendant; unknown times cannot prove
    // reuse.
    let crafted = ProcessSummary {
        summary_version: "water.process-summary.v1".to_string(),
        taken_at: "2026-10-26T05:00:00Z".to_string(),
        process_count: 5,
        processes: vec![
            entry(100, 4, Some("2026-10-26T04:00:00Z")),
            entry(200, 100, Some("2026-10-26T04:00:00Z")),
            entry(300, 100, Some("2026-10-26T03:59:59Z")),
            entry(400, 300, Some("2026-10-26T04:10:00Z")),
            entry(500, 200, None),
        ],
        issues: Vec::new(),
    };
    assert_eq!(crafted.descendant_pids(100), vec![200, 500]);
    assert_eq!(crafted.descendant_pids(300), vec![400]);

    let mut session = Session::new();
    let alive = session.spawn_ping("alive-parent");
    let orphan_signal = session.spawn_orphan();
    let (snap, orphan_waiter) = session.wait_for_orphan(&orphan_signal, &[alive]);
    let none = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::None,
            processes: vec!["cmd.exe".to_string()],
        },
        &session.path_str(),
    );
    assert!(none.is_none());
    let running = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::ProcessRunning,
            processes: vec!["cmd.exe".to_string()],
        },
        &session.path_str(),
    )
    .expect("cmd.exe is running");
    assert!(running.process_name.eq_ignore_ascii_case("cmd.exe"));
    let missing = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::ProcessRunning,
            processes: vec!["water-process-scan-missing.exe".to_string()],
        },
        &session.path_str(),
    );
    assert!(missing.is_none());
    let referenced = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::CmdlineReferencesItem,
            processes: Vec::new(),
        },
        &session.path_str(),
    )
    .expect("cmdline references fixture");
    assert!(referenced.process_name.eq_ignore_ascii_case("cmd.exe"));
    let filtered = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::CmdlineReferencesItem,
            processes: vec!["chrome.exe".to_string()],
        },
        &session.path_str(),
    );
    assert!(filtered.is_none());

    let chains = orphans(&snap, &[session.token.as_str()]);
    assert_eq!(chains.len(), 1, "exactly the homemade orphan is listed");
    let orphan = &chains[0];
    assert_ne!(
        orphan.root_pid, alive,
        "parent still running is not an orphan"
    );
    assert!(orphan.name.eq_ignore_ascii_case("cmd.exe"));
    assert_eq!(orphan.matched_pattern, session.token);
    assert_eq!(
        orphan.descendant_pids,
        vec![orphan_waiter],
        "descendant_pids must be exactly the token-bearing waitfor child"
    );
    assert_eq!(
        snap.summary().descendant_pids(orphan.root_pid),
        orphan.descendant_pids
    );
    // Compare against an independent sample of the quiet waiting cmd root;
    // the extra waiter must contribute too, so root-only nc2 is rejected.
    let root_bytes = session.root_working_set(orphan.root_pid);
    let child_bytes = session.root_working_set(orphan_waiter);
    let memory_snap = snapshot().expect("PROC_M3C_MEMORY: fresh snapshot");
    let memory_chains = orphans(&memory_snap, &[session.token.as_str()]);
    let chain = memory_chains
        .iter()
        .find(|c| c.root_pid == orphan.root_pid)
        .expect("PROC_M3C_MEMORY: orphan remains alive");
    let bytes = chain
        .memory_bytes
        .as_ref()
        .expect("PROC_M3C_MEMORY: known working set")
        .parse::<u64>()
        .expect("PROC_M3C_MEMORY: decimal bytes");
    assert!(
        root_bytes > 0 && bytes > root_bytes,
        "PROC_M3C_MEMORY: root and descendant working sets must be summed"
    );
    // Leave room for normal working-set movement between the independent
    // samples and snapshot, while still rejecting a root-only sum.
    assert!(
        child_bytes > 0 && bytes >= root_bytes + child_bytes / 2,
        "PROC_M3C_MEMORY: descendant contributes its working set"
    );
    let serialized = serde_json::to_value(chain).expect("PROC_M3C_MEMORY: chain JSON");
    assert!(
        serialized["memory_bytes"].is_string(),
        "PROC_M3C_MEMORY: bytes serialize as a string"
    );

    session.spawn_image();
    let (image_snap, image_pid) = session.wait_for_image();
    for names in [Vec::new(), vec!["ping.exe".to_string()]] {
        let hit = in_use(
            &image_snap,
            &InUseSpec {
                mode: InUseMode::RunsFromOrReferencesItem,
                processes: names,
            },
            &session.path_str(),
        )
        .expect("PROC_M3C_IMAGE: executable in item hits");
        assert_eq!(
            hit.pid, image_pid,
            "PROC_M3C_ANY: empty or matching names allow image"
        );
        assert!(
            hit.process_name.eq_ignore_ascii_case("PING.EXE"),
            "PROC_M3C_ANY: original hit shape"
        );
    }
    assert!(
        in_use(
            &image_snap,
            &InUseSpec {
                mode: InUseMode::RunsFromOrReferencesItem,
                processes: vec!["water-process-scan-missing.exe".to_string()]
            },
            &session.path_str()
        )
        .is_none(),
        "PROC_M3C_ANY: name filter excludes image"
    );
    for mode in [
        InUseMode::None,
        InUseMode::ProcessRunning,
        InUseMode::CmdlineReferencesItem,
    ] {
        assert!(
            in_use(
                &image_snap,
                &InUseSpec {
                    mode,
                    processes: vec!["PING.EXE".to_string()]
                },
                &session.path_str()
            )
            .is_some()
                == (mode == InUseMode::ProcessRunning),
            "PROC_M3C_ANY: three original modes unchanged"
        );
    }
    for (mode, spelling) in [
        (InUseMode::None, "none"),
        (InUseMode::ProcessRunning, "process_running"),
        (InUseMode::CmdlineReferencesItem, "cmdline_references_item"),
        (
            InUseMode::RunsFromOrReferencesItem,
            "runs_from_or_references_item",
        ),
    ] {
        assert_eq!(
            serde_json::to_value(mode).expect("mode JSON"),
            spelling,
            "PROC_M3C_ANY: mode spelling"
        );
    }
    session.stop();
}

#[test]
fn process_privacy() {
    let mut session = Session::new();
    let worker = session.spawn_ping("jest-worker");
    session.spawn_image();
    let (_, image_pid) = session.wait_for_image();
    let orphan_signal = session.spawn_orphan();
    let (snap, _) = session.wait_for_orphan(&orphan_signal, &[worker]);
    let token = session.token.clone();
    let path = session.path_str();
    let summary_json = summary_to_json(&snap.summary());
    let snap_dbg = format!("{snap:?}");
    let hit = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::CmdlineReferencesItem,
            processes: Vec::new(),
        },
        &path,
    )
    .expect("in-use hit");
    let hit_json = serde_json::to_string(&hit).expect("InUseHit json");
    let hit_dbg = format!("{hit:?}");
    let ours = orphans(&snap, &[token.as_str()]);
    assert_eq!(ours.len(), 1, "the homemade orphan is present");
    let chains: Vec<OrphanChain> = orphans(&snap, &["jest-worker"]);
    assert!(
        !chains.is_empty(),
        "orphan chains must be non-empty before the privacy check"
    );
    let mine = chains
        .iter()
        .find(|c| c.root_pid == ours[0].root_pid)
        .expect("the homemade orphan matches jest-worker");
    assert!(
        !mine.descendant_pids.is_empty(),
        "the homemade orphan chain has its child"
    );
    let orphan_json = serde_json::to_string(&chains).expect("OrphanChain json");
    let orphan_dbg = format!("{chains:?}");
    let image_hit = in_use(
        &snap,
        &InUseSpec {
            mode: InUseMode::RunsFromOrReferencesItem,
            processes: vec!["PING.EXE".to_string()],
        },
        &path,
    )
    .expect("PROC_M3C_IMAGE: image hit for privacy");
    assert_eq!(
        image_hit.pid, image_pid,
        "PROC_M3C_PRIVACY: copied image present"
    );
    let image_hit_json = serde_json::to_string(&image_hit).expect("image hit JSON");
    let image_hit_debug = format!("{image_hit:?}");
    let summary_debug = format!("{:?}", snap.summary());
    let image_path = session
        .dir
        .join("bin")
        .join("PING.EXE")
        .to_string_lossy()
        .into_owned();
    for text in [
        &summary_json,
        &summary_debug,
        &snap_dbg,
        &hit_json,
        &hit_dbg,
        &image_hit_json,
        &image_hit_debug,
        &orphan_json,
        &orphan_dbg,
    ] {
        for secret in [&token, &path, &image_path] {
            let encoded = serde_json::to_string(secret).expect("secret JSON");
            assert!(
                !text.to_lowercase().contains(&secret.to_lowercase())
                    && !text
                        .to_lowercase()
                        .contains(&encoded[1..encoded.len() - 1].to_lowercase()),
                "PROC_M3C_PRIVACY: no token, item or executable path in output"
            );
        }
    }
    for (label, text) in [
        ("summary json", summary_json.as_str()),
        ("snapshot debug", snap_dbg.as_str()),
        ("InUseHit json", hit_json.as_str()),
        ("InUseHit debug", hit_dbg.as_str()),
        ("OrphanChain json", orphan_json.as_str()),
        ("OrphanChain debug", orphan_dbg.as_str()),
    ] {
        assert!(
            !text
                .to_ascii_lowercase()
                .contains(&token.to_ascii_lowercase()),
            "{label} leaked the test token"
        );
        assert!(
            !text
                .to_ascii_lowercase()
                .contains(&path.to_ascii_lowercase()),
            "{label} leaked the fixture path"
        );
    }
    session.stop();
}

fn regex_is_rfc3339(value: &str) -> bool {
    if value.len() != 20 {
        return false;
    }
    let bytes = value.as_bytes();
    bytes[4] == b'-'
        && bytes[7] == b'-'
        && bytes[10] == b'T'
        && bytes[13] == b':'
        && bytes[16] == b':'
        && bytes[19] == b'Z'
        && bytes.iter().enumerate().all(|(i, c)| match i {
            4 | 7 | 10 | 13 | 16 | 19 => true,
            _ => c.is_ascii_digit(),
        })
}
